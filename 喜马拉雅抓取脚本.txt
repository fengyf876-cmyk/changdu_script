  const normalizeAlbumText = (value) =>
    cleanText(value)
      .replace(/[.。…\u2026·•]/g, "")
      .replace(/[|｜]/g, "")
      .replace(/[【】\[\]《》<>〈〉「」『』()（）{}“”"'‘’：:，,、；;！!？?＿_\-—–~～]/g, "")
      .replace(/\s+/g, "")

      .trim();

  const albumTitleMatchScore = (candidateTitle, targetTitle) => {
    const candidate = normalizeAlbumText(candidateTitle);
    const target = normalizeAlbumText(targetTitle);

    if (!candidate || !target) {
      return 0;
    }

    if (candidate === target) {
      return 10000 + candidate.length;
    }

    const shorter = candidate.length <= target.length ? candidate : target;
    const longer = candidate.length > target.length ? candidate : target;

    /* 页面标题被省略号截断时，允许较长的稳定前缀匹配。 */
    if (shorter.length >= 8 && longer.startsWith(shorter)) {
      return 8500 + shorter.length;
    }

    let commonPrefix = 0;
    const maxPrefix = Math.min(candidate.length, target.length);
    while (commonPrefix < maxPrefix && candidate[commonPrefix] === target[commonPrefix]) {
      commonPrefix += 1;
    }

    if (commonPrefix >= 8 && commonPrefix / shorter.length >= 0.8) {
      return 7000 + commonPrefix;
    }

    /* 仅对足够长的文本允许包含匹配，避免“快更版”等短前缀误判。 */
    if (shorter.length >= 12 && longer.includes(shorter)) {
      return 6000 + shorter.length;
    }

    return 0;
  };

      sanitized.traffic = { ...EMPTY_TRAFFIC };

    return sanitized;

      deduped.set(retentionRecordKey(record), record);
    }

    const numberFromWeek = (value) => {
      const match = cleanText(value).match(/\d+/);
      return match ? Number(match[0]) : Number.NaN;
    };

    const sorted = Array.from(deduped.values()).sort((a, b) => {
      const ai = albumOrder.has(String(a.albumId)) ? albumOrder.get(String(a.albumId)) : Number.MAX_SAFE_INTEGER;
      const bi = albumOrder.has(String(b.albumId)) ? albumOrder.get(String(b.albumId)) : Number.MAX_SAFE_INTEGER;
      if (ai !== bi) return ai - bi;

      const aw = numberFromWeek(a.weekIndex);
      const bw = numberFromWeek(b.weekIndex);
      if (Number.isFinite(aw) && Number.isFinite(bw) && aw !== bw) return bw - aw;

      const dateCompare = String(b.startDate || "").localeCompare(String(a.startDate || ""));
      if (dateCompare) return dateCompare;
      return String(a.albumName || "").localeCompare(String(b.albumName || ""));
    });

    let previousAlbumId = null;
    return sorted.map((record) => {
      const albumId = String(record.albumId);
      const showAlbumName = albumId !== previousAlbumId;
      previousAlbumId = albumId;
      return { ...record, showAlbumName };
    });
  };

    const endDate = cleanText(record && record.endDate);
    const dateText = startDate && endDate ? `${startDate}~${endDate}` : (startDate || endDate);
    return [weekText, dateText].filter(Boolean).join(" ");
  };

      const bp = periodOrder.has(b.period) ? periodOrder.get(b.period) : Number.MAX_SAFE_INTEGER;
      return ai - bi || ap - bp || String(a.albumName || "").localeCompare(String(b.albumName || ""));
    });

  const isPanelElement = (element) => {
    if (!element || element.nodeType !== 1) {
      return false;
    }

    try {
      if (element.id && String(element.id).startsWith("xm-export-panel")) {
        return true;
      }

      return Boolean(
        element.closest && element.closest('[id^="xm-export-panel"]')
      );
    } catch (error) {
      return false;
    }
  };

  const isVisible = (element) => {
    if (!element || element.nodeType !== 1 || isPanelElement(element)) {
      return false;
    }

    try {
      const view = element.ownerDocument.defaultView || window;
      const style = view.getComputedStyle(element);

      if (
        style.display === "none" ||
        style.visibility === "hidden" ||
        Number(style.opacity) === 0
      ) {
        return false;
      }

      const rect = element.getBoundingClientRect();
      return rect.width > 0 && rect.height > 0;
    } catch (error) {
      return false;
    }
  };

  const getOwnText = (element) => {
    let text = "";

    try {
      text = Array.from(element.childNodes || [])
        .filter((node) => node.nodeType === 3)
        .map((node) => node.nodeValue || "")
        .join(" ");
    } catch (error) {
      text = "";
    }

    text = cleanText(text);

    if (!text) {
      try {
        text = cleanText(
          element.getAttribute("aria-label") ||
            element.getAttribute("title") ||
            element.getAttribute("value") ||
            ""
        );
      } catch (error) {
        text = "";
      }
    }

    return text;
  };

  const getElementText = (element) => {
    if (!element) {
      return "";
    }

    try {
      if (String(element.tagName || "").toLowerCase() === "input") {
        return cleanText(
          element.value ||
            element.getAttribute("placeholder") ||
            element.getAttribute("title") ||
            ""
        );
      }

      if (String(element.tagName || "").toLowerCase() === "select") {
        const option = element.options[element.selectedIndex];
        return cleanText(option ? option.textContent : "");
      }

      return cleanText(
        element.getAttribute("title") ||
          element.getAttribute("aria-label") ||
          element.innerText ||
          element.textContent ||
          ""
      );
    } catch (error) {
      return "";
    }
  };

  const collectRoots = () => {
    const roots = [];
    const seen = new Set();

    const visit = (root) => {
      if (!root || seen.has(root)) {
        return;
      }

      seen.add(root);
      roots.push(root);

      let elements = [];

      try {
        elements = Array.from(root.querySelectorAll("*"));
      } catch (error) {
        elements = [];
      }

      for (const element of elements) {
        if (element.shadowRoot) {
          visit(element.shadowRoot);
        }
      }
    };

    visit(document);
    return roots;
  };

  const queryAllLocal = (selector) => {
    const result = [];
    const seen = new Set();

    for (const root of collectRoots()) {
      let elements = [];

      try {
        elements = Array.from(root.querySelectorAll(selector));
      } catch (error) {
        elements = [];
      }

      for (const element of elements) {
        if (!seen.has(element)) {
          seen.add(element);
          result.push(element);
        }
      }
    }

    return result;
  };

  const getParent = (element) => {
    if (!element) {
      return null;
    }

    if (element.parentElement) {
      return element.parentElement;
    }

    try {
      const root = element.getRootNode();
      return root && root.host ? root.host : null;
    } catch (error) {
      return null;
    }
  };

  const elementContains = (container, element) => {
    if (!container || !element) {
      return false;
    }

    let current = element;

    for (let depth = 0; depth < 30 && current; depth += 1) {
      if (current === container) {
        return true;
      }
      current = getParent(current);
    }

    return false;
  };

  const buildTokens = (container = null) => {
    const tokens = [];

    for (const element of queryAllLocal("*")) {
      if (!isVisible(element)) {
        continue;
      }

      if (container && !elementContains(container, element)) {
        continue;
      }

      const text = getOwnText(element);

      if (!text || text.length > 120) {
        continue;
      }

      let rect;
      let fontSize = 0;
      let fontWeight = 400;

      try {
        rect = element.getBoundingClientRect();
        const style = (element.ownerDocument.defaultView || window).getComputedStyle(
          element
        );
        fontSize = parseFloat(style.fontSize) || 0;
        fontWeight = parseInt(style.fontWeight, 10) || 400;
      } catch (error) {
        continue;
      }

      const parts = text.split(/\n+/).map(cleanText).filter(Boolean);

      for (const part of parts) {
        tokens.push({
          element,
          text: part,
          normalized: normalizeText(part),
          rect,
          fontSize,
          fontWeight,
        });
      }
    }

    return tokens;
  };

  const findSectionRoot = (titleAliases, requiredWords = []) => {
    const aliases = titleAliases.map(normalizeText);
    const candidates = [];

    for (const element of queryAllLocal("div,section,article,main,h1,h2,h3,h4,p,span")) {
      if (!isVisible(element)) {
        continue;
      }

      const own = normalizeText(getOwnText(element));
      if (!aliases.includes(own)) {
        continue;
      }

      let current = element;

      for (let depth = 0; depth < 12 && current; depth += 1) {
        try {
          const rect = current.getBoundingClientRect();
          const text = normalizeText(
            current.innerText || current.textContent || ""
          );
          const wordCount = requiredWords.filter((word) =>
            text.includes(normalizeText(word))
          ).length;

          if (
            rect.width >= 300 &&
            rect.height >= 100 &&
            rect.height <= 1200 &&
            (!requiredWords.length || wordCount >= Math.min(2, requiredWords.length))
          ) {
            candidates.push({
              element: current,
              score:
                rect.width * rect.height +
                depth * 5000 -
                wordCount * 120000,
            });
          }
        } catch (error) {
          // 继续向上查找。
        }

        current = getParent(current);
      }
    }

    candidates.sort((a, b) => a.score - b.score);
    return candidates.length ? candidates[0].element : null;
  };

  const findClickableAncestor = (element) => {
    let current = element;

    for (let depth = 0; depth < 5 && current; depth += 1) {
      const tag = String(current.tagName || "").toLowerCase();
      const role = cleanText(current.getAttribute && current.getAttribute("role"));

      if (
        ["button", "a", "label", "li"].includes(tag) ||
        ["button", "tab", "option"].includes(role) ||
        typeof current.onclick === "function"
      ) {
        return current;
      }

      current = getParent(current);
    }

    return element;

  const dispatchMouseLikeEvent = (target, type, clientX, clientY, view) => {
    const common = {
      bubbles: true,
      cancelable: true,
      composed: true,
      view,
      clientX,
      clientY,
      screenX: clientX,
      screenY: clientY,
      button: 0,
      buttons: type === "mouseup" || type === "pointerup" || type === "click" ? 0 : 1,
    };

    if (type.startsWith("pointer") && typeof view.PointerEvent === "function") {
      target.dispatchEvent(
        new view.PointerEvent(type, {
          ...common,
          pointerId: 1,
          pointerType: "mouse",
          isPrimary: true,
        })
      );
      return;
    }

    target.dispatchEvent(new view.MouseEvent(type, common));
  };

  const clickElement = (element) => {
    if (!element) {
      return false;
    }

    let target = findClickableAncestor(element) || element;
    const doc = target.ownerDocument || document;
    const view = doc.defaultView || window;

    try {
      target.scrollIntoView({ block: "center", inline: "nearest", behavior: "auto" });
    } catch (error) {
      // 忽略。
    }

    let rect = null;
    try {
      rect = target.getBoundingClientRect();
      const x = Math.max(rect.left + 2, Math.min(rect.right - 2, rect.left + rect.width / 2));
      const y = Math.max(rect.top + 2, Math.min(rect.bottom - 2, rect.top + rect.height / 2));
      const hit = doc.elementFromPoint ? doc.elementFromPoint(x, y) : null;

      /* 虚拟列表常把点击事件绑在最上层命中节点或其行容器上。 */
      if (hit && (elementContains(target, hit) || elementContains(element, hit))) {
        target = findClickableAncestor(hit) || hit;
        rect = target.getBoundingClientRect();
      }
    } catch (error) {
      rect = null;
    }

    try {
      if (typeof target.focus === "function") {
        target.focus({ preventScroll: true });
      }
    } catch (error) {
      // 忽略。
    }

    try {
      const clientX = rect ? rect.left + rect.width / 2 : 0;
      const clientY = rect ? rect.top + rect.height / 2 : 0;

      ["mouseover", "mouseenter", "mousemove"].forEach((type) => {
        dispatchMouseLikeEvent(target, type, clientX, clientY, view);
      });
      dispatchMouseLikeEvent(target, "pointerdown", clientX, clientY, view);
      dispatchMouseLikeEvent(target, "mousedown", clientX, clientY, view);
      dispatchMouseLikeEvent(target, "pointerup", clientX, clientY, view);
      dispatchMouseLikeEvent(target, "mouseup", clientX, clientY, view);

      /* 只触发一次 click，避免双击导致下拉框刚选中又重新打开。 */
      if (typeof target.click === "function") {
        target.click();
      } else {
        dispatchMouseLikeEvent(target, "click", clientX, clientY, view);
      }

      return true;
    } catch (error) {
      console.warn("[xm-export] 模拟点击失败", error, target);
      return false;
    }
  };

        descriptor.set.call(input, value);

      input.dispatchEvent(new Event("input", { bubbles: true }));
      input.dispatchEvent(new Event("change", { bubbles: true }));
      return true;
    } catch (error) {
      return false;
    }
  };

  const parseCardLines = (rawText, label) => {
    const lines = String(rawText == null ? "" : rawText)
      .replace(/\u00a0/g, " ")
      .split(/\n+/)
      .map(cleanText)
      .filter(Boolean);

    for (let i = 0; i < lines.length; i += 1) {
      if (!normalizeText(lines[i]).includes(normalizeText(label))) {
        continue;
      }

      let value = "";
      let comparison = "";

      for (let j = i + 1; j < Math.min(lines.length, i + 15); j += 1) {
        const line = cleanText(lines[j]).replace(/[?？ⓘℹ]/g, "").trim();

        if (!line) {
          continue;
        }

        const otherMetric = CORE_METRICS.some(
          (name) =>
            name !== label &&
            normalizeText(line).includes(normalizeText(name))
        );

        if (otherMetric) {
          break;
        }

        if (line === "环比") {
          const next = cleanText(lines[j + 1] || "");
          if (isValue(next)) {
            comparison = next;
          }
          continue;
        }

        if (line.startsWith("环比")) {
          const tail = cleanText(line.replace(/^环比\s*/, ""));
          if (isValue(tail)) {
            comparison = tail;
          }
          continue;
        }

        if (!value && isValue(line)) {
          value = line;
        }
      }

      if (value) {
        return { value, comparison };
      }
    }

    return null;
  };

  const parseCardCompact = (rawText, label) => {
    const text = String(rawText == null ? "" : rawText)
      .replace(/\u00a0/g, " ")
      .replace(/[?？ⓘℹ]/g, " ")
      .replace(/\s+/g, " ")
      .trim();

    const start = text.indexOf(label);
    if (start < 0) {
      return null;
    }

    let end = Math.min(text.length, start + 360);

    for (const other of CORE_METRICS) {
      if (other === label) {
        continue;
      }

      const index = text.indexOf(other, start + label.length);
      if (index >= 0 && index < end) {
        end = index;
      }
    }

    const segment = text.slice(start, end);
    const afterLabel = segment.slice(label.length);
    const valueMatch = afterLabel.match(
      new RegExp(`^[^\\d+\\-—]{0,90}(${VALUE_SOURCE})`)
    );
    const comparisonMatch = segment.match(
      new RegExp(`环比[^\\d+\\-—]{0,40}(${VALUE_SOURCE})`)
    );

    if (!valueMatch) {
      return null;
    }

    return {
      value: valueMatch[1],
      comparison: comparisonMatch ? comparisonMatch[1] : "",
    };
  };

  const readMetricFromTokens = (label, tokens) => {
    const target = normalizeText(label);
    const labels = tokens.filter(
      (token) =>
        token.normalized === target ||
        (token.normalized.startsWith(target) &&
          token.normalized.length <= target.length + 2)
    );

    let best = null;

    for (const labelToken of labels) {
      const card = findMetricCard(labelToken, label);
      if (!card) {
        continue;
      }

      const lineResult = parseCardLines(card.text, label);
      const compactResult = parseCardCompact(card.text, label);
      const parsed = lineResult || compactResult;

      const inCard = tokens.filter((token) =>
        elementContains(card.element, token.element)
      );
      const ringTokens = inCard.filter((token) => token.text.includes("环比"));
      const ringY = ringTokens.length
        ? Math.min(...ringTokens.map((token) => token.rect.top))
        : Number.POSITIVE_INFINITY;

      const valueCandidates = inCard.filter((token) => {
        if (!isValue(token.text)) {
          return false;
        }
        if (token.element === labelToken.element) {
          return false;
        }
        if (token.rect.top < labelToken.rect.top - 5) {
          return false;
        }
        if (token.rect.top > ringY + 3) {
          return false;
        }
        return true;
      });

      valueCandidates.sort((a, b) => {
        const score = (token) => {
          let result = 0;
          result += Math.abs(token.rect.top - labelToken.rect.top) * 4;
          result += Math.abs(token.rect.left - labelToken.rect.left) * 0.25;
          result -= token.fontSize * 18;
          result -= token.fontWeight >= 600 ? 30 : 0;

          if (label.includes("率") && token.text.includes("%")) {
            result -= 120;
          }
          if (!label.includes("率") && token.text.includes("%")) {
            result += 180;
          }
          return result;
        };

        return score(a) - score(b);
      });

      let value = parsed ? parsed.value : "";
      if (valueCandidates.length) {
        value = valueCandidates[0].text;
      }

      if (!value) {
        continue;
      }

      const score =
        card.score +
        (lineResult ? -100000 : 0) +
        (valueCandidates.length ? -70000 : 0);

      if (!best || score < best.score) {
        best = {
          label,
          value,
          comparison: parsed ? parsed.comparison : "",
          score,
          source: "DOM卡片",
          sourceText: card.text.slice(0, 500),
        };
      }
    }

    return best;
  };

  const parseTrafficFromText = (rawText) => {

    /*
     * 页面实际排列通常为：

    const beforeLabel = Array.from(

        playFinish: beforeLabel[2] || "",
        score: 0,
        source: "转化率文字顺序",
      };
    }

    const afterLabel = Array.from(
      text.matchAll(/转化率\s*([+\-]?\d[\d,]*(?:\.\d+)?%)/g)
    ).map((match) => match[1]);

    if (afterLabel.length >= 3) {

        playFinish: afterLabel[2] || "",

  const readTrafficByPosition = (container = null) => {
    const tokens = buildTokens(container);
    const labels = ["曝光", "点击", "播放", "完播"];
    const labelTokens = labels.map((label) => {
      const normalized = normalizeText(label);
      return tokens
        .filter((token) => token.normalized === normalized)
        .sort((a, b) => a.rect.left - b.rect.left || a.rect.top - b.rect.top)[0] || null;
    });

    if (labelTokens.filter(Boolean).length < 4) {
      return null;
    }

    const orderedLabels = labelTokens.slice().sort(
      (a, b) => a.rect.left - b.rect.left || a.rect.top - b.rect.top
    );

    const rateTokens = tokens.filter((token) => RATE_RE.test(cleanText(token.text)));
    const result = [];

    for (let index = 0; index < 3; index += 1) {
      const leftLabel = orderedLabels[index];
      const rightLabel = orderedLabels[index + 1];
      const leftCenter = leftLabel.rect.left + leftLabel.rect.width / 2;
      const rightCenter = rightLabel.rect.left + rightLabel.rect.width / 2;
      const top = Math.min(leftLabel.rect.top, rightLabel.rect.top) - 140;
      const bottom = Math.max(leftLabel.rect.bottom, rightLabel.rect.bottom) + 30;

      const candidates = rateTokens
        .filter((token) => {
          const center = token.rect.left + token.rect.width / 2;
          return (
            center > leftCenter &&
            center < rightCenter &&
            token.rect.top >= top &&
            token.rect.top <= bottom

  const verticalDistanceToRect = (rect, anchorRect) => {
    if (!rect || !anchorRect) return Number.POSITIVE_INFINITY;
    const center = rect.top + rect.height / 2;
    const anchorCenter = anchorRect.top + anchorRect.height / 2;
    return Math.abs(center - anchorCenter);

   * v2.4.1 核心修复：周期按钮必须属于目标板块。
   * 旧版虽然找到了“核心数据”板块，但随后仍扫描全页面，并用 active 状态大幅加权，
   * 因此“流量转化”里已经激活的近7天可能压过真正的“核心数据-近7天”。
   * 新版先做硬性板块隔离，再评分；跨板块按钮不再有资格成为候选。
   */
  const periodElementBelongsToSection = (
    sectionType,
    element,
    section,
    oppositeSection,
    targetAnchor,
    oppositeAnchor
  ) => {
    if (!section || !element || !elementContains(section, element)) {
      return false;
    }

    const insideOpposite = Boolean(
      oppositeSection && elementContains(oppositeSection, element)
    );

    if (!insideOpposite) {
      return true;
    }

    /*
     * 某些页面结构中两个 findSectionRoot 可能找到有重叠的祖先容器。
     * 这时不能仅靠 contains，要比较候选按钮离两个板块标题哪个更近。
     */
    const rect = getRectSafe(element);
    const targetRect = targetAnchor && targetAnchor.rect;
    const oppositeRect = oppositeAnchor && oppositeAnchor.rect;

    if (rect && targetRect && oppositeRect) {
      const targetDistance = verticalDistanceToRect(rect, targetRect);
      const oppositeDistance = verticalDistanceToRect(rect, oppositeRect);
      return targetDistance + 12 < oppositeDistance;
    }

    /*
     * 如果对方板块只是目标板块外层的宽泛祖先，仍允许目标板块内部元素；
     * 反之若目标板块把对方板块包进来了，则拒绝对方板块内部候选。
     */
    if (oppositeSection && elementContains(oppositeSection, section)) {
      return true;
    }

    return false;
  };

  const collectPeriodCandidatesLocal = (sectionType, aliases) => {
    const section = getPeriodSection(sectionType);
    const oppositeSection = getOppositePeriodSection(sectionType);
    const targetAnchor = findPeriodSectionAnchor(sectionType, section);
    const oppositeAnchor = findPeriodSectionAnchor(
      sectionType === "traffic" ? "core" : "traffic",
      oppositeSection
    );
    const normalizedAliases = aliases.map(normalizeText).filter(Boolean);
    const candidates = [];

    if (!section) {
      return {
        section: null,
        oppositeSection,
        targetAnchor,
        oppositeAnchor,
        candidates,
        strictSectionFound: false,
      };
    }

    for (const element of queryAllLocal(
      "button,[role='tab'],[role='radio'],[role='menuitem'],[role='option'],div,span,label,li,a"
    )) {
      if (!isVisible(element) || isPanelElement(element)) continue;

      const ownText = normalizeText(getOwnText(element));
      const fullText = normalizeText(getElementText(element));
      const matchedAlias = normalizedAliases.find((alias) =>
        ownText === alias ||
        fullText === alias ||
        (ownText.startsWith(alias) && ownText.length <= alias.length + 4) ||
        (fullText.startsWith(alias) && fullText.length <= alias.length + 4)
      );
      if (!matchedAlias) continue;

      if (!periodElementBelongsToSection(
        sectionType,
        element,
        section,
        oppositeSection,
        targetAnchor,
        oppositeAnchor
      )) {
        continue;
      }

      const rect = getRectSafe(element);
      if (!rect) continue;
      if (rect.width < 18 || rect.height < 10 || rect.width > 700 || rect.height > 150) {
        continue;
      }

      const clickTarget = findClickableAncestor(element) || element;
      if (!periodElementBelongsToSection(
        sectionType,
        clickTarget,
        section,
        oppositeSection,
        targetAnchor,
        oppositeAnchor
      )) {
        continue;
      }

      const active = isActiveTab(clickTarget) || isActiveTab(element);
      let score = rect.width * rect.height;

      // 到板块标题越近越优先；active 只在“已严格确认属于本板块”以后参与排序。
      if (targetAnchor && targetAnchor.rect) {
        score += verticalDistanceToRect(rect, targetAnchor.rect) * 80;
      }
      if (active) score -= 180000;
      if (String(element.getAttribute && element.getAttribute("role")) === "tab") score -= 80000;
      if (String(element.tagName || "").toLowerCase() === "button") score -= 50000;

      candidates.push({
        element,
        clickTarget,
        active,
        score,
        text: cleanText(getOwnText(element) || getElementText(element)),
        matchedAlias,
        strictSection: true,
        sectionType,
      });
    }

    candidates.sort((a, b) => a.score - b.score);
    return {
      section,
      oppositeSection,
      targetAnchor,
      oppositeAnchor,
      candidates,
      strictSectionFound: true,
    };
  };

        ) {
          continue;
        }
      } catch (error) {
        // 忽略选择器判断失败。
      }

      const variants = getAlbumTextVariants(element);
      if (!variants.length) {
        continue;
      }

      const value = getAlbumElementValue(element);
      const exactId = Boolean(targetId && value && targetId === String(value));
      const titleScore = Math.max(
        0,
        ...variants.map((candidateTitle) =>
          albumTitleMatchScore(candidateTitle, album.title)
        )
      );

      if (!exactId && titleScore <= 0) {
        continue;
      }

      const combinedText = variants.join(" ");
      if (
        combinedText.includes("搜索你想找的专辑") ||
        normalizeText(combinedText) === "加载更多"
      ) {
        continue;
      }

      const clickTarget = resolveAlbumOptionClickTarget(element, album);
      if (!clickTarget || seen.has(clickTarget)) {
        continue;
      }
      if (dropdownRoot && !elementContains(dropdownRoot, clickTarget)) {
        continue;
      }
      try {
        const targetRect = clickTarget.getBoundingClientRect();
        if (dropdownInputRect && targetRect.top < dropdownInputRect.bottom - 2) {
          continue;
        }
      } catch (error) {
        // 忽略。
      }
      seen.add(clickTarget);

      let clickRect = rect;
      try {
        clickRect = clickTarget.getBoundingClientRect();
      } catch (error) {
        // 使用原 rect。
      }

      candidates.push({
        element: clickTarget,
        exactId,
        titleScore,
        area: clickRect.width * clickRect.height,
        top: clickRect.top,
        text: variants[0],
        value,
      });
    }

    /* 找不到下拉根节点时，再做一次全页面兜底扫描。 */
    if (!candidates.length && dropdownRoot) {
      const originalRoot = dropdownRoot;
      try {
        originalRoot.setAttribute("data-xm-scan-fallback", "1");
      } catch (error) {
        // 忽略。
      }

      for (const element of queryAllLocal("[title],[aria-label],li,div,span")) {
        if (!isVisible(element) || isPanelElement(element)) {
          continue;
        }

        let rect;
        try {
          rect = element.getBoundingClientRect();
        } catch (error) {
          continue;
        }

        if (rect.height < 18 || rect.height > 100 || rect.width < 120) {
          continue;
        }

        if (dropdownInputRect && rect.top < dropdownInputRect.bottom - 2) {
          continue;
        }

        try {
          if (
            element.matches(".ant-select-selection-item,[class*='selection-item']") ||
            element.closest(".ant-select-selector") ||
            (element.closest("[role='combobox']") && !element.closest("[role='option']"))
          ) {
            continue;
          }
        } catch (error) {
          // 忽略选择器判断失败。
        }

        const variants = getAlbumTextVariants(element);
        const titleScore = Math.max(
          0,
          ...variants.map((candidateTitle) =>
            albumTitleMatchScore(candidateTitle, album.title)
          )
        );

        if (titleScore <= 0) {
          continue;
        }

        const clickTarget = resolveAlbumOptionClickTarget(element, album);
        if (!clickTarget || seen.has(clickTarget)) {
          continue;
        }
        if (dropdownRoot && !elementContains(dropdownRoot, clickTarget)) {
          continue;
        }
        try {
          const targetRect = clickTarget.getBoundingClientRect();
          if (dropdownInputRect && targetRect.top < dropdownInputRect.bottom - 2) {
            continue;
          }
        } catch (error) {
          // 忽略。
        }
        seen.add(clickTarget);

        candidates.push({
          element: clickTarget,
          exactId: false,
          titleScore,
          area: rect.width * rect.height,
          top: rect.top,
          text: variants[0] || "",
          value: "",
        });
      }
    }

    candidates.sort(
      (a, b) =>
        Number(b.exactId) - Number(a.exactId) ||
        b.titleScore - a.titleScore ||
        a.area - b.area ||
        a.top - b.top
    );

    if (candidates.length) {
      console.log("[xm-export] 找到目标专辑选项", {
        album,
        candidate: candidates[0],
        candidateCount: candidates.length,
      });
    }

    return candidates.length ? candidates[0].element : null;
  };

  const findAlbumSearchInput = (trigger = null) => {
    const candidates = [];
    const triggerRect = trigger && trigger.getBoundingClientRect
      ? trigger.getBoundingClientRect()
      : null;

    for (const input of queryAllLocal(
      [
        "input[placeholder*='搜索你想找的专辑']",
        "input[placeholder*='专辑名称']",
        "input[placeholder*='专辑名']",
        "input[placeholder*='专辑']",
        "input[role='combobox']",
        ".ant-select-selection-search-input",
        "input[aria-controls]",
      ].join(",")
    )) {
      if (!isVisible(input) || isPanelElement(input)) {
        continue;
      }

      let rect;
      try {
        rect = input.getBoundingClientRect();
      } catch (error) {
        continue;
      }

      const placeholder = normalizeText(
        input.getAttribute("placeholder") ||
          input.getAttribute("aria-label") ||
          input.getAttribute("title") ||
          ""
      );

      let score = 0;
      if (placeholder.includes("搜索") && placeholder.includes("专辑")) {
        score -= 5000;
      } else if (placeholder.includes("专辑")) {
        score -= 2500;
      }
      if (input.getAttribute("role") === "combobox") {
        score -= 500;
      }
      if (String(input.className || "").includes("selection-search")) {
        score -= 400;
      }
      if (triggerRect) {
        score += Math.abs(rect.left - triggerRect.left) * 0.2;
        score += Math.abs(rect.top - triggerRect.bottom) * 0.15;
      }

      candidates.push({ input, score });
    }

    candidates.sort((a, b) => a.score - b.score);
    return candidates.length ? candidates[0].input : null;
  };

    input.dispatchEvent(

    if (

        "近7日",
        "近7天",

        "近365日",
        "近365天",
        "最近1年",

        "累计",

        "请选择专辑",
        "专辑",
      ].includes(text)
    ) {
      return false;
    }

    if (/^\d{4}-\d{2}-\d{2}/.test(text)) {
      return false;
    }

    if (text.includes("正在加载") || text.includes("手动输入albumId")) {
      return false;

    return true;

  const getLocalCurrentAlbumIdentities = () => {
    const candidates = [];
    const selector = [
      ".ant-select-selection-item",
      "[class*='selection-item']",
      ".ant-select-selector",
      "select",
      "[role='combobox']",
    ].join(",");

    for (const element of queryAllLocal(selector)) {
      if (!isVisible(element) || isPanelElement(element)) {
        continue;
      }

      let title = "";
      let id = "";
      let score = 0;
      const tag = String(element.tagName || "").toLowerCase();
      const className = String(element.className || "");

      if (tag === "select") {
        const option = element.options && element.options[element.selectedIndex];
        title = cleanText(option ? option.textContent : "");
        id = cleanText(option ? option.value : "");
        score -= 700;
      } else {
        let selectedItem = null;
        try {
          selectedItem = element.matches(".ant-select-selection-item,[class*='selection-item']")
            ? element
            : element.querySelector(
                ".ant-select-selection-item,[class*='selection-item']"
              );
        } catch (error) {
          selectedItem = null;
        }

        const source = selectedItem || element;
        title = cleanText(
          source.getAttribute("title") ||
            source.getAttribute("aria-label") ||
            source.innerText ||
            source.textContent ||
            ""
        );
        id = cleanText(
          source.getAttribute("data-value") ||
            source.getAttribute("value") ||
            element.getAttribute("data-value") ||
            ""
        );

        if (selectedItem) {
          score -= 900;
        }
        if (className.includes("ant-select")) {
          score -= 350;
        }
        if (element.getAttribute("role") === "combobox") {
          score -= 180;
        }
      }

      if (!isPossibleAlbumTitle(title)) {
        continue;
      }

      let contextText = "";
      let parent = element;
      for (let depth = 0; depth < 4 && parent; depth += 1) {
        contextText += normalizeText(getOwnText(parent) || "");
        parent = getParent(parent);
      }

      if (contextText.includes("专辑")) {
        score -= 1200;
      }

      let rect;
      try {
        rect = element.getBoundingClientRect();
        score += Math.abs(rect.top - 120) * 0.05;
      } catch (error) {
        rect = { top: 0 };
      }

      candidates.push({
        title,
        normalizedTitle: normalizeAlbumText(title),
        id,
        score,
        source: `${tag || "node"}.${className}`,
      });
    }

    candidates.sort((a, b) => a.score - b.score);

    const unique = [];
    const seen = new Set();

    for (const item of candidates) {
      const key = `${String(item.id || "")}|${normalizeAlbumText(item.title || "")}`;
      if (!key || seen.has(key)) {
        continue;
      }
      seen.add(key);
      unique.push(item);
    }

    return unique.slice(0, 20);

  const albumIdentityMatches = (identity, album) => {
    if (!identity || !album) {
      return false;
    }

    const targetId = String(album.id || "");
    const currentId = String(identity.id || "");
    const currentTitle = identity.title || identity.normalizedTitle || "";

    if (targetId && currentId && targetId === currentId) {
      return true;
    }

    return Boolean(
      album.title &&
        currentTitle &&
        albumTitlesMatch(currentTitle, album.title)
    );
  };

  const albumIdentitiesMatch = (left, right) => {
    if (!left || !right) {
      return false;
    }

    const leftId = String(left.id || "");
    const rightId = String(right.id || "");

    if (leftId && rightId && leftId === rightId) {
      return true;
    }

    return albumTitlesMatch(
      left.title || left.normalizedTitle || "",
      right.title || right.normalizedTitle || ""
    );
  };

  const localSelectedAlbumMatches = (album) =>
    albumIdentityMatches(getLocalCurrentAlbumIdentity(), album);

  const waitForLocalAlbumSelection = async (album, beforeIdentity, timeoutMs = 15000) => {
    const startedAt = Date.now();
    let lastIdentity = getLocalCurrentAlbumIdentity();
    let lastCandidates = getLocalCurrentAlbumIdentities();

    while (Date.now() - startedAt < timeoutMs) {
      lastCandidates = getLocalCurrentAlbumIdentities();
      const matched = lastCandidates.find((item) => albumIdentityMatches(item, album));
      lastIdentity = matched || lastCandidates[0] || getLocalCurrentAlbumIdentity();

      if (matched) {
        const targetWasAlreadySelected = albumIdentityMatches(beforeIdentity, album);
        const changedFromBefore = !albumIdentitiesMatch(matched, beforeIdentity);

        if (targetWasAlreadySelected || changedFromBefore || !beforeIdentity.title) {
          return {
            success: true,
            current: matched,
            candidates: lastCandidates,
          };
        }
      }

      await sleep(160);
    }

    return {
      success: false,
      current: lastIdentity,
      candidates: lastCandidates,
    };
  };

  const selectAlbumLocal = async (album) => {
    if (!album || !album.id) {
      return { success: false, reason: "专辑参数为空" };
    }

    const beforeIdentity = getLocalCurrentAlbumIdentity();

    if (albumIdentityMatches(beforeIdentity, album)) {
      return {
        success: true,
        alreadySelected: true,
        before: beforeIdentity,
        current: beforeIdentity,
      };
    }

    await scrollLocalPageToTop();

    const targetId = String(album.id);
    const selectors = getLocalAlbumSelectorCandidates(album);
    const debugAttempts = [];

    for (const candidate of selectors.slice(0, 20)) {
      const element = candidate.element;
      const tag = String(element.tagName || "").toLowerCase();

      if (tag === "select") {
        const options = Array.from(element.options || []);
        const match = options.find(
          (option) =>
            String(option.value) === targetId ||
            albumTitlesMatch(option.textContent, album.title)
        );

        if (match) {
          element.value = match.value;
          element.dispatchEvent(new Event("input", { bubbles: true }));
          element.dispatchEvent(new Event("change", { bubbles: true }));

          const waited = await waitForLocalAlbumSelection(album, beforeIdentity);
          if (waited.success) {
            return {
              success: true,
              method: "native-select",
              before: beforeIdentity,
              current: waited.current,
            };
          }
        }

        continue;
      }

      const trigger =
        element.closest && element.closest(".ant-select-selector")
          ? element.closest(".ant-select-selector")
          : element.closest && element.closest("[role='combobox']")
            ? element.closest("[role='combobox']")
            : element;

      debugAttempts.push({
        selectorText: candidate.text,
        selectorValue: candidate.value,
        selectorScore: candidate.score,
      });

      clickElement(trigger);
      await sleep(350);

      /* 第一优先：目标专辑已在当前展开列表中，直接点击。 */
      let option = findVisibleAlbumOption(album);
      if (option) {
        clickElement(option);
        const waited = await waitForLocalAlbumSelection(album, beforeIdentity);

        if (waited.success) {
          return {
            success: true,
            method: "visible-option-fuzzy",
            before: beforeIdentity,
            current: waited.current,
          };
        }

        /* 点击后未切换成功，重新打开下拉框，再尝试搜索。 */
        clickElement(trigger);
        await sleep(280);
      }

      const input = findAlbumSearchInput(trigger);

      if (input) {
        /* 搜索框明确支持名称或 ID，先用 ID 可避免标题截断和符号差异。 */
        const terms = [targetId, album.title]
          .map(cleanText)
          .filter((value, index, list) => value && list.indexOf(value) === index);

        for (const term of terms) {
          await typeAlbumSearchTerm(input, term);
          option = findVisibleAlbumOption(album);

          if (!option) {
            continue;
          }

          clickElement(option);
          const waited = await waitForLocalAlbumSelection(album, beforeIdentity);

          if (waited.success) {
            return {
              success: true,
              method: term === targetId ? "search-by-id" : "search-by-title",
              searchTerm: term,
              before: beforeIdentity,
              current: waited.current,
            };
          }

          /* 搜索结果被点击但页面未变，重新打开下拉框继续下一种搜索。 */
          clickElement(trigger);
          await sleep(280);
        }
      }

      try {
        document.dispatchEvent(
          new KeyboardEvent("keydown", {
            key: "Escape",
            code: "Escape",
            bubbles: true,
          })
        );
      } catch (error) {
        // 忽略。
      }

      await sleep(250);
    }

    const currentIdentity = getLocalCurrentAlbumIdentity();

    return {
      success: false,
      reason: `页面专辑下拉框中未能选中：${album.title}`,
      before: beforeIdentity,
      current: currentIdentity,
      selectorCount: selectors.length,
      selectorTexts: selectors.slice(0, 10).map((item) => item.text || "(无文本)"),
      debugAttempts,
      dropdownRootFound: Boolean(findAlbumDropdownRoot()),
      searchInputFound: Boolean(findAlbumSearchInput()),
    };
  };

  const forwardToChildren = (message) => {
    let count = 0;

    try {
      for (let i = 0; i < window.frames.length; i += 1) {
        try {
          window.frames[i].postMessage(message, "*");
          count += 1;
        } catch (error) {
          // 忽略不可访问的框架。
        }
      }
    } catch (error) {
      // 忽略。
    }

    return count;
  };

  const executeLocalCommand = async (command, payload) => {
    switch (command) {
      case "SCAN_CORE": {
        const scan = scanLocalCoreMetrics();
        const aliases = Array.isArray(payload.aliases) ? payload.aliases : [];
        const periodState = aliases.length
          ? getPeriodActiveStateLocal("core", aliases)
          : { verified: true, active: true, activeText: "", sectionFound: true };
        return {
          ...scan,
          periodVerified: Boolean(periodState.verified),
          periodActiveText: periodState.activeText || "",
          periodSectionFound: Boolean(periodState.sectionFound),
        };
      }

      case "SCAN_TRAFFIC": {
        const scan = scanLocalTraffic();
        const aliases = Array.isArray(payload.aliases) ? payload.aliases : [];
        const periodState = aliases.length
          ? getPeriodActiveStateLocal("traffic", aliases)
          : { verified: true, active: true, activeText: "", sectionFound: true };
        return {
          ...scan,
          periodVerified: Boolean(periodState.verified),
          periodActiveText: periodState.activeText || "",
          periodSectionFound: Boolean(periodState.sectionFound),
        };
      }

      case "SELECT_CORE_PERIOD":
        return clickPeriodLocal("core", payload.aliases || []);

      case "SELECT_TRAFFIC_PERIOD":
        return clickPeriodLocal("traffic", payload.aliases || []);

      case "SELECT_ALBUM":
        return selectAlbumLocal(payload.album);

      case "CHECK_ALBUM": {
        const current = getLocalCurrentAlbumIdentity();
        return {
          success: albumIdentityMatches(current, payload.album),
          current,
        };
      }

      case "GET_CURRENT_ALBUM": {
        const currents = getLocalCurrentAlbumIdentities();
        return {
          success: true,
          current: currents[0] || getLocalCurrentAlbumIdentity(),
          currents,
        };
      }

      default:
        return {
          success: false,
          reason: `未知命令：${command}`,
        };
    }
  };

  window.addEventListener("message", (event) => {
    const data = event.data;

    if (!data || data.flag !== MSG_FLAG) {
      return;
    }

    if (data.type === MSG_COMMAND) {
      (async () => {
        let result;

        try {
          result = await executeLocalCommand(data.command, data.payload || {});
        } catch (error) {
          result = {
            success: false,
            error: error.message,
          };
        }

        try {
          event.source.postMessage(
            {
              flag: MSG_FLAG,
              type: MSG_RESULT,
              requestId: data.requestId,
              command: data.command,
              payload: {
                ...result,
                frameUrl: location.href,
                frameTitle: document.title || "",
              },
            },
            "*"
          );
        } catch (error) {
          // 忽略。
        }

        forwardToChildren(data);
      })();

      return;
    }

    if (data.type === MSG_RESULT && window.self !== window.top) {
      try {
        window.parent.postMessage(data, "*");
      } catch (error) {
        // 忽略。
      }
    }
  });

  if (window.self !== window.top) {
    return;

  const executeAcrossFrames = async (command, payload = {}, options = {}) => {
    const {
      minWaitMs = 220,
      quietWaitMs = 160,
      maxWaitMs = 1400,
      stopWhen = null,
    } = options;
    const requestId = `${Date.now()}_${Math.random().toString(36).slice(2)}`;
    const results = [];
    let lastChangeAt = Date.now();
    let localFinished = false;

    const pushResult = (payloadValue) => {
      if (!payloadValue) return;
      results.push(payloadValue);
      lastChangeAt = Date.now();
    };

    const listener = (event) => {
      const data = event.data;
      if (!data || data.flag !== MSG_FLAG || data.type !== MSG_RESULT ||
          data.requestId !== requestId || data.command !== command) return;
      pushResult(data.payload);
    };

    window.addEventListener("message", listener);

    // 先通知 iframe，再并行执行顶层页面，避免原版先等顶层执行完才开始 iframe。
    forwardToChildren({ flag: MSG_FLAG, type: MSG_COMMAND, requestId, command, payload });

    Promise.resolve()
      .then(() => executeLocalCommand(command, payload))
      .then((localResult) => pushResult({
        ...localResult,
        frameUrl: location.href,
        frameTitle: document.title || "",
      }))
      .catch((error) => pushResult({
        success: false,
        error: error.message,
        frameUrl: location.href,
      }))
      .finally(() => { localFinished = true; });

    const startedAt = Date.now();
    try {
      while (Date.now() - startedAt < maxWaitMs) {
        const elapsed = Date.now() - startedAt;
        if (elapsed >= minWaitMs && typeof stopWhen === "function" && stopWhen(results)) break;
        if (elapsed >= minWaitMs && localFinished && Date.now() - lastChangeAt >= quietWaitMs) break;
        await sleep(45);
      }
    } finally {
      window.removeEventListener("message", listener);
    }
    return results;
  };

  const mergeCoreResults = (sources) => CORE_METRICS.map((label) => {
    const candidates = [];
    for (const source of sources) {
      if (!source || !Array.isArray(source.metrics)) continue;
      const item = source.metrics.find((metric) => metric.label === label && metric.value !== "");
      if (item) candidates.push({ ...item, frameUrl: source.frameUrl || source.url || "" });
    }
    candidates.sort((a, b) => (Number.isFinite(a.score) ? a.score : Number.MAX_SAFE_INTEGER) -
      (Number.isFinite(b.score) ? b.score : Number.MAX_SAFE_INTEGER));
    return candidates[0] || { label, value: "", comparison: "", score: Number.POSITIVE_INFINITY };
  });

  const readCoreAcrossFrames = async (expectedAliases = []) => {
    const aliases = Array.isArray(expectedAliases) ? expectedAliases.filter(Boolean) : [];
    const requirePeriodVerification = aliases.length > 0;
    const sources = await executeAcrossFrames("SCAN_CORE", { aliases }, {
      minWaitMs: 180,
      quietWaitMs: 130,
      maxWaitMs: 1100,
      stopWhen: (items) => items.some((item) =>
        item &&
        (!requirePeriodVerification || item.periodVerified) &&
        Array.isArray(item.metrics) &&
        item.metrics.filter((m) => m.value !== "").length === CORE_METRICS.length
      ),
    });

    const usableSources = requirePeriodVerification
      ? sources.filter((item) => item && item.periodVerified)
      : sources;

    if (requirePeriodVerification && !usableSources.length) {
      return CORE_METRICS.map((label) => ({
        label,
        value: "",
        comparison: "",
        score: Number.POSITIVE_INFINITY,
      }));
    }

    return mergeCoreResults(usableSources);
  };

  const readTrafficAcrossFrames = async (expectedAliases = []) => {
    const aliases = Array.isArray(expectedAliases) ? expectedAliases.filter(Boolean) : [];
    const requirePeriodVerification = aliases.length > 0;
    const sources = await executeAcrossFrames("SCAN_TRAFFIC", { aliases }, {
      minWaitMs: 180,
      quietWaitMs: 130,
      maxWaitMs: 1100,
      stopWhen: (items) => items.some((item) =>
        item &&

  const readStableCore = async (
    statusPrefix,
    previousSignature = "",
    periodVerified = false,
    expectedAliases = []
  ) => {
    let best = null;
    let bestCount = -1;
    let bestSignature = "";
    let lastSignature = "";
    let sameSignatureReads = 0;

    for (let attempt = 1; attempt <= 8; attempt += 1) {
      const metrics = await readCoreAcrossFrames(expectedAliases);
      const count = metrics.filter((item) => item.value !== "").length;
      const signature = metrics.map((item) => `${item.label}:${item.value}`).join("|");

      if (count > bestCount) {
        best = metrics;
        bestCount = count;
        bestSignature = signature;
      }

      sameSignatureReads = signature && signature === lastSignature
        ? sameSignatureReads + 1
        : 1;
      const changedFromPrevious = !previousSignature || signature !== previousSignature;
      const stableEnough = sameSignatureReads >= 2;

      if (count === CORE_METRICS.length && changedFromPrevious && stableEnough) {
        return metrics;
      }

      // 两个周期的数据确实可能完全相同；v2.4.1 只有“同一数据来源 frame 内、目标板块的目标周期严格激活”时才允许接收。
      if (
        count === CORE_METRICS.length &&
        periodVerified &&
        stableEnough &&
        attempt >= 3
      ) {
        return metrics;
      }

      if (
        count >= 7 &&
        changedFromPrevious &&
        stableEnough &&
        attempt >= 4
      ) {
        return metrics;
      }

      lastSignature = signature;
      if (attempt < 8) {
        setStatus(
          `${statusPrefix}，核心数据已识别 ${count}/9，正在确认页面已刷新（${attempt}/8）…`
        );
        await sleep(360);
      }
    }

    if (
      best &&
      bestCount >= 7 &&
      (!previousSignature || bestSignature !== previousSignature || periodVerified)
    ) {
      return best;
    }

    throw new Error("核心数据目标周期未在对应板块内完成刷新/验证，为防止把上一周期旧数据重复保存，本周期已停止并保留断点");
  };

      const run = () => {
        activeHttpRequests += 1;

          .then(task)

            activeHttpRequests = Math.max(0, activeHttpRequests - 1);
            const next = httpWaitQueue.shift();
            if (next) next();
          });
      };

      if (activeHttpRequests < getHttpConcurrency()) run();
      else httpWaitQueue.push(run);
    });

      () =>
        new Promise((resolve, reject) => {
          if (typeof GM_xmlhttpRequest !== "function") {
            reject(new Error("GM_xmlhttpRequest 不可用，请检查 Tampermonkey 权限"));
            return;

          GM_xmlhttpRequest({
            method: "GET",
            url,
            headers: { Accept: "application/json, text/plain, */*" },
            timeout: 20000,
            anonymous: false,
            onload: (response) => {

                reject(
                  new Error(
                    `${label}返回 HTTP ${response.status}，请重新登录并刷新页面`
                  )
                );
                return;

              if (response.status < 200 || response.status >= 300) {
                reject(new Error(`${label}返回 HTTP ${response.status}`));
                return;
              }
              try {
                resolve(JSON.parse(response.responseText));
              } catch (error) {
                reject(new Error(`${label} JSON 解析失败：${error.message}`));
              }
            },
            onerror: () => reject(new Error(`${label}网络请求失败`)),
            ontimeout: () => reject(new Error(`${label}请求超时`)),
          });

    );

  const mapWithConcurrency = async (items, concurrency, worker) => {
    const list = Array.from(items || []);
    if (!list.length) return [];
    const results = new Array(list.length);
    let cursor = 0;
    const workerCount = Math.max(1, Math.min(Number(concurrency) || 1, list.length));

    const runners = Array.from({ length: workerCount }, async () => {
      while (true) {
        const index = cursor;
        cursor += 1;
        if (index >= list.length) return;
        results[index] = await worker(list[index], index);
      }
    });

    await Promise.all(runners);
    return results;
  };

        tx.oncomplete = () => resolve();
        tx.onerror = () => reject(tx.error || new Error("IndexedDB 保存失败"));
        tx.onabort = () => reject(tx.error || new Error("IndexedDB 保存被中止"));

      const rows = XLSX.utils.sheet_to_json(workbook.Sheets["留存断点恢复"], { defval: "" });
      for (const row of rows) {
        const albumId = String(row.albumId || "");
        if (!albumId) continue;
        const record = sanitizeRetentionRecord({
          albumId,
          albumName: cleanText(row.albumName) || albumId,
          weekIndex: row.weekIndex,
          startDate: row.startDate,
          endDate: row.endDate,
          added: row.added,
          rates: Array.from({ length: WEEK_SLOTS }, (_, index) => row[`rate${index + 1}`]),
        });
        if (!record) continue;
        importedRetentionRecords.push(record);
        importedRetentionCompletedAlbumIds.add(albumId);
      }
    } else if (workbook.Sheets["留存分析"]) {
      const rows = XLSX.utils.sheet_to_json(workbook.Sheets["留存分析"], { header: 1, defval: "" });
      const headerIndex = rows.findIndex(
        (row) => row.includes("首次收听时间") && row.includes("新增听众") && row.includes("1周后")
      );
      if (headerIndex >= 0) {
        let currentAlbumName = "";
        for (let index = headerIndex + 1; index < rows.length; index += 1) {
          const row = rows[index];
          if (cleanText(row[0])) currentAlbumName = cleanText(row[0]);
          if (!currentAlbumName || !cleanText(row[1])) continue;
          const album = findAlbumForImportedName(currentAlbumName);
          if (!album) continue;
          const timeText = cleanText(row[1]);
          const match = timeText.match(/第\s*(\d+)\s*周\s*(\d{4}-\d{2}-\d{2})\s*[~～-]\s*(\d{4}-\d{2}-\d{2})/);
          const record = sanitizeRetentionRecord({
            albumId: album.id,
            albumName: album.title,
            weekIndex: match ? match[1] : "",
            startDate: match ? match[2] : "",
            endDate: match ? match[3] : "",
            added: row[2],
            rates: Array.from({ length: WEEK_SLOTS }, (_, rateIndex) => row[rateIndex + 3]),
          });
          if (!record) continue;
          importedRetentionRecords.push(record);
          importedRetentionCompletedAlbumIds.add(String(album.id));
        }
      }
    }

    const failureSheet = workbook.Sheets["失败专辑"];
    if (failureSheet) {
      const rows = XLSX.utils.sheet_to_json(failureSheet, { defval: "" });
      for (const row of rows) {
        importedFailures.push({
          albumId: String(row["专辑ID"] || ""),
          albumName: cleanText(row["专辑名称"]),
          stage: cleanText(row["失败阶段"]),
          reason: cleanText(row["失败原因"]),
        });
      }
    }

    if (!importedRecords.length) throw new Error("备份文件中没有识别到可恢复的数据行");

    const existing = await loadAllExportProgress();
    const merged = new Map();
    for (const record of existing && Array.isArray(existing.records) ? existing.records : []) {
      const sanitizedRecord = sanitizeRecordTraffic(record);
      merged.set(
        recordKey(sanitizedRecord.albumId, sanitizedRecord.period),
        sanitizedRecord
      );
    }
    for (const record of importedRecords) {
      const sanitizedRecord = sanitizeRecordTraffic(record);
      merged.set(
        recordKey(sanitizedRecord.albumId, sanitizedRecord.period),
        sanitizedRecord
      );
    }

    const mergedRetention = new Map();
    for (const record of existing && Array.isArray(existing.retentionRecords) ? existing.retentionRecords : []) {
      const sanitized = sanitizeRetentionRecord(record);
      if (sanitized) mergedRetention.set(retentionRecordKey(sanitized), sanitized);
    }
    for (const record of importedRetentionRecords) {
      const sanitized = sanitizeRetentionRecord(record);
      if (sanitized) mergedRetention.set(retentionRecordKey(sanitized), sanitized);
    }

    const retentionCompletedAlbumIds = new Set(
      existing && Array.isArray(existing.retentionCompletedAlbumIds)
        ? existing.retentionCompletedAlbumIds.map(String)
        : []
    );
    for (const albumId of importedRetentionCompletedAlbumIds) {
      retentionCompletedAlbumIds.add(String(albumId));
    }

    const progress = {
      version: VERSION,
      signature: cachedAlbums.map((item) => String(item.id)).join("|"),
      records: Array.from(merged.values()),
      retentionRecords: Array.from(mergedRetention.values()),
      retentionCompletedAlbumIds: Array.from(retentionCompletedAlbumIds),
      failures: importedFailures.length
        ? importedFailures
        : existing && Array.isArray(existing.failures)
          ? existing.failures
          : [],
      updatedAt: new Date().toISOString(),
      importedFrom: file.name,
    };
    await saveAllExportProgress(progress);
    setStatus(
      `✅ 已导入 ${importedRecords.length} 个核心周期、${importedRetentionRecords.length} 行留存；当前断点共 ${progress.records.length} 个周期、${progress.retentionRecords.length} 行留存；点击“极速导出全部专辑（API并发）”只补抓缺失数据`
    );
  };

  window.addEventListener("pagehide", () => {
    if (activeProgressSnapshot) {
      safeStore(ALL_EXPORT_PROGRESS_KEY, JSON.stringify(activeProgressSnapshot));
    }
  });

  const runAllAlbumsExport = async () => {
    if (exportingAll) {
      setStatus("极速并发接口抓取正在执行，请不要重复点击", true);
      return;
    }

    exportingAll = true;
    const concurrencyProfile = getConcurrencyProfile();
    [allButton, retentionButton, reloadButton, albumSelect, concurrencySelect].forEach((item) => {
      if (item) item.disabled = true;
    });
    allButton.textContent = `极速并发抓取中（${concurrencyProfile.album}路）…`;

    let records = [];
    let retentionRecords = [];
    let retentionCompletedAlbumIds = new Set();
    let failures = [];
    let forcePersistProgress = async () => {};

    try {
      if (!cachedAlbums.length) {
        setStatus("正在通过接口获取全部专辑列表…");
        cachedAlbums = await loadAlbums((page, pages, count) =>
          setStatus(`专辑列表 ${page}/${pages}，已获取 ${count} 个`)
        );
      }
      if (!cachedAlbums.length) throw new Error("没有获取到任何专辑");

      const allAlbums = cachedAlbums.slice();
      const currentAlbumIds = new Set(allAlbums.map((item) => String(item.id)));
      const signature = allAlbums.map((item) => String(item.id)).join("|");
      const savedProgress = await loadAllExportProgress();

      if (savedProgress && Array.isArray(savedProgress.records)) {
        const dedupe = new Map();
        for (const item of savedProgress.records) {
          if (!item || !currentAlbumIds.has(String(item.albumId)) || !item.period) {
            continue;
          }
          const sanitized = sanitizeRecordTraffic({
            ...item,
            period: normalizePeriodName(item.period),
          });
          dedupe.set(recordKey(sanitized.albumId, sanitized.period), sanitized);
        }
        records = Array.from(dedupe.values());

        failures = Array.isArray(savedProgress.failures)
          ? savedProgress.failures.filter((item) =>
              currentAlbumIds.has(String(item.albumId))
            )
          : [];

        const retentionDedupe = new Map();
        for (const item of Array.isArray(savedProgress.retentionRecords)
          ? savedProgress.retentionRecords
          : []) {
          const sanitized = sanitizeRetentionRecord(item);
          if (!sanitized || !currentAlbumIds.has(String(sanitized.albumId))) {
            continue;
          }
          retentionDedupe.set(retentionRecordKey(sanitized), sanitized);
        }
        retentionRecords = Array.from(retentionDedupe.values());

        retentionCompletedAlbumIds = new Set(
          (Array.isArray(savedProgress.retentionCompletedAlbumIds)
            ? savedProgress.retentionCompletedAlbumIds
            : [])
            .map(String)
            .filter((albumId) => currentAlbumIds.has(albumId))
        );
      }

      // v3.2.0：沿用 Map 索引，避免并发运行时反复线性扫描。
      const recordMap = new Map();
      for (const item of records) {
        recordMap.set(recordKey(item.albumId, item.period), item);
      }

      const getRecord = (albumId, period) =>
        recordMap.get(recordKey(albumId, period)) || null;

      const coreIsComplete = (record) =>
        Boolean(
          record &&
          record.core &&
          CORE_METRICS.every((name) => cleanText(record.core[name]) !== "")
        );

      const trafficIsComplete = (record) => {
        if (!record) return false;
        if (isTrafficDisabledPeriod(record.period)) return true;
        return Boolean(
          record.traffic &&
          cleanText(record.traffic.exposureClick) !== "" &&
          cleanText(record.traffic.clickPlay) !== "" &&
          cleanText(record.traffic.playFinish) !== ""
        );
      };

      const periodIsComplete = (albumId, periodName) => {
        const record = getRecord(albumId, periodName);
        return coreIsComplete(record) && trafficIsComplete(record);
      };

      const isCoreAlbumComplete = (albumId) =>
        EXPORT_PERIODS.every((period) =>
          periodIsComplete(String(albumId), period.outputName)
        );

      const isRetentionComplete = (albumId) =>
        retentionCompletedAlbumIds.has(String(albumId));

      const isAlbumComplete = (albumId) =>
        isCoreAlbumComplete(albumId) && isRetentionComplete(albumId);

      const completedAlbumCount = () =>
        allAlbums.filter((album) => isAlbumComplete(album.id)).length;

      const upsertRecord = (record) => {
        const sanitized = sanitizeRecordTraffic(record);
        const key = recordKey(sanitized.albumId, sanitized.period);
        const existing = recordMap.get(key);
        if (existing) {
          const index = records.indexOf(existing);
          if (index >= 0) records[index] = sanitized;
        } else {
          records.push(sanitized);
        }
        recordMap.set(key, sanitized);
      };

      const clearFailure = (albumId, stage) => {
        failures = failures.filter(
          (item) =>
            !(
              String(item.albumId) === String(albumId) &&
              cleanText(item.stage) === cleanText(stage)
            )
        );
      };

      const addFailure = (album, stage, error) => {
        clearFailure(album.id, stage);
        failures.push({
          albumId: String(album.id),
          albumName: album.title,
          stage,
          reason: error && error.message ? error.message : String(error || "未知错误"),
        });
      };

      const buildProgressPayload = () => {
        const completedPeriodKeys = [];
        for (const album of allAlbums) {
          for (const period of EXPORT_PERIODS) {
            if (periodIsComplete(String(album.id), period.outputName)) {
              completedPeriodKeys.push(recordKey(album.id, period.outputName));
            }
          }
        }

        return {
          version: VERSION,
          mode: "api-turbo",
          signature,
          records,
          retentionRecords,
          retentionCompletedAlbumIds: Array.from(retentionCompletedAlbumIds),
          completedAlbumIds: allAlbums
            .filter((album) => isAlbumComplete(album.id))
            .map((album) => String(album.id)),
          completedPeriodKeys,
          failures,
          updatedAt: new Date().toISOString(),
        };
      };

      // v3.1.0：断点不再每个 API 小步骤都写整份 JSON。
      // 状态先留在内存，每 1.2 秒或每完成 5 张专辑批量写一次；最终导出前强制落盘。
      let progressDirty = false;
      let checkpointTimer = null;
      let checkpointChain = Promise.resolve();
      let albumsSinceCheckpoint = 0;

      const flushProgress = async (force = false) => {
        if (!progressDirty && !force) return checkpointChain;
        progressDirty = false;
        albumsSinceCheckpoint = 0;
        if (checkpointTimer) {
          clearTimeout(checkpointTimer);
          checkpointTimer = null;
        }
        const payload = buildProgressPayload();
        checkpointChain = checkpointChain
          .catch(() => {})
          .then(() => saveAllExportProgress(payload));
        await checkpointChain;
        if (progressDirty && !checkpointTimer) {
          checkpointTimer = setTimeout(() => {
            checkpointTimer = null;
            flushProgress(false).catch((error) =>
              console.warn("[xm-export] 批量断点保存失败", error)
            );
          }, CHECKPOINT_INTERVAL_MS);
        }
        return checkpointChain;
      };

      const markProgressDirty = (albumFinished = false) => {
        progressDirty = true;
        if (albumFinished) albumsSinceCheckpoint += 1;

        if (albumsSinceCheckpoint >= CHECKPOINT_ALBUM_BATCH) {
          if (checkpointTimer) {
            clearTimeout(checkpointTimer);
            checkpointTimer = null;
          }
          Promise.resolve().then(() =>
            flushProgress(false).catch((error) =>
              console.warn("[xm-export] 批量断点保存失败", error)
            )
          );
          return;
        }

        if (!checkpointTimer) {
          checkpointTimer = setTimeout(() => {
            checkpointTimer = null;
            flushProgress(false).catch((error) =>
              console.warn("[xm-export] 批量断点保存失败", error)
            );
          }, CHECKPOINT_INTERVAL_MS);
        }
      };

      forcePersistProgress = async () => {
        progressDirty = true;
        await flushProgress(true);
        // 如果保存过程中又有 worker 更新，继续补一次，直到完全干净。
        while (progressDirty) await flushProgress(true);
      };

      await forcePersistProgress();

      if (records.length || retentionRecords.length) {
        setStatus(
          `已恢复断点：核心 ${records.length} 行，留存 ${retentionRecords.length} 行；将以 ${concurrencyProfile.album} 路·${concurrencyProfile.label}档并发只补抓缺失部分…`
        );
      }

      const pendingAlbums = allAlbums
        .map((album, albumIndex) => ({ album, albumIndex }))
        .filter(({ album }) => !isAlbumComplete(String(album.id)));

      // 平台的数据截止日通常全账号统一。先只预取 1 次 KPI 确定截止日，
      // 后续专辑即可让 KPI / 7天流量 / 30天流量 / 留存同时起跑。
      // 若某张专辑返回的 KPI 截止日不同，会自动用该专辑真实截止日重抓流量。
      let sharedDataEndDate = null;
      const prefetchedKpiBundles = new Map();
      const preflightTarget = pendingAlbums.find(({ album }) => {
        const albumId = String(album.id);
        const missingCorePeriods = EXPORT_PERIODS.some(
          (period) => !coreIsComplete(getRecord(albumId, period.outputName))
        );
        const missingTraffic =
          !trafficIsComplete(getRecord(albumId, "近7天")) ||
          !trafficIsComplete(getRecord(albumId, "近30天"));
        return missingCorePeriods || missingTraffic;
      });

      if (preflightTarget) {
        try {
          setStatus(`正在预取数据截止日：${preflightTarget.album.title}…`);
          const bundle = await fetchKpiTotalBundle(preflightTarget.album);
          prefetchedKpiBundles.set(String(preflightTarget.album.id), bundle);
          sharedDataEndDate = bundle.endDate;
        } catch (error) {
          console.warn("[xm-export] 截止日预取失败，将自动降级为逐专辑确认", error);
        }
      }

      let finishedJobs = 0;
      let activeJobs = 0;

      const renderTurboStatus = (message = "") => {
        const done = completedAlbumCount();
        setStatus(
          `并发 ${concurrencyProfile.album} 路·${concurrencyProfile.label}｜HTTP ${activeHttpRequests}/${concurrencyProfile.http}｜` +
            `完整 ${done}/${allAlbums.length}｜任务 ${finishedJobs}/${pendingAlbums.length}` +
            (message ? `｜${message}` : "")
        );
      };

      const fetchTrafficPair = async (
        album,
        endDate,
        missing7dTraffic,
        missing30dTraffic
      ) => {
        const jobs = [];
        if (missing7dTraffic) {
          jobs.push(
            fetchConversionTraffic(album, 7, endDate)
              .then((traffic) => ({ period: "近7天", traffic }))
              .catch((error) => ({ period: "近7天", error }))
          );
        }
        if (missing30dTraffic) {
          jobs.push(
            fetchConversionTraffic(album, 30, endDate)
              .then((traffic) => ({ period: "近30天", traffic }))
              .catch((error) => ({ period: "近30天", error }))
          );
        }
        return Promise.all(jobs);
      };

      const applyTrafficResults = (album, albumId, kpiBundle, results) => {
        for (const result of Array.isArray(results) ? results : []) {
          const stage = `${result.period}流量转化API`;
          if (result.error) {
            console.warn(`[xm-export] ${stage}失败`, {
              album,
              error: result.error,
            });
            addFailure(album, stage, result.error);
            continue;
          }

          const existing = getRecord(albumId, result.period);
          upsertRecord({
            albumId,
            albumName: album.title,
            period: result.period,
            core:
              existing && existing.core
                ? existing.core
                : kpiBundle && kpiBundle.cores
                  ? kpiBundle.cores[result.period]
                  : {},
            traffic: result.traffic,
          });
          clearFailure(albumId, stage);
        }
      };

      const processAlbum = async ({ album, albumIndex }) => {
        const albumId = String(album.id);
        const prefix = `[${albumIndex + 1}/${allAlbums.length}] ${album.title}`;
        activeJobs += 1;
        renderTurboStatus(`${prefix} 开始`);

        try {
          await waitForOnline(prefix);

          const missingCorePeriods = EXPORT_PERIODS.filter((period) => {
            const record = getRecord(albumId, period.outputName);
            return !coreIsComplete(record);
          });
          const missing7dTraffic = !trafficIsComplete(getRecord(albumId, "近7天"));
          const missing30dTraffic = !trafficIsComplete(getRecord(albumId, "近30天"));
          const needKpi =
            missingCorePeriods.length > 0 || missing7dTraffic || missing30dTraffic;
          const needRetention = !isRetentionComplete(albumId);

          const cachedBundle = prefetchedKpiBundles.get(albumId) || null;
          const kpiPromise = needKpi
            ? cachedBundle
              ? Promise.resolve({ bundle: cachedBundle, error: null })
              : fetchKpiTotalBundle(album)
                  .then((bundle) => ({ bundle, error: null }))
                  .catch((error) => ({ bundle: null, error }))
            : Promise.resolve({ bundle: null, error: null });

          // 有共享截止日时，流量不再等待本专辑 KPI，直接和 KPI 同时发送。
          const earlyTrafficPromise =
            sharedDataEndDate && (missing7dTraffic || missing30dTraffic)
              ? fetchTrafficPair(
                  album,
                  sharedDataEndDate,
                  missing7dTraffic,
                  missing30dTraffic
                )
              : Promise.resolve(null);

          const retentionPromise = needRetention
            ? fetchAllRetentionRows(albumId)
                .then((rawRows) => ({ rawRows, error: null }))
                .catch((error) => ({ rawRows: null, error }))
            : Promise.resolve({ rawRows: null, error: null });

          // 三大分支同步推进：KPI、提前流量、留存。
          const [{ bundle: kpiBundle, error: kpiError }, earlyTraffic, retentionResult] =
            await Promise.all([kpiPromise, earlyTrafficPromise, retentionPromise]);

          if (needKpi) {
            if (kpiError || !kpiBundle) {
              const error = kpiError || new Error("核心数据API未返回有效结果");
              console.warn("[xm-export] 核心数据API失败", { album, error });
              addFailure(album, "核心数据API", error);
            } else {
              for (const period of EXPORT_PERIODS) {
                const existing = getRecord(albumId, period.outputName);
                const core = kpiBundle.cores[period.outputName];
                if (!core) continue;
                upsertRecord({
                  albumId,
                  albumName: album.title,
                  period: period.outputName,
                  core,
                  traffic:
                    existing && existing.traffic
                      ? existing.traffic
                      : { ...EMPTY_TRAFFIC },
                });
              }
              clearFailure(albumId, "核心数据API");

              const actualEndDate = kpiBundle.endDate;
              const sharedYmd = sharedDataEndDate
                ? formatLocalYmd(sharedDataEndDate)
                : "";
              const actualYmd = actualEndDate ? formatLocalYmd(actualEndDate) : "";

              let trafficResults = earlyTraffic;

              // 无共享截止日：此处才补发流量；有共享截止日但该专辑日期不同：重抓一次纠偏。
              if (
                (missing7dTraffic || missing30dTraffic) &&
                (!trafficResults || (sharedYmd && actualYmd && sharedYmd !== actualYmd))
              ) {
                trafficResults = await fetchTrafficPair(
                  album,
                  actualEndDate,
                  missing7dTraffic,
                  missing30dTraffic
                );
              }

              if (trafficResults) {
                applyTrafficResults(
                  album,
                  albumId,
                  kpiBundle,
                  trafficResults
                );
              }
            }
          }

          if (needRetention) {
            if (retentionResult.error) {
              console.warn("[xm-export] 留存API失败", {
                album,
                error: retentionResult.error,
              });
              addFailure(album, "留存分析", retentionResult.error);
            } else {
              const normalizedRows = (retentionResult.rawRows || [])
                .map((item) => retentionApiItemToRecord(album, item))
                .filter(Boolean);

              retentionRecords = retentionRecords.filter(
                (item) => String(item.albumId) !== albumId
              );
              retentionRecords.push(...normalizedRows);
              retentionCompletedAlbumIds.add(albumId);
              clearFailure(albumId, "留存分析");
            }
          }

          markProgressDirty(false);

          if (isAlbumComplete(albumId)) {
            failures = failures.filter((item) => String(item.albumId) !== albumId);
          }
          markProgressDirty(true);
        } finally {
          activeJobs = Math.max(0, activeJobs - 1);
          finishedJobs += 1;
          renderTurboStatus(`${prefix} 完成`);
        }
      };

      await mapWithConcurrency(
        pendingAlbums,
        concurrencyProfile.album,
        processAlbum
      );

      await forcePersistProgress();
      downloadAllAlbumsXlsx(records, failures, "", retentionRecords);
      const done = completedAlbumCount();
      await forcePersistProgress();

      if (!failures.length && done === allAlbums.length) {
        setStatus(
          `✅ 极速并发导出完成：${done}/${allAlbums.length} 个专辑；核心 ${records.length} 行，留存 ${retentionRecords.length} 行。使用 ${concurrencyProfile.album} 路·${concurrencyProfile.label}档（HTTP ${concurrencyProfile.http}），核心/留存并行、7/30天流量并行。`
        );
      } else {
        setStatus(
          `⚠️ 极速并发导出完成：完整专辑 ${done}/${allAlbums.length}，核心 ${records.length} 行，留存 ${retentionRecords.length} 行，失败 ${failures.length} 项。再次点击只补抓缺失项。`,
          true
        );
      }
    } catch (error) {
      console.error("[xm-export] 极速并发接口导出失败", error);
      try {
        await forcePersistProgress();
      } catch (persistError) {
        console.warn("[xm-export] 异常前强制断点保存失败", persistError);
      }
      try {
        const progress = activeProgressSnapshot || (await loadAllExportProgress());
        if (progress && Array.isArray(progress.records) && progress.records.length) {
          downloadAllAlbumsXlsx(
            progress.records,
            Array.isArray(progress.failures) ? progress.failures : [],
            "全部专辑数据_API极速并发版_异常中断自动备份.xlsx",
            Array.isArray(progress.retentionRecords)
              ? progress.retentionRecords
              : []
          );
          setStatus(
            `任务异常：${error.message}。已自动保存并导出中断前备份；再次点击可从断点继续。`,
            true
          );
        } else {
          setStatus(`极速并发接口导出失败：${error.message}`, true);
        }
      } catch (backupError) {
        setStatus(
          `极速并发接口导出失败：${error.message}；自动备份也失败：${backupError.message}`,
          true
        );
      }
    } finally {
      exportingAll = false;
      [allButton, retentionButton, reloadButton, albumSelect, concurrencySelect].forEach((item) => {
        if (item) item.disabled = false;
      });
      allButton.textContent = "极速导出全部专辑（API并发）";
      renderConcurrencyChoice(false);
    }
  };

  const getChosenAlbum = () => {
    const manual = cleanText(albumInput.value);

    if (manual) {
      return {
        id: manual,
        title: "",
      };
    }

    const option = albumSelect.options[albumSelect.selectedIndex];

    return {
      id: albumSelect.value,
      title: option ? cleanText(option.textContent) : "",
    };
  };

  progressButton.addEventListener("click", exportSavedProgressNow);
  exportYesterdayButton.addEventListener("click", () =>
    exportSavedPeriodNow("昨日")
  );
  exportWeekButton.addEventListener("click", () =>
    exportSavedPeriodNow("近7天")
  );
  exportMonthButton.addEventListener("click", () =>
    exportSavedPeriodNow("近30天")
  );

    exportSavedPeriodNow("近1年")
  );

    exportSavedPeriodNow("累计")
  );

    const file = importFileInput.files && importFileInput.files[0];
    if (!file) return;
    importButton.disabled = true;
    try {
      setStatus(`正在导入阶段备份：${file.name}…`);
      await importProgressWorkbook(file);
    } catch (error) {
      console.error("[xm-export] 导入阶段备份失败", error);
      setStatus(`导入阶段备份失败：${error.message}`, true);
    } finally {
      importButton.disabled = false;
      importFileInput.value = "";
    }
  });

  clearProgressButton.addEventListener("click", async () => {
    const confirmed = window.confirm(
      "确定清空全部断点吗？清空后将从第一个专辑重新抓取。建议先导出当前已保存进度。"
    );
    if (!confirmed) return;
    await clearAllExportProgress();
    setStatus("已清空断点，下次将从第一个专辑重新开始");
  });

  retentionButton.addEventListener("click", () => {
    const chosen = getChosenAlbum();
    runRetentionExport(chosen.id, chosen.title);
  });

    GM_registerMenuCommand("📊 极速导出全部专辑（API并发）", runAllAlbumsExport);

      runRetentionExport(chosen.id, chosen.title);
    });

    GM_registerMenuCommand("🔄 重新加载专辑列表", refreshAlbumList);
  }

    if (
      event.altKey &&
      event.shiftKey &&
      String(event.key).toLowerCase() === "e"
    ) {
      event.preventDefault();
      runAllAlbumsExport();
    }
  });

  console.log(`[xm-export] v${VERSION} loaded on`, location.href);
  refreshAlbumList();
  loadAllExportProgress().then((progress) => {
    if (progress && Array.isArray(progress.records) && progress.records.length) {
      setStatus(`检测到已保存断点：${progress.records.length} 个周期。点击“极速导出全部专辑（API并发）”将自动续传。`);
    }
  }).catch(() => {});
