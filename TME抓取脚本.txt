    function openStorageDb() {
        if (storageDbPromise) {
            return storageDbPromise;
        }

        storageDbPromise = new Promise((resolve, reject) => {
            if (!window.indexedDB) {
                reject(new Error('当前浏览器不支持 IndexedDB，大量数据无法安全保存。'));
                return;
            }

            const request = indexedDB.open(CFG.dbName, CFG.dbVersion);

            request.onupgradeneeded = () => {
                const db = request.result;

                if (!db.objectStoreNames.contains(CFG.dbMetaStore)) {
                    db.createObjectStore(CFG.dbMetaStore);
                }

                if (!db.objectStoreNames.contains(CFG.dbRecordsStore)) {
                    db.createObjectStore(CFG.dbRecordsStore, {
                        keyPath: 'key'
                    });
                }
            };

            request.onsuccess = () => resolve(request.result);
            request.onerror = () => reject(
                request.error || new Error('IndexedDB打开失败')
            );
            request.onblocked = () => reject(
                new Error('IndexedDB升级被其他页面阻塞，请关闭同站点旧页面后刷新。')
            );
        });

        return storageDbPromise;
    }

    function requestPromise(request) {
        return new Promise((resolve, reject) => {
            request.onsuccess = () => resolve(request.result);
            request.onerror = () => reject(
                request.error || new Error('IndexedDB请求失败')
            );
        });
    }

    function transactionPromise(transaction) {
        return new Promise((resolve, reject) => {
            transaction.oncomplete = () => resolve();
            transaction.onerror = () => reject(
                transaction.error || new Error('IndexedDB事务失败')
            );
            transaction.onabort = () => reject(
                transaction.error || new Error('IndexedDB事务已中止')
            );
        });
    }

        );

        return value === undefined ? fallback : value;
    }

    function parseNumber(value) {
        const text =
            compact(value)
                .replace(/,/g, '')
                .replace(
                    /分钟|min/gi,
                    ''
                );

        /* 接口读取到任一种横杠时，按有效零值处理。 */
        if (isDashPlaceholder(text)) {
            return 0;
        }

        let multiplier = 1;

        if (text.includes('亿')) {
            multiplier =
                100000000;
        } else if (
            text.includes('万')
        ) {
            multiplier =
                10000;
        }

        const numericText =
            text.replace(
                /[^\d.]/g,
                ''
            );

        if (!numericText) {
            return null;
        }

        const number =
            Number(numericText);

        return Number.isFinite(number)
            ? number * multiplier
            : null;
    }

    function normalizeRecord(item) {
        let value =
            item.value ??
            item.numericValue ??
            '';

        if (
            value === '' &&
            item.rawValue != null
        ) {
            value =
                parseNumber(
                    item.rawValue
                ) ?? '';
        }

        if (value !== '') {
            const number =
                Number(
                    String(value)
                        .replace(/,/g, '')
                );

            value =
                Number.isFinite(number)
                    ? number
                    : '';
        }

        let percent = '';

        const percentCandidates = [
            item.percent,
            item.percentage,
            item.rawPercent
        ];

        for (const candidate of percentCandidates) {
            if (
                candidate === '' ||
                candidate === null ||
                candidate === undefined
            ) {
                continue;
            }

            const text =
                compact(candidate)
                    .replace(/[％]/g, '%');

            if (isDashPlaceholder(text)) {
                percent = 0;
                break;
            }

            const numericText =
                text
                    .replace(/%/g, '')
                    .replace(/,/g, '');

            if (/^\d+(?:\.\d+)?$/.test(numericText)) {
                const number = Number(numericText);

                if (Number.isFinite(number)) {
                    percent =
                        number < CFG.percentZeroThreshold
                            ? 0
                            : number;
                    break;
                }
            }

            const parsed =
                parsePercentValue(
                    text.includes('占比')
                        ? text
                        : `占比:${text}`
                );

            if (parsed.ready) {
                percent = parsed.value;
                break;
            }
        }

        if (
            percent === '' &&
            clean(item.sourceText)
        ) {
            const parsed =
                parsePercentValue(
                    item.sourceText
                );

            if (parsed.ready) {
                percent = parsed.value;
            }
        }

        let status =
            clean(item.status) ||
            (
                value !== '' &&
                percent !== ''
                    ? '成功'
                    : '未识别'
            );

        if (
            status === '成功' &&
            (
                value === '' ||
                percent === ''
            )
        ) {
            status = '未识别';
        }

        return {
            album:
                clean(
                    item.album ??
                    item.albumName
                ) ||
                '未识别专辑',

            period:
                normalizePeriod(
                    item.period
                ),

            platform:
                normalizePlatform(
                    item.platform
                ),

            metric:
                clean(item.metric),

            unit:
                clean(item.unit),

            value,

            percent,

            status,

            note:
                clean(item.note)
        };
    }

    function parseIsoDate(value) {
        const match = String(value || '').match(/^(\d{4})-(\d{2})-(\d{2})$/);

        if (!match) {
            return null;
        }

        return Date.UTC(
            Number(match[1]),
            Number(match[2]) - 1,
            Number(match[3])
        );
    }

    function classifyPeriodRange(startDate, endDate) {
        const start = parseIsoDate(startDate);
        const end = parseIsoDate(endDate);

        if (start === null || end === null || end < start) {
            return '';
        }

        const days = Math.round((end - start) / 86400000);

        if (days === 0) {
            return '昨天';
        }

        if (days === 6) {
            return '近七天';
        }

        if (days === 29) {
            return '近30天';
        }

        /*
         * 累计不是页面右上角的固定快捷周期，而是整本专辑的全历史区间。
         * 本脚本会用一个足够早的起始日期生成fanChart请求，因此凡是
         * 超过30天的区间都按“累计”识别。
         */
        if (days > 29) {
            return '累计';
        }

        return '';
    }

    function metricFromRequestType(value) {
        const type = clean(value).toLowerCase();

        if (/playduration|duration|playtime/.test(type)) {
            return '播放时长';
        }

        if (/peoplecnt|peoplecount|personcnt|listener|usercnt/.test(type)) {
            return '播放人数';
        }

        if (/playcnt|playcount|playnum|playvolume|plays/.test(type)) {
            return '播放量';
        }

        return '';
    }

    function requestMeta(value) {
        let url;

        try {
            url = new URL(String(value || ''), location.href);
        } catch (error) {
            return null;
        }

        if (
            !/\/api\/v1\/datacenter\/platform\/fanChart\/?$/i.test(
                url.pathname
            )
        ) {
            return null;
        }

        const startDate = url.searchParams.get('startDate') || '';
        const endDate = url.searchParams.get('endDate') || '';
        const period = classifyPeriodRange(startDate, endDate);
        const metric = metricFromRequestType(
            url.searchParams.get('type')
        );
        const albumId = url.searchParams.get('albumId') || '';

        if (!period || !metric || !albumId) {
            return null;
        }

        return {
            url: url.href,
            period,
            metric,
            type: url.searchParams.get('type') || '',
            albumId,
            startDate,
            endDate
        };
    }

    function headersObject(value) {
        const result = {};

        if (!value) {
            return result;
        }

        try {
            const headers = new Headers(value);

            headers.forEach((headerValue, name) => {
                result[String(name).toLowerCase()] = String(headerValue);
            });
        } catch (error) {
            if (typeof value === 'object') {
                for (const [name, headerValue] of Object.entries(value)) {
                    result[String(name).toLowerCase()] = String(headerValue);
                }
            }
        }

        return result;
    }

    function albumAtRequestStart() {
        try {

            return name === '未识别专辑' ? '' : name;
        } catch (error) {

    }

    function beginTrackedRequest(value, options = {}) {
        const meta = requestMeta(value);

        if (!meta) {
            return 0;
        }

        const id = ++requestTracker.seq;

        requestTracker.pending.set(id, {
            id,
            ...meta,
            method: clean(options.method || 'GET').toUpperCase(),
            requestHeaders: headersObject(options.headers),
            albumName: albumAtRequestStart(),
            startedAt: Date.now()
        });

        return id;
    }

    function decodeJsonText(value) {
        const text = String(value ?? '').trim();

        if (!text) {
            return {
                payload: null,
                parseError: '接口响应为空'
            };
        }

        try {
            return {
                payload: JSON.parse(text),
                parseError: ''
            };
        } catch (error) {
            return {
                payload: null,

    function finishTrackedRequest(id, result = {}) {
        if (!id) {
            return;
        }

        const item = requestTracker.pending.get(id);

        if (!item) {
            return;
        }

        requestTracker.pending.delete(id);
        requestTracker.completed.push({
            ...item,
            ok: Boolean(result.ok),
            status: Number(result.status || 0),
            payload:
                result.payload === undefined
                    ? null
                    : result.payload,
            parseError: clean(result.parseError),
            completedAt: Date.now()
        });

        if (requestTracker.completed.length > 300) {
            requestTracker.completed.splice(
                0,
                requestTracker.completed.length - 300
            );
        }
    }

    function primitiveObjectField(node, preferredKeys) {

            preferredKeys.map(normalizedApiKey)
        );

        for (const [key, value] of Object.entries(node || {})) {

    function extractAlbumCatalog(payload, requestUrl = '') {
        const result = [];
        const seenNodes = new Set();
        let order = 0;

        const strongNameKeys = [
            'albumName',
            'albumTitle',
            'workName',
            'workTitle',
            'programName',
            'programTitle',
            'bookName',
            'audioName',
            'podcastName'
        ];

        const genericNameKeys = [
            'name',
            'title'
        ];

        const strongIdKeys = [
            'albumId',
            'workId',
            'programId',
            'bookId',
            'audioId',
            'podcastId'
        ];

        const genericIdKeys = [
            'id'
        ];

        function visit(node, path = [], depth = 0) {
            if (
                depth > 12 ||
                !node ||
                typeof node !== 'object' ||
                seenNodes.has(node)
            ) {
                return;
            }

            seenNodes.add(node);

            if (!Array.isArray(node)) {
                const strongName = primitiveObjectField(node, strongNameKeys);
                const genericName = primitiveObjectField(node, genericNameKeys);
                const strongId = primitiveObjectField(node, strongIdKeys);
                const genericId = primitiveObjectField(node, genericIdKeys);
                const nameField = strongName || genericName;
                const idField = strongId || genericId;

                if (nameField) {
                    const name = clean(nameField.value);
                    const contextText = `${requestUrl}|${path.join('.')}`.toLowerCase();
                    const hasEntityHint = /album|work|program|podcast|book|audio|voice|content|专辑|作品/.test(
                        contextText
                    );
                    const hasListHint = /list|search|page|query|option|select/.test(
                        contextText
                    );

                    let score = 0;

                    if (strongName) {
                        score += 190;
                    } else {
                        score += 45;
                    }

                    if (strongId) {
                        score += 170;
                    } else if (genericId) {
                        score += 45;
                    }

                    if (hasEntityHint) {
                        score += 80;
                    }

                    if (hasListHint) {
                        score += 30;
                    }

                    if (
                        validAlbumName(name) &&
                        (
                            strongName ||
                            strongId
                        ) &&
                        score >= 180
                    ) {
                        result.push({
                            name,
                            albumId:
                                idField
                                    ? clean(idField.value)
                                    : '',
                            score,
                            order: order++,
                            path: path.join('.')
                        });
                    }
                }
            }

            for (const [key, child] of Object.entries(node)) {
                if (child && typeof child === 'object') {
                    visit(
                        child,
                        path.concat(key),
                        depth + 1
                    );
                }
            }
        }

        visit(payload, [], 0);

        const byKey = new Map();

        for (const item of result) {
            const key = item.albumId
                ? `id:${item.albumId}`
                : `name:${compact(item.name)}`;
            const old = byKey.get(key);

            if (
                !old ||
                item.score > old.score
            ) {
                byKey.set(key, item);
            }
        }

        return Array.from(byKey.values())
            .sort((a, b) => a.order - b.order);
    }

    function extractAlbumTotal(payload, requestUrl = '') {
        const seen = new Set();
        const candidates = [];

        function visit(node, path = [], depth = 0) {
            if (
                depth > 10 ||
                !node ||
                typeof node !== 'object' ||
                seen.has(node)
            ) {
                return;
            }

            seen.add(node);

            for (const [key, value] of Object.entries(node)) {
                const normalized = normalizedApiKey(key);
                const contextText = `${requestUrl}|${path.join('.')}|${key}`.toLowerCase();

                if (
                    value !== null &&
                    value !== undefined &&
                    typeof value !== 'object'
                ) {
                    const number = Number(value);

                    if (
                        Number.isInteger(number) &&
                        number > 0 &&
                        number <= 20000
                    ) {
                        let score = 0;

                        if (/^(?:albumtotal|albumcount|totalcount|recordcount)$/.test(normalized)) {
                            score += 150;
                        } else if (normalized === 'total') {
                            score += 90;
                        } else if (normalized === 'count') {
                            score += 30;
                        }

                        if (/album|work|program|podcast|book|audio|专辑|作品/.test(contextText)) {
                            score += 70;
                        }

                        if (/list|search|page|query/.test(contextText)) {
                            score += 35;
                        }

                        if (score >= 120) {
                            candidates.push({
                                value: number,
                                score
                            });
                        }
                    }
                } else if (value && typeof value === 'object') {
                    visit(
                        value,
                        path.concat(key),
                        depth + 1
                    );
                }
            }
        }

        visit(payload, [], 0);

        candidates.sort((a, b) => {
            return (
                b.score - a.score ||
                b.value - a.value
            );
        });

        return candidates[0]?.value || 0;
    }

    function recordAlbumCatalogResponse(value, payload, status = 0) {

            id: ++albumCatalogTracker.seq,
            ...meta,
            status: Number(status || 0),
            albums,
            total: extractAlbumTotal(
                payload,
                meta.url

            completedAt: Date.now()
        });

        if (albumCatalogTracker.responses.length > 100) {
            albumCatalogTracker.responses.splice(

            );
        }

        console.debug(
            '[TME] 捕获到疑似专辑目录接口',
            meta.pathname,
            `专辑候选=${albums.length}`
        );
    }

    function collectTrackedAlbumCatalog(afterId = 0, expectedCount = 0) {
        const responses = albumCatalogTracker.responses
            .filter(item => item.id > afterId)
            .sort((a, b) => a.id - b.id);
        const source = responses.length
            ? responses
            : albumCatalogTracker.responses.slice(-20);
        const merged = [];
        const byKey = new Map();
        let sequence = 0;
        let totalHint = 0;

        for (const response of source) {
            if (response.total > totalHint) {
                totalHint = response.total;
            }

            for (const item of response.albums) {
                const key = item.albumId
                    ? `id:${item.albumId}`
                    : `name:${compact(item.name)}`;
                const old = byKey.get(key);
                const candidate = {
                    ...item,
                    sequence: sequence++,
                    sourceUrl: response.url
                };

                if (
                    !old ||
                    candidate.score > old.score
                ) {
                    byKey.set(key, candidate);
                }
            }
        }

        merged.push(...byKey.values());

        let selected = merged;

        if (
            expectedCount > 0 &&
            merged.length > expectedCount
        ) {
            const top = [...merged]
                .sort((a, b) => {
                    return (
                        b.score - a.score ||
                        a.sequence - b.sequence
                    );
                })
                .slice(0, expectedCount);
            const keep = new Set(top);

            selected = merged.filter(item => keep.has(item));
        }

        selected.sort((a, b) => a.sequence - b.sequence);

        const names = [];
        const nameSeen = new Set();

        for (const item of selected) {
            const key = compact(item.name);

            if (!key || nameSeen.has(key)) {
                continue;
            }

            nameSeen.add(key);
            names.push(item.name);
        }

        return {
            names,
            entries: selected,
            totalHint,
            responseCount: source.length,
            urls: Array.from(
                new Set(
                    source.map(item => item.pathname)
                )
            )
        };
    }

    async function waitTrackedAlbumCatalog(
        afterId,
        expectedCount = 0,
        timeout = 4200
    ) {
        const start = Date.now();
        let best = collectTrackedAlbumCatalog(
            afterId,
            expectedCount
        );
        let stableSince = 0;
        let previousCount = -1;

        while (Date.now() - start < timeout) {
            const current = collectTrackedAlbumCatalog(
                afterId,
                expectedCount
            );

            if (current.names.length >= best.names.length) {
                best = current;
            }

            const target =
                expectedCount ||
                current.totalHint ||
                0;

            if (
                target > 0 &&
                current.names.length >= target
            ) {
                return current;
            }

            if (
                current.names.length > 0 &&
                current.names.length === previousCount
            ) {
                if (!stableSince) {
                    stableSince = Date.now();
                }

                if (
                    !target &&
                    Date.now() - stableSince >= 900
                ) {
                    return current;
                }
            } else {
                stableSince = 0;
                previousCount = current.names.length;
            }

            await sleep(180);
        }

        return best;
    }



    function catalogPaginationInfo(response) {
        if (!response?.url) {
            return null;
        }

        let url;

        try {
            url = new URL(response.url, location.href);
        } catch (error) {
            return null;
        }

        const keys = Array.from(url.searchParams.keys());
        const findKey = patterns => keys.find(key => {
            return patterns.some(pattern => pattern.test(key));
        }) || '';

        const pageKey = findKey([
            /^page$/i,
            /^pageNum$/i,
            /^pageNo$/i,
            /^pageIndex$/i,
            /^current$/i,
            /^currentPage$/i
        ]);
        const offsetKey = findKey([
            /^offset$/i,
            /^start$/i,
            /^startIndex$/i,
            /^from$/i
        ]);
        const sizeKey = findKey([
            /^pageSize$/i,
            /^size$/i,
            /^limit$/i,
            /^rows$/i,
            /^perPage$/i
        ]);

        if (!pageKey && !offsetKey) {
            return null;
        }

        const mode = offsetKey ? 'offset' : 'page';
        const key = offsetKey || pageKey;
        const raw = Number(url.searchParams.get(key));
        const current = Number.isFinite(raw) ? raw : (mode === 'offset' ? 0 : 1);
        const rawSize = sizeKey ? Number(url.searchParams.get(sizeKey)) : NaN;
        const size = Number.isFinite(rawSize) && rawSize > 0
            ? rawSize
            : Math.max(1, response.albums?.length || 20);

        return {
            url,
            mode,
            key,
            sizeKey,
            current,
            size
        };
    }

    function refreshCatalogCacheBusters(url) {
        if (url.searchParams.has('timestamp')) {
            const old = url.searchParams.get('timestamp') || '';
            url.searchParams.set(
                'timestamp',
                String(old.length >= 13 ? Date.now() : Math.floor(Date.now() / 1000))
            );
        }

        if (url.searchParams.has('_')) {
            url.searchParams.set('_', String(Date.now()));
        }

        if (url.searchParams.has('nonceStr')) {
            const old = url.searchParams.get('nonceStr') || '';
            url.searchParams.set('nonceStr', randomNonce(old.length || 6));
        }
    }

    async function expandAlbumCatalogFromCapturedApi(afterId = 0, expectedCount = 0) {
        /*
         * 9.6.0：如果下拉框后台接口本身带分页参数，直接复用它逐页读取。
         * 这样大量专辑不需要依赖浏览器的虚拟滚动层，跨电脑、缩放和DPI更稳定。
         * 只有已经捕获到真实同源目录请求，并且URL里明确存在分页参数时才启用；
         * 无分页参数或接口拒绝时立即退出，继续走DOM兜底，不猜接口地址。
         */
        if (!nativeFetch) {
            return collectTrackedAlbumCatalog(afterId, expectedCount);
        }

        const sources = albumCatalogTracker.responses
            .filter(item => item.id > afterId && item.albums?.length >= 2)
            .sort((a, b) => {
                return (
                    Number(b.total || 0) - Number(a.total || 0) ||
                    Number(b.albums?.length || 0) - Number(a.albums?.length || 0) ||
                    b.id - a.id
                );
            });

        let source = null;
        let info = null;

        for (const candidate of sources) {
            const parsed = catalogPaginationInfo(candidate);

            if (parsed) {
                source = candidate;
                info = parsed;
                break;
            }
        }

        if (!source || !info) {
            return collectTrackedAlbumCatalog(afterId, expectedCount);
        }

        let collected = collectTrackedAlbumCatalog(afterId, expectedCount);
        let target = expectedCount || source.total || collected.totalHint || 0;
        let cursor = info.current;
        let previousNewCount = -1;
        const maxRequests = 120;

        setStatus(
            target
                ? `检测到专辑目录分页接口，正在直接读取全部 ${target} 个专辑……`
                : '检测到专辑目录分页接口，正在直接逐页读取全部专辑……'
        );

        for (let requestIndex = 0; requestIndex < maxRequests; requestIndex += 1) {
            if (target > 0 && collected.names.length >= target) {
                break;
            }

            const nextUrl = new URL(info.url.href);

            if (info.mode === 'offset') {
                cursor += info.size;
            } else {
                cursor += 1;
            }

            nextUrl.searchParams.set(info.key, String(cursor));
            refreshCatalogCacheBusters(nextUrl);

            try {
                const response = await nativeFetch(nextUrl.href, {
                    method: 'GET',
                    credentials: 'include',
                    cache: 'no-store',
                    headers: {
                        Accept: 'application/json, text/plain, */*'
                    }
                });

                if (!response.ok) {
                    console.debug('[TME] 专辑目录分页接口停止：HTTP', response.status);
                    break;
                }

                const decoded = decodeJsonText(await response.text());

                if (decoded.parseError || !decoded.payload || typeof decoded.payload !== 'object') {
                    break;
                }

                const pageAlbums = extractAlbumCatalog(decoded.payload, nextUrl.href);

                if (!pageAlbums.length) {
                    break;
                }

                const before = collected.names.length;
                recordAlbumCatalogResponse(nextUrl.href, decoded.payload, response.status);
                collected = collectTrackedAlbumCatalog(afterId, target);
                const added = collected.names.length - before;
                const totalHint = extractAlbumTotal(decoded.payload, nextUrl.href);

                if (totalHint > target) {
                    target = totalHint;
                }

                setStatus(
                    target
                        ? `后台分页读取专辑：${collected.names.length}/${target}`
                        : `后台分页读取专辑：已识别 ${collected.names.length} 个`
                );

                if (added <= 0 || added === previousNewCount && pageAlbums.length === 0) {
                    break;
                }

                previousNewCount = added;
                await sleep(70);
            } catch (error) {
                console.debug('[TME] 专辑目录分页接口直读失败，回退DOM扫描', error.message);
                break;
            }
        }

        return collectTrackedAlbumCatalog(afterId, target);
    }

    function textCandidates(root) {
        const selector = [
            '[title]',
            '[class*="title"]',
            '[class*="Title"]',
            '[class*="name"]',
            '[class*="Name"]',
            'h1',
            'h2',
            'h3',
            'h4',
            'strong',
            'b',
            'p',
            'span',
            'div'
        ].join(',');

        const result = [];

        for (
            const element of
            all(
                selector,
                root
            )
        ) {
            const values = [
                clean(
                    element.getAttribute(
                        'title'
                    )
                ),

                clean(
                    element.innerText ||
                    element.textContent
                )
            ].filter(Boolean);

            for (const value of values) {
                if (
                    !validAlbumName(value)
                ) {
                    continue;
                }

                const childHasSameText =
                    Array.from(
                        element.children
                    ).some(child => {
                        return (
                            visible(child) &&
                            clean(
                                child.innerText ||
                                child.textContent
                            ) === value
                        );
                    });

                if (childHasSameText) {
                    continue;
                }

                const style =
                    getComputedStyle(
                        element
                    );

                const rect =
                    element
                        .getBoundingClientRect();

                let score = 0;

                if (
                    /title|name|album/i.test(
                        String(
                            element.className ||
                            ''
                        )
                    )
                ) {
                    score += 45;
                }

                const size =
                    parseFloat(
                        style.fontSize
                    ) || 0;

                const weight =
                    parseInt(
                        style.fontWeight,
                        10
                    ) || 400;

                if (size >= 18) {
                    score += 28;
                } else if (
                    size >= 15
                ) {
                    score += 18;
                }

                if (weight >= 600) {
                    score += 18;
                }

                if (
                    value.length >= 8 &&
                    value.length <= 180
                ) {
                    score += 15;
                }

                if (
                    value.includes('|')
                ) {
                    score += 10;
                }

                result.push({
                    value,
                    score,

                    area:
                        rect.width *
                        rect.height,

                    top:
                        rect.top
                });
            }
        }

        result.sort((a, b) => {
            return (
                b.score -
                a.score ||
                a.area -
                b.area ||
                a.top -
                b.top
            );
        });

        return result;
    }

    function bestTitle(root) {
        const titleAttributes =
            Array.from(
                root.querySelectorAll(
                    '[title]'
                )
            )
                .map(element => {
                    return clean(
                        element.getAttribute(
                            'title'
                        )
                    );
                })
                .filter(
                    validAlbumName
                );

        if (titleAttributes.length) {
            return titleAttributes
                .sort((a, b) => {
                    return (
                        b.length -
                        a.length
                    );
                })[0];
        }

        return (
            textCandidates(
                root
            )[0]?.value ||
            ''
        );
    }

    function findSelector() {
        /*
         * 9.6.0：跨电脑/缩放兼容。
         * 旧版要求选择框宽度>=350且必须已经渲染img；在另一台电脑、浏览器缩放、
         * DPI或窗口较窄时，这两个条件都可能不成立，导致“根本无法开始扫描”。
         * 现在主要依靠“选择专辑”标题附近的位置 + 可点击/下拉特征评分，图片只加分不再硬性要求。
         */
        const heading = findHeading();
        const headingRect = heading?.getBoundingClientRect();
        const seen = new Set();
        const candidates = [];
        const selectors = [
            '[role="combobox"]',
            '[aria-haspopup="listbox"]',
            'button',
            '[role="button"]',
            '[class*="select"]',
            '[class*="Select"]',
            '[class*="album"]',
            '[class*="Album"]',
            'div'
        ];

        for (const selector of selectors) {
            for (const raw of all(selector)) {
                if (seen.has(raw)) {
                    continue;
                }

                seen.add(raw);

                const element = raw.closest(
                    'button,[role="button"],[role="combobox"]'
                ) || raw;

                if (!visible(element) || inPanel(element)) {
                    continue;
                }

                const rect = element.getBoundingClientRect();

                /* 不再写死350px，适配不同缩放和屏幕。 */
                if (
                    rect.width < Math.max(150, Math.min(220, window.innerWidth * 0.16)) ||
                    rect.height < 42 ||
                    rect.height > 240
                ) {
                    continue;
                }

                const rawText = clean(element.innerText || element.textContent);
                const title = bestTitle(element) || rawText;
                const noApproved = /暂无(?:已)?(?:通过审核|审核通过)(?:的)?专辑|暂无专辑通过审核/.test(
                    compact(rawText)
                );

                if (!validAlbumName(title) && !noApproved) {
                    continue;
                }

                if (headingRect) {
                    const maxDistance = Math.max(300, Math.min(520, window.innerHeight * 0.55));

                    if (
                        rect.top < headingRect.bottom - 40 ||
                        rect.top > headingRect.bottom + maxDistance
                    ) {
                        continue;
                    }
                }

                let score = 60;
                const classText = String(element.className || '').toLowerCase();

                if (element.getAttribute('role') === 'combobox') {
                    score += 70;
                }

                if (element.getAttribute('aria-haspopup') === 'listbox') {
                    score += 65;
                }

                if (element.matches('button,[role="button"]')) {
                    score += 28;
                }

                if (/album|select|work|program|dropdown|picker/.test(classText)) {
                    score += 36;
                }

                if (element.querySelector('img')) {
                    score += 32;
                }

                if (element.querySelector('svg,[class*="arrow"],[class*="down"]')) {
                    score += 35;
                }

                if (noApproved) {
                    score += 55;
                }

                if (rect.width >= 260) {
                    score += 15;
                }

                if (headingRect) {
                    score += Math.max(0, 80 - Math.abs(rect.top - headingRect.bottom) / 4);
                }

                candidates.push({
                    element,
                    title: validAlbumName(title) ? title : '',
                    score,
                    top: rect.top,
                    area: rect.width * rect.height
                });
            }
        }

        candidates.sort((a, b) => {
            return (
                b.score - a.score ||
                a.top - b.top ||
                b.area - a.area
            );
        });

        return candidates[0] || null;
    }

    function humanClick(element) {
        if (!element) {
            return false;
        }

        /*
         * 9.5.5 安全保护：抓取脚本任何流程都不需要创建专辑。
         * 当“暂无专辑通过审核”出现时，旧版 closeDropdown 的坐标兜底
         * 可能正好落在右侧“创建专辑”按钮上，从而误跳到创建页。
         * 因此所有脚本触发的点击，在执行前先拦截创建专辑入口。
         */
        if (
            (processing || task.running) &&
            isCreateAlbumAction(element)
        ) {
            console.warn(
                '[TME] 已阻止脚本误点“创建专辑/创建新专辑”',
                element
            );
            return false;
        }

        const options = {
            bubbles: true,
            cancelable: true,
            view: window,
            button: 0,
            buttons: 1
        };

        try {
            element.dispatchEvent(
                new PointerEvent(
                    'pointerdown',
                    options
                )
            );
        } catch (error) {
            console.debug(error);
        }

        element.dispatchEvent(
            new MouseEvent(
                'mousedown',
                options
            )
        );

        element.dispatchEvent(
            new MouseEvent(
                'mouseup',
                {
                    ...options,
                    buttons: 0
                }
            )
        );

        try {
            element.dispatchEvent(
                new PointerEvent(
                    'pointerup',
                    {
                        ...options,
                        buttons: 0
                    }
                )
            );
        } catch (error) {
            console.debug(error);
        }

        element.click();
        return true;
    }

    function clickPoint(x, y) {
        const element =
            document.elementFromPoint(
                x,
                y
            );

        if (element) {
            return humanClick(element);
        }

        return false;
    }

    function findScrollBox(
        root,
        input,
        createRow
    ) {
        const rootRect =
            root.getBoundingClientRect();

        const inputRect =
            input.getBoundingClientRect();

        const createTop =
            createRow
                ? createRow
                    .getBoundingClientRect()
                    .top
                : rootRect.bottom;

        const listTop =
            Math.max(
                inputRect.bottom - 25,
                rootRect.top
            );

        const listBottom =
            Math.max(
                listTop + 80,
                Math.min(
                    createTop + 25,
                    window.innerHeight + 80
                )
            );

        /*
         * 9.5.8沿用：绝不再把下拉框里的全部DOM节点塞进候选池。
         * 大量专辑时 root.querySelectorAll('*') 会一次遍历成千上万个节点，
         * 再逐个执行 getComputedStyle/getBoundingClientRect，容易直接堵塞浏览器主线程。
         * 真实滚动层一定会出现在下拉根节点、输入框祖先或列表可视区域命中节点的祖先链上，
         * 因此只检查这些少量节点即可。
         */
        const pool = new Set([root]);

        /*
         * 9.1.1起：旧逻辑只允许“下拉根节点的子元素”成为滚动层。
         * TME 当前页面在部分账号/窗口尺寸下会把真实滚动层放在更外层，
         * 因此前面已经能识别专辑条目，但这里仍会误判为“没有滚动区域”。
         * 现在把输入框、下拉根节点的祖先，以及列表区域命中的节点一起纳入候选。
         */
        for (const origin of [root, input]) {
            let current = origin;

            for (
                let level = 0;
                level < 10 &&
                current &&
                current !== document.body &&
                current !== document.documentElement;
                level += 1
            ) {
                pool.add(current);
                current = current.parentElement;
            }
        }

        const sampleX =
            rootRect.left +
            rootRect.width / 2;

        const sampleYs = [
            listTop + 18,
            listTop +
                (listBottom - listTop) * 0.35,
            listTop +
                (listBottom - listTop) * 0.65,
            listBottom - 18
        ];

        for (const y of sampleYs) {
            if (
                y < 0 ||
                y > window.innerHeight
            ) {
                continue;
            }

            for (
                const element of
                document.elementsFromPoint(
                    sampleX,
                    y
                )
            ) {
                pool.add(element);

                let current =
                    element.parentElement;

                for (
                    let level = 0;
                    level < 6 &&
                    current &&
                    current !== document.body &&
                    current !== document.documentElement;
                    level += 1
                ) {
                    pool.add(current);
                    current = current.parentElement;
                }
            }
        }

        const candidates = [];

        for (const element of pool) {
            if (
                !(element instanceof Element) ||
                element === document.body ||
                element === document.documentElement ||
                !visible(element) ||
                inPanel(element)
            ) {
                continue;
            }

            const rect =
                element
                    .getBoundingClientRect();

            const style =
                getComputedStyle(element);

            const range =
                element.scrollHeight -
                element.clientHeight;

            if (range <= 5) {
                continue;
            }

            const overlapWidth =
                Math.max(
                    0,
                    Math.min(
                        rect.right,
                        rootRect.right + 35
                    ) -
                    Math.max(
                        rect.left,
                        rootRect.left - 35
                    )
                );

            const overlapHeight =
                Math.max(
                    0,
                    Math.min(
                        rect.bottom,
                        listBottom
                    ) -
                    Math.max(
                        rect.top,
                        listTop
                    )
                );

            if (
                overlapWidth <
                    Math.min(
                        120,
                        rootRect.width * 0.35
                    ) ||
                overlapHeight < 30
            ) {
                continue;
            }

            let score =
                Math.min(
                    range,
                    1600
                );

            const overflowText =
                `${style.overflowY} ${style.overflow}`;

            if (
                /auto|scroll/.test(
                    overflowText
                )
            ) {
                score += 1800;
            }

            if (
                /list|scroll|menu|option|content|virtual|holder|dropdown/i.test(
                    String(
                        element.className ||
                        ''
                    )
                )
            ) {
                score += 800;
            }

            if (element === root) {
                score += 420;
            } else if (root.contains(element)) {
                score += 360;
            } else if (element.contains(root)) {
                score += 220;
            }

            if (
                rect.top >=
                    inputRect.bottom - 35 &&
                rect.top <
                    createTop
            ) {
                score += 360;
            }

            if (
                rect.bottom <=
                    createTop + 40
            ) {
                score += 260;
            }

            if (
                rect.width >=
                    rootRect.width * 0.58 &&
                rect.width <=
                    rootRect.width * 1.45
            ) {
                score += 300;
            }

            if (
                rect.height >
                    window.innerHeight * 0.92
            ) {
                score -= 900;
            }

            if (
                rect.width >
                    rootRect.width * 2.2
            ) {
                score -= 650;
            }

            candidates.push({
                element,
                score,
                range,
                area:
                    rect.width *
                    rect.height
            });
        }

        candidates.sort((a, b) => {
            return (
                b.score -
                a.score ||
                b.range -
                a.range ||
                a.area -
                b.area
            );
        });

        return (
            candidates[0]?.element ||
            null
        );
    }

    function dropdownAlbumCount(context) {
        const roots = [];
        const seen = new Set();

        function addRoot(element) {
            if (
                !element ||
                seen.has(element) ||
                !(element instanceof Element)
            ) {
                return;
            }

            seen.add(element);
            roots.push(element);
        }

        addRoot(context.root);

        let current =
            context.root?.parentElement;

        for (
            let level = 0;
            level < 4 &&
            current &&
            current !== document.body;
            level += 1
        ) {
            addRoot(current);
            current = current.parentElement;
        }

        for (const root of roots) {
            const text =
                clean(
                    root.textContent ||
                    root.innerText
                );

            const matches = [
                ...text.matchAll(
                    /(?:共\s*)?(\d{1,5})\s*(?:个|张)\s*专辑/g
                )
            ];

            for (const match of matches) {
                const count =
                    Number(match[1]);

                if (
                    Number.isInteger(count) &&
                    count > 0 &&
                    count <= 20000
                ) {
                    return count;
                }
            }
        }

        return 0;
    }

    function zone(context) {
        const rootRect =
            context.root
                .getBoundingClientRect();

        const inputRect =
            context.input
                .getBoundingClientRect();

        const createRect =
            context.createRow
                ?.getBoundingClientRect();

        const scrollRect =
            context.scrollBox
                ?.getBoundingClientRect();

        return {
            left:
                Math.max(
                    rootRect.left,
                    scrollRect?.left ??
                    rootRect.left
                ),

            right:
                Math.min(
                    rootRect.right,
                    scrollRect?.right ??
                    rootRect.right
                ),

            top:
                Math.max(
                    inputRect.bottom + 5,
                    scrollRect?.top ??
                    rootRect.top
                ),

            bottom:
                Math.min(
                    createRect
                        ? createRect.top - 3
                        : rootRect.bottom - 3,

                    scrollRect?.bottom ??
                    rootRect.bottom
                )
        };
    }

    function centerIn(
        rect,
        area
    ) {
        const x =
            rect.left +
            rect.width / 2;

        const y =
            rect.top +
            rect.height / 2;

        return (
            x >= area.left &&
            x <= area.right &&
            y >= area.top &&
            y <= area.bottom
        );
    }

    function rowClickTarget(
        row,
        context
    ) {
        const area =
            zone(context);

        let current =
            row;

        let best =
            row;

        for (
            let level = 0;
            level < 8 &&
            current &&
            current !== context.root;
            level += 1
        ) {
            const rect =
                current
                    .getBoundingClientRect();

            if (
                !centerIn(
                    rect,
                    area
                )
            ) {
                break;
            }

            if (
                current.matches(
                    'button,a,li,' +
                    '[role="button"],' +
                    '[role="option"],' +
                    '[role="menuitem"]'
                )
            ) {
                return current;
            }

            if (
                rect.width >=
                (
                    area.right -
                    area.left
                ) * 0.58 &&
                rect.height >= 38 &&
                rect.height <= 190
            ) {
                best =
                    current;
            }

            current =
                current.parentElement;
        }

        return best;
    }

    function collectRows(context) {
        const area =
            zone(context);

        const map =
            new Map();

        function add(
            row,
            bonus = 0
        ) {
            if (
                !row ||
                !visible(row)
            ) {
                return;
            }

            const rect =
                row
                    .getBoundingClientRect();

            if (
                !centerIn(
                    rect,
                    area
                ) ||
                rect.width <
                (
                    area.right -
                    area.left
                ) * 0.5 ||
                rect.height < 30 ||
                rect.height > 220
            ) {
                return;
            }

            const name =
                bestTitle(row);

            if (
                !validAlbumName(name)
            ) {
                return;
            }

            const target =
                rowClickTarget(
                    row,
                    context
                );

            const targetRect =
                target
                    .getBoundingClientRect();

            let score =
                bonus;

            if (
                target.querySelector(
                    'img'
                )
            ) {
                score += 45;
            }

            if (
                target.matches(
                    'li,button,a,' +
                    '[role="button"],' +
                    '[role="option"]'
                )
            ) {
                score += 25;
            }

            if (
                /item|option|album|card|virtual/i.test(
                    String(
                        target.className ||
                        ''
                    )
                )
            ) {
                score += 25;
            }

            const old =
                map.get(name);

            const targetArea =
                targetRect.width *
                targetRect.height;

            if (
                !old ||
                score >
                old.score ||
                (
                    score ===
                    old.score &&
                    targetArea <
                    old.area
                )
            ) {
                map.set(
                    name,
                    {
                        name,
                        element:
                            target,

                        score,

                        area:
                            targetArea,

                        top:
                            targetRect.top
                    }
                );
            }
        }

        for (
            const image of
            all(
                'img',
                context.root
            )
        ) {
            const rect =
                image
                    .getBoundingClientRect();

            if (
                !centerIn(
                    rect,
                    area
                )
            ) {
                continue;
            }

            let current =
                image.parentElement;

            for (
                let level = 0;
                level < 9 &&
                current &&
                current !== context.root;
                level += 1
            ) {
                const currentRect =
                    current
                        .getBoundingClientRect();

                if (
                    currentRect.width >=
                    (
                        area.right -
                        area.left
                    ) * 0.55 &&
                    currentRect.height >= 40 &&
                    currentRect.height <= 190
                ) {
                    add(
                        current,
                        40
                    );
                }

                current =
                    current.parentElement;
            }
        }

        const explicit = [
            '[role="option"]',
            '[data-option-index]',
            'li',

            '[class*="album-item"]',
            '[class*="albumItem"]',

            '[class*="list-item"]',
            '[class*="listItem"]',

            '[class*="option-item"]',
            '[class*="optionItem"]',

            '[class*="virtual"]',
            '[class*="Virtual"]',

            '[class*="item"]',
            '[class*="Item"]'
        ].join(',');

        for (
            const element of
            all(
                explicit,
                context.root
            )
        ) {
            add(
                element,
                15
            );
        }

        /*
         * 文字兜底。
         * 虚拟列表滚动时，部分图片节点不更新，
         * 但标题文字已经改变。
         */
        for (
            const element of
            all(
                '[title],span,div,p,strong,b',
                context.root
            )
        ) {
            const rect =
                element
                    .getBoundingClientRect();

            if (
                !centerIn(
                    rect,
                    area
                )
            ) {
                continue;
            }

            const value =
                clean(
                    element.getAttribute(
                        'title'
                    )
                ) ||
                clean(
                    element.innerText ||
                    element.textContent
                );

            if (
                !validAlbumName(value)
            ) {
                continue;
            }

            const childHasSameText =
                Array.from(
                    element.children
                ).some(child => {
                    return (
                        visible(child) &&
                        clean(
                            child.innerText ||
                            child.textContent
                        ) === value
                    );
                });

            if (childHasSameText) {
                continue;
            }

            let current =
                element;

            for (
                let level = 0;
                level < 7 &&
                current &&
                current !== context.root;
                level += 1
            ) {
                const currentRect =
                    current
                        .getBoundingClientRect();

                if (
                    currentRect.width >=
                    (
                        area.right -
                        area.left
                    ) * 0.55 &&
                    currentRect.height >= 35 &&
                    currentRect.height <= 190
                ) {
                    add(
                        current,
                        10
                    );
                }

                current =
                    current.parentElement;
            }
        }

        return Array.from(
            map.values()
        ).sort((a, b) => {
            return (
                a.top -
                b.top
            );
        });
    }


    /*
     * 9.5.8沿用：大量专辑扫描只看当前可视区域。
     * 旧版 collectRows 会对整个下拉DOM执行多轮 querySelectorAll，专辑数多时非常重。
     * 现在按列表视口每隔约22px取样，用 elementsFromPoint 找当前屏幕真正可见的行，
     * 每次只处理几十个节点。只有完全取不到可见行时，才偶尔回退一次旧版 collectRows。
     */
    function collectScanRowsFast(context) {
        const area = zone(context);
        const width = area.right - area.left;
        const height = area.bottom - area.top;

        if (
            !Number.isFinite(width) ||
            !Number.isFinite(height) ||
            width < 120 ||
            height < 40
        ) {
            return [];
        }

        const map = new Map();
        const seenElements = new Set();
        const xs = [
            area.left + width * 0.18,
            area.left + width * 0.48,
            area.left + width * 0.78
        ];
        const stepY = Math.max(18, Math.min(26, Math.floor(height / 15) || 22));

        function remember(row, bonus = 0) {
            if (
                !row ||
                !(row instanceof Element) ||
                seenElements.has(row) ||
                !visible(row) ||
                inPanel(row)
            ) {
                return;
            }

            seenElements.add(row);

            const rect = row.getBoundingClientRect();

            if (
                rect.width < width * 0.45 ||
                rect.height < 30 ||
                rect.height > 220 ||
                rect.bottom < area.top ||
                rect.top > area.bottom
            ) {
                return;
            }

            const name = bestTitle(row);

            if (!validAlbumName(name)) {
                return;
            }

            const key = compact(name);
            const old = map.get(key);
            const score = bonus +
                (row.querySelector('img') ? 35 : 0) +
                (/item|option|album|card|virtual/i.test(String(row.className || '')) ? 20 : 0);

            if (!old || score > old.score) {
                map.set(key, {
                    name,
                    element: row,
                    score,
                    area: rect.width * rect.height,
                    top: rect.top
                });
            }
        }

        for (let y = area.top + 8; y <= area.bottom - 8; y += stepY) {
            if (y < 0 || y > window.innerHeight) {
                continue;
            }

            for (const x of xs) {
                if (x < 0 || x > window.innerWidth) {
                    continue;
                }

                const stack = document.elementsFromPoint(x, y).slice(0, 6);

                for (const seed of stack) {
                    let current = seed;

                    for (
                        let level = 0;
                        level < 9 &&
                        current &&
                        current !== document.body &&
                        current !== document.documentElement;
                        level += 1
                    ) {
                        const rect = current.getBoundingClientRect();

                        if (
                            rect.width >= width * 0.45 &&
                            rect.height >= 30 &&
                            rect.height <= 220 &&
                            rect.bottom >= area.top &&
                            rect.top <= area.bottom
                        ) {
                            remember(current, Math.max(0, 40 - level * 4));
                        }

                        if (current === context.root) {
                            break;
                        }

                        current = current.parentElement;
                    }
                }
            }
        }

        return Array.from(map.values())
            .sort((a, b) => Number(a.top || 0) - Number(b.top || 0));
    }

    function collectScanRows(context, allowHeavyFallback = false) {
        const fast = collectScanRowsFast(context);

        if (fast.length || !allowHeavyFallback) {
            return fast;
        }

        /* 只有轻量扫描完全失败时才执行一次旧版DOM兜底。 */
        return collectRows(context);
    }

    function signature(context) {
        return collectScanRows(context)
            .map(item => compact(item.name))
            .join('||');
    }

    /*
     * 从“已经看得见的专辑行”反向寻找最近的真实滚动祖先。
     * 这比只根据宽高、overflow 和页面坐标猜滚动层更可靠，尤其适用于
     * 专辑很多、列表使用虚拟滚动的账号。
     */
    function findRowScrollBox(context) {
        const rows = collectScanRows({
            ...context,
            scrollBox: null
        }, true);

        if (!rows.length) {
            return null;
        }

        const rootRect = context.root.getBoundingClientRect();
        const candidates = new Map();

        for (const row of rows.slice(-5)) {
            let current = row.element;

            for (
                let level = 0;
                level < 12 &&
                current &&
                current !== document.body &&
                current !== document.documentElement;
                level += 1
            ) {
                if (current instanceof Element && visible(current) && !inPanel(current)) {
                    const range = current.scrollHeight - current.clientHeight;

                    if (range > 5) {
                        const rect = current.getBoundingClientRect();
                        const style = getComputedStyle(current);
                        const overflowText = `${style.overflowY} ${style.overflow}`;
                        const overlapWidth = Math.max(
                            0,
                            Math.min(rect.right, rootRect.right + 30) -
                            Math.max(rect.left, rootRect.left - 30)
                        );
                        const overlapHeight = Math.max(
                            0,
                            Math.min(rect.bottom, rootRect.bottom + 30) -
                            Math.max(rect.top, rootRect.top - 30)
                        );

                        if (overlapWidth > 100 && overlapHeight > 40) {
                            let score = 0;

                            if (/auto|scroll/.test(overflowText)) {
                                score += 1800;
                            }

                            if (/list|scroll|menu|option|content|virtual|holder|dropdown/i.test(String(current.className || ''))) {
                                score += 900;
                            }

                            if (context.root.contains(current)) {
                                score += 550;
                            } else if (current.contains(context.root)) {
                                score += 100;
                            }

                            score += Math.max(0, 500 - level * 55);
                            score += Math.min(range, 1400) / 4;

                            if (rect.height > window.innerHeight * 0.9) {
                                score -= 1100;
                            }

                            const old = candidates.get(current);

                            if (!old || score > old.score) {
                                candidates.set(current, {
                                    element: current,
                                    score,
                                    level,
                                    range
                                });
                            }
                        }
                    }
                }

                current = current.parentElement;
            }
        }

        return Array.from(candidates.values())
            .sort((a, b) => {
                return (
                    b.score - a.score ||
                    a.level - b.level ||
                    b.range - a.range
                );
            })[0]?.element || null;
    }



    function scrollBoxCandidates(context) {
        const rows = collectScanRows({
            ...context,
            scrollBox: null
        }, true);
        const pool = new Set();

        for (const origin of [context.root, context.input]) {
            let current = origin;

            for (
                let level = 0;
                level < 12 &&
                current &&
                current !== document.body &&
                current !== document.documentElement;
                level += 1
            ) {
                pool.add(current);
                current = current.parentElement;
            }
        }

        for (const row of rows.slice(0, 10)) {
            let current = row.element;

            for (
                let level = 0;
                level < 14 &&
                current &&
                current !== document.body &&
                current !== document.documentElement;
                level += 1
            ) {
                pool.add(current);
                current = current.parentElement;
            }
        }

        const rootRect = context.root.getBoundingClientRect();
        const result = [];

        for (const element of pool) {
            if (!(element instanceof Element) || !visible(element) || inPanel(element)) {
                continue;
            }

            const range = Number(element.scrollHeight || 0) - Number(element.clientHeight || 0);

            if (range <= 5) {
                continue;
            }

            const rect = element.getBoundingClientRect();
            const overlapWidth = Math.max(
                0,
                Math.min(rect.right, rootRect.right + 40) -
                Math.max(rect.left, rootRect.left - 40)
            );

            if (overlapWidth < Math.min(100, rootRect.width * 0.3)) {
                continue;
            }

            const style = getComputedStyle(element);
            let score = Math.min(range, 2400) / 4;

            if (/auto|scroll/.test(`${style.overflowY} ${style.overflow}`)) {
                score += 1500;
            }

            if (/list|scroll|menu|option|content|virtual|holder|dropdown/i.test(String(element.className || ''))) {
                score += 900;
            }

            if (context.root.contains(element)) {
                score += 500;
            }

            if (rect.height > 90 && rect.height < window.innerHeight * 0.8) {
                score += 350;
            }

            result.push({ element, score, range });
        }

        return result
            .sort((a, b) => b.score - a.score || b.range - a.range)
            .slice(0, 12)
            .map(item => item.element);
    }

    async function probeTrueScrollBox(
        context,
        requireVerified = false,
        options = {}
    ) {
        const forwardOnly = Boolean(options.forwardOnly);
        /*
         * 静态“看起来像滚动层”在不同电脑上不可靠。
         * 这里对少量候选容器做一次真实试滚，只有“可见专辑签名发生变化”的容器
         * 才被认定为真正的专辑列表滚动层。
         */
        const originalBox = context.scrollBox;
        const baselineContext = {
            ...context,
            scrollBox: originalBox || null
        };
        const baseline = signature(baselineContext);
        const candidates = scrollBoxCandidates(context);

        if (originalBox && !candidates.includes(originalBox)) {
            candidates.unshift(originalBox);
        }

        for (const candidate of candidates) {
            if (!candidate?.isConnected) {
                continue;
            }

            const oldTop = Number(candidate.scrollTop || 0);
            const maxTop = Math.max(0, candidate.scrollHeight - candidate.clientHeight);

            if (maxTop <= 5) {
                continue;
            }

            const step = Math.max(120, Math.floor(candidate.clientHeight * 0.78));
            let targetTop = Math.min(maxTop, oldTop + step);
            let restoreAfterProbe = false;

            /*
             * 9.6.2关键修复：扫描到底部时，只允许“向前还能滚动”的候选通过验证。
             * 旧版在已到最底部时会向上反向试滚，只要专辑签名变化就把同一个
             * 滚动层再次认定为“可继续”，从而在底部无限循环。
             */
            if (Math.abs(targetTop - oldTop) <= 1 && oldTop > 2) {
                if (forwardOnly) {
                    continue;
                }

                targetTop = Math.max(0, oldTop - step);
                restoreAfterProbe = true;
            }

            if (Math.abs(targetTop - oldTop) <= 1) {
                continue;
            }

            context.scrollBox = candidate;
            const candidateBaseline = signature(context) || baseline;
            candidate.scrollTop = targetTop;
            candidate.dispatchEvent(new Event('scroll', { bubbles: true }));
            await sleep(260);

            const next = signature(context);

            if (next && candidateBaseline && next !== candidateBaseline) {
                if (restoreAfterProbe) {
                    candidate.scrollTop = oldTop;
                    candidate.dispatchEvent(new Event('scroll', { bubbles: true }));
                    await sleep(120);
                }

                context.scrollBox = candidate;
                console.info('[TME] 已通过试滚确认真实专辑滚动层', {
                    range: maxTop,
                    oldTop,
                    probeTop: targetTop
                });
                return candidate;
            }

            candidate.scrollTop = oldTop;
            candidate.dispatchEvent(new Event('scroll', { bubbles: true }));
            await sleep(70);
        }

        const fallback = originalBox || findRowScrollBox(context) || findScrollBox(
            context.root,
            context.input,
            context.createRow
        );

        context.scrollBox = fallback || null;

        /*
         * 恢复阶段必须只接受“试滚后确实让专辑列表变化”的容器，
         * 否则同一个错误容器会被不断当成恢复成功，造成无限循环。
         */
        return requireVerified ? null : (fallback || null);
    }

    function setInput(
        input,
        value
    ) {
        const descriptor =
            Object.getOwnPropertyDescriptor(
                Object.getPrototypeOf(input),
                'value'
            );

        if (descriptor?.set) {
            descriptor.set.call(
                input,
                value
            );
        } else {
            input.value =
                value;
        }

        input.dispatchEvent(
            new Event(
                'input',
                {
                    bubbles: true
                }
            )
        );

        input.dispatchEvent(
            new Event(
                'change',
                {
                    bubbles: true
                }
            )
        );
    }

    function searchButton(root) {
        return (
            all(
                'button,a,[role="button"],span,div',
                root
            )
                .filter(element => {
                    return (
                        compact(
                            element.innerText ||
                            element.textContent
                        ) ===
                        '搜索'
                    );
                })
                .map(element => {
                    return (
                        element.closest(
                            'button,a,' +
                            '[role="button"]'
                        ) ||
                        element
                    );
                })[0] ||
            null
        );
    }

    async function searchAlbum(
        context,
        value
    ) {
        /*
         * 9.5.8：搜索阶段只做“输入 + 一次搜索触发”，不再固定等待1秒、
         * 不再每次搜索后重新扫描滚动容器，也不再同时按Enter又点击搜索按钮。
         * 搜索结果是否就绪交给 selectAlbum 的轻量轮询判断。
         */
        let liveContext = albumSearchContextUsable(context)
            ? context
            : null;

        if (!liveContext) {
            liveContext = await getAlbumSearchContext();
        }

        setInput(
            liveContext.input,
            value
        );

        await sleep(45);

        const button = searchButton(liveContext.root);

        if (button) {
            humanClick(button);
        } else {
            liveContext.input.dispatchEvent(
                new KeyboardEvent(
                    'keydown',
                    {
                        key: 'Enter',
                        code: 'Enter',
                        bubbles: true,
                        cancelable: true
                    }
                )
            );
        }

        /* 给站点一次最小渲染机会，不做长时间固定等待。 */
        await sleep(80);

        /* 只有搜索框节点真的被替换时，才执行一次完整上下文重定位。 */
        if (!albumSearchContextUsable(liveContext)) {
            const refreshed = discoverOpenAlbumSearchContext();

            if (albumSearchContextUsable(refreshed)) {
                liveContext = refreshed;
            }
        }

        preserveAlbumSearchContext(liveContext);
        return liveContext;
    }

    async function waitDropdown(
        selectorElement,
        timeout =
            CFG.openTimeout
    ) {
        const selectorRect =
            selectorElement
                .getBoundingClientRect();

        const start =
            Date.now();

        while (
            Date.now() -
            start <
            timeout
        ) {
            const input =
                findSearchInput(
                    selectorRect
                );

            if (input) {
                const root =
                    findDropdownRoot(
                        input,
                        selectorRect
                    );

                if (root) {
                    const createRow =
                        closestWideRow(
                            findCreateText(root),
                            root
                        );

                    const scrollBox =
                        findScrollBox(
                            root,
                            input,
                            createRow
                        );

                    const context = {
                        selectorElement,
                        input,
                        root,
                        createRow,
                        scrollBox
                    };

                    if (
                        createRow ||
                        scrollBox ||
                        root.querySelector('img,[role="option"],li') ||
                        /创建新?专辑|输入专辑名搜索|搜索/.test(clean(root.textContent || ''))
                    ) {
                        return context;
                    }
                }
            }

            await sleep(220);
        }

        return null;
    }

    async function openDropdown() {
        const selector =
            findSelector();

        if (!selector) {
            throw new Error(
                '没有找到“选择专辑”下方带封面的专辑选择框。'
            );
        }

        let context =
            await waitDropdown(
                selector.element,
                900
            );

        if (context) {
            return context;
        }

        selector.element
            .scrollIntoView({
                behavior: 'auto',
                block: 'center',
                inline: 'center'
            });

        await sleep(180);

        const rect =
            selector.element
                .getBoundingClientRect();

        const points = [
            [rect.left + rect.width * 0.92, rect.top + rect.height / 2],
            [rect.left + rect.width * 0.60, rect.top + rect.height / 2],
            [rect.left + rect.width * 0.30, rect.top + rect.height / 2],
            [rect.left + Math.min(70, rect.width * 0.18), rect.top + rect.height / 2]
        ];

        for (
            let index = 0;
            index < points.length;
            index += 1
        ) {
            clickPoint(
                points[index][0],
                points[index][1]
            );

            context =
                await waitDropdown(
                    selector.element,
                    index === 0
                        ? 3500
                        : 2600
                );

            if (context) {
                return context;
            }

            await sleep(250);
        }

        humanClick(
            selector.element
        );

        context =
            await waitDropdown(
                selector.element,
                3000
            );

        if (context) {
            return context;
        }

        throw new Error(
            '已自动点击专辑选择框，但没有识别到展开后的搜索框。'
        );
    }

    async function closeDropdown() {
        const options = {
            key: 'Escape',
            code: 'Escape',
            keyCode: 27,
            which: 27,
            bubbles: true,
            cancelable: true
        };

        /*
         * 优先只用键盘事件和失焦关闭。
         * 这是下架专辑场景下最安全的方式，因为页面右侧可能出现
         * “创建专辑”按钮，绝不能再用盲目坐标点击去强制收起。
         */
        for (let attempt = 0; attempt < 2; attempt += 1) {
            document.dispatchEvent(
                new KeyboardEvent(
                    'keydown',
                    options
                )
            );

            document.dispatchEvent(
                new KeyboardEvent(
                    'keyup',
                    options
                )
            );

            try {
                if (
                    document.activeElement &&
                    typeof document.activeElement.blur === 'function'
                ) {
                    document.activeElement.blur();
                }
            } catch (error) {
                console.debug(error);
            }

            await sleep(180);
        }

        const noApproved = noApprovedAlbumSignal();

        /*
         * 强规则：只要出现“暂无专辑通过审核”，禁止任何坐标兜底点击。
         * 旧版在这里重新调用 findSelector 后，可能把“暂无专辑通过审核”
         * 卡片当成专辑选择框，rect.right - 55 恰好命中“创建专辑”。
         */
        if (noApproved.active) {
            console.debug(
                '[TME] 下架状态下仅使用Escape/失焦关闭下拉框，跳过坐标点击'
            );
            return;
        }

        const selector = findSelector();

        if (!selector) {
            return;
        }

        const searchInput = findSearchInput(
            selector.element.getBoundingClientRect()
        );

        if (!searchInput) {
            return;
        }

        /*
         * 普通页面保留原来的选择框兜底，但点击前检查实际落点。
         * 如果落点是“创建专辑/创建新专辑”，直接放弃，不执行点击。
         */
        const rect = selector.element.getBoundingClientRect();
        const x = rect.right - 55;
        const y = rect.top + rect.height / 2;
        const pointTarget = document.elementFromPoint(x, y);

        if (
            !pointTarget ||
            isCreateAlbumAction(pointTarget)
        ) {
            console.debug(
                '[TME] 关闭下拉框的兜底落点不安全，已取消坐标点击'
            );
            return;
        }

        clickPoint(x, y);
        await sleep(250);
    }

    async function waitListChange(
        context,
        oldSignature,
        oldTop,
        box
    ) {
        const start = Date.now();
        let lastSignature = oldSignature;

        while (Date.now() - start < CFG.scanSettleTimeout) {
            const newSignature = signature(context);
            lastSignature = newSignature;

            /*
             * 9.5.6关键修复：不能再把“scrollTop发生变化”本身当成列表成功翻页。
             * 如果滚错了外层容器，scrollTop也会变化，但专辑行根本不会更新。
             * 只有可见专辑签名真的变化，才算滚动成功。
             */
            if (
                newSignature &&
                newSignature !== oldSignature
            ) {
                return {
                    changed: true,
                    signature: newSignature,
                    topChanged: Math.abs(Number(box?.scrollTop || 0) - oldTop) > 1
                };
            }

            await sleep(90);
        }

        return {
            changed: false,
            signature: lastSignature,
            topChanged: Math.abs(Number(box?.scrollTop || 0) - oldTop) > 1
        };
    }

    async function scrollList(context) {
        let box = context.scrollBox;

        if (!box) {
            return {
                oldTop: 0,
                newTop: 0,
                oldSignature: signature(context),
                newSignature: signature(context),
                changed: false,
                boxChanged: false
            };
        }

        const oldSignature = signature(context);
        const firstOldTop = Number(box.scrollTop || 0);

        async function move(targetBox, multiplier = 1) {
            const oldTop = Number(targetBox.scrollTop || 0);
            const maxTop = Math.max(
                0,
                targetBox.scrollHeight - targetBox.clientHeight
            );
            const step = Math.max(
                110,
                Math.floor(
                    targetBox.clientHeight *
                    CFG.scanStepRatio *
                    multiplier
                )
            );

            targetBox.scrollTop = Math.min(maxTop, oldTop + step);
            targetBox.dispatchEvent(new Event('scroll', { bubbles: true }));

            const settled = await waitListChange(
                context,
                oldSignature,
                oldTop,
                targetBox
            );

            return {
                oldTop,
                newTop: Number(targetBox.scrollTop || 0),
                maxTop,
                settled
            };
        }

        let result = await move(box, 1);
        let boxChanged = false;

        if (!result.settled.changed) {
            const rowBox = await probeTrueScrollBox(context, true);

            if (
                rowBox &&
                rowBox !== box
            ) {
                console.info(
                    '[TME] 大列表扫描重新定位真实滚动层',
                    {
                        oldRange: box.scrollHeight - box.clientHeight,
                        newRange: rowBox.scrollHeight - rowBox.clientHeight
                    }
                );

                context.scrollBox = rowBox;
                box = rowBox;
                boxChanged = true;
                result = await move(box, 1.25);
            }
        }

        /*
         * 已经是正确滚动层但虚拟列表仍没有刷新时，再做一次较大的跳跃，
         * 避免大量专辑账号每次只移动很小距离。
         */
        if (
            !result.settled.changed &&
            result.newTop < result.maxTop - 2
        ) {
            result = await move(box, 1.9);
        }

        return {
            oldTop: firstOldTop,
            newTop: Number(box.scrollTop || 0),
            oldSignature,
            newSignature: signature(context),
            changed: Boolean(result.settled.changed),
            boxChanged
        };
    }

    function appendAlbums(names) {
        const existing =
            new Set(
                task.albums.map(
                    compact
                )
            );

        let added = 0;

        for (const name of names) {
            const key =
                compact(name);

            if (
                !key ||
                existing.has(key)
            ) {
                continue;
            }

            existing.add(key);

            task.albums.push(
                name
            );

            added += 1;
        }

        if (added) {
            saveTask();

            setStatus(
                `正在扫描专辑，已记录 ${task.albums.length} 个。` +
                `最近：${task.albums.slice(-3).join('；')}`
            );
        }

        return added;
    }

    async function commitScannedAlbums(
        names,
        message
    ) {
        task.albums = [];
        task.scanCompleted = false;
        saveTask();

        appendAlbums(names);

        const selected = currentAlbum();

        if (
            selected !== '未识别专辑'
        ) {
            appendAlbums([
                selected
            ]);
        }

        task.scanCompleted = true;
        saveTask();
        await closeDropdown();

        if (!task.albums.length) {
            throw new Error('没有识别到任何专辑。');
        }

        setStatus(
            message.replace(
                '{count}',
                String(task.albums.length)
            )
        );

        return [
            ...task.albums
        ];
    }

    async function scanAlbums(
        options = {}
    ) {
        const allowCachedFallback =
            options.allowCachedFallback !== false;

        const cachedAlbums =
            task.scanCompleted
                ? [
                    ...(task.albums || [])
                ]
                : [];

        const catalogMarker =
            albumCatalogTracker.seq;

        setStatus(
            '正在打开专辑下拉框，并监听后台专辑目录接口……'
        );

        const context =
            await openDropdown();

        await searchAlbum(
            context,
            ''
        );

        context.createRow =
            closestWideRow(
                findCreateText(
                    context.root
                ),
                context.root
            );

        const firstNames =
            collectScanRows(context, true)
                .map(item => item.name);

        const domExpectedCount =
            dropdownAlbumCount(
                context
            );

        setStatus(
            domExpectedCount
                ? `已识别页面专辑总数 ${domExpectedCount} 个，正在直接读取后台专辑目录接口……`
                : '正在直接读取后台专辑目录接口……'
        );

        let catalog =
            await waitTrackedAlbumCatalog(
                catalogMarker,
                domExpectedCount,
                4200
            );

        let expectedCount =
            domExpectedCount ||
            catalog.totalHint ||
            0;

        /*
         * 如果只拿到目录接口第一页，优先尝试按真实URL分页直读。
         * 不再把“expectedCount未知 + 只捕获一页”误当成完整目录。
         */
        if (catalog.names.length && (!expectedCount || catalog.names.length < expectedCount)) {
            const expanded = await expandAlbumCatalogFromCapturedApi(
                catalogMarker,
                expectedCount
            );

            if (expanded.names.length >= catalog.names.length) {
                catalog = expanded;
            }

            expectedCount = expectedCount || catalog.totalHint || 0;
        }

        if (
            catalog.names.length &&
            expectedCount > 0 &&
            catalog.names.length >= expectedCount
        ) {
            console.info(
                '[TME] 已通过后台接口获取专辑目录',
                {
                    expectedCount,
                    captured:
                        catalog.names.length,
                    apiPaths:
                        catalog.urls
                }
            );

            return commitScannedAlbums(
                catalog.names,
                '已从后台接口直接识别 {count} 个专辑，不需要扫描滚动区域。'
            );
        }

        /*
         * 接口优先失败时才进入 DOM 兜底。
         * 这一段保留用于兼容后台接口分页、缓存命中而未重新发请求等特殊情况。
         */
        context.scrollBox =
            findRowScrollBox(context) ||
            findScrollBox(
                context.root,
                context.input,
                context.createRow
            );

        /* 静态识别后再做一次真实试滚校验，跨电脑更可靠。 */
        context.scrollBox = await probeTrueScrollBox(context) || context.scrollBox;

        const combinedNames = [];
        const combinedSeen = new Set();

        for (const name of [
            ...catalog.names,
            ...firstNames
        ]) {
            const key = compact(name);

            if (
                key &&
                !combinedSeen.has(key)
            ) {
                combinedSeen.add(key);
                combinedNames.push(name);
            }
        }

        if (
            expectedCount > 0 &&
            combinedNames.length >= expectedCount
        ) {
            return commitScannedAlbums(
                combinedNames.slice(0, expectedCount),
                '后台接口与当前可见列表合并后已识别 {count} 个专辑，不需要滚动扫描。'
            );
        }

        /*
         * 没有滚动层并不一定代表扫描失败：
         * 1. 少量专辑可能一次性全部渲染；
         * 2. 页面结构变化时，批量抓取可以安全回退到上一次已完整扫描的专辑清单。
         */
        if (!context.scrollBox) {
            if (
                firstNames.length &&
                expectedCount > 0 &&
                firstNames.length >=
                    expectedCount
            ) {
                task.albums = [];
                task.scanCompleted = false;
                saveTask();

                appendAlbums(
                    firstNames
                );

                const selected =
                    currentAlbum();

                if (
                    selected !==
                    '未识别专辑'
                ) {
                    appendAlbums([
                        selected
                    ]);
                }

                task.scanCompleted = true;
                saveTask();
                await closeDropdown();

                setStatus(
                    `专辑列表无需滚动，已直接识别 ${task.albums.length} 个专辑。`
                );

                return [
                    ...task.albums
                ];
            }

            if (
                allowCachedFallback &&
                cachedAlbums.length
            ) {
                await closeDropdown();

                setStatus(
                    `本次未定位到下拉滚动层，已自动改用上一次完整扫描的 ${cachedAlbums.length} 个专辑继续批量抓取。`
                );

                return cachedAlbums;
            }

            await closeDropdown();

            throw new Error(
                '后台专辑目录接口未返回完整清单，DOM兜底也没有定位到可滚动容器。' +
                `接口已识别 ${catalog.names.length} 个，当前可见 ${firstNames.length} 个` +
                (expectedCount
                    ? `，预计共 ${expectedCount} 个`
                    : '') +
                '。请打开控制台查看[TME]捕获到疑似专辑目录接口日志。'
            );
        }

        task.albums = [];
        task.scanCompleted = false;

        saveTask();

        /* 先记录首屏，避免滚动初始化时首屏条目被虚拟列表替换。 */
        appendAlbums(
            firstNames
        );

        context.scrollBox.scrollTop = 0;

        context.scrollBox.dispatchEvent(
            new Event(
                'scroll',
                {
                    bubbles: true
                }
            )
        );

        await sleep(600);

        let sameView = 0;
        let bottomStable = 0;
        let previousSignature = '';
        let noNewSteps = 0;
        let lastProgressAt = Date.now();

        for (
            let step = 0;
            step < CFG.scanMaxSteps;
            step += 1
        ) {
            const names = collectScanRows(context)
                .map(item => item.name);

            let added = appendAlbums(names);

            /*
             * 滚动过程中如果站点懒加载了下一页目录接口，立即把后台返回的专辑合并进来。
             * 这能绕过“DOM停在55本，但后台已经加载下一页”的情况。
             */
            const liveCatalog = collectTrackedAlbumCatalog(
                catalogMarker,
                expectedCount
            );

            if (liveCatalog.totalHint > expectedCount) {
                expectedCount = liveCatalog.totalHint;
            }

            added += appendAlbums(liveCatalog.names);

            if (added > 0) {
                noNewSteps = 0;
                lastProgressAt = Date.now();
            } else {
                noNewSteps += 1;
            }

            if (
                expectedCount > 0 &&
                task.albums.length >= expectedCount
            ) {
                setStatus(
                    `已达到页面显示的专辑总数 ${expectedCount} 个，结束扫描。`
                );
                break;
            }

            const currentSignature = names
                .map(compact)
                .join('||');

            sameView =
                currentSignature &&
                currentSignature === previousSignature
                    ? sameView + 1
                    : 0;

            previousSignature = currentSignature;

            let box = context.scrollBox;

            if (!box) {
                const recovered = findRowScrollBox(context) ||
                    findScrollBox(
                        context.root,
                        context.input,
                        context.createRow
                    );

                if (recovered) {
                    context.scrollBox = recovered;
                    box = recovered;
                    noNewSteps = 0;
                } else {
                    throw new Error(
                        `专辑列表扫描过程中丢失滚动层；当前仅识别 ${task.albums.length} 个专辑。`
                    );
                }
            }

            const maxTop = Math.max(
                0,
                box.scrollHeight - box.clientHeight
            );
            const atBottom = Number(box.scrollTop || 0) >= maxTop - 3;

            bottomStable = atBottom
                ? bottomStable + 1
                : 0;

            if (
                atBottom &&
                bottomStable >= 3 &&
                sameView >= 2
            ) {
                const beforeBottomProbe = task.albums.length;

                /*
                 * 到底部先给懒加载接口一点时间，不马上判定结束。
                 * 大量账号常见“当前页到底 -> 过几百毫秒才追加下一页”。
                 */
                box.scrollTop = Math.max(0, box.scrollHeight - box.clientHeight);
                box.dispatchEvent(new Event('scroll', { bubbles: true }));
                await sleep(650);

                const bottomCatalog = collectTrackedAlbumCatalog(
                    catalogMarker,
                    expectedCount
                );

                if (bottomCatalog.totalHint > expectedCount) {
                    expectedCount = bottomCatalog.totalHint;
                }

                appendAlbums(bottomCatalog.names);
                appendAlbums(collectScanRows(context).map(item => item.name));

                if (task.albums.length > beforeBottomProbe) {
                    noNewSteps = 0;
                    bottomStable = 0;
                    sameView = 0;
                    lastProgressAt = Date.now();
                    continue;
                }

                if (!expectedCount || task.albums.length >= expectedCount) {
                    setStatus(
                        `已到专辑列表底部，确认没有新增，扫描结束，共 ${task.albums.length} 个专辑。`
                    );
                    break;
                }

                /*
                 * 9.6.2：到底后只寻找“还能继续向下滚”的其他真实滚动层。
                 * 禁止通过向上反向试滚来验证当前容器，否则同一底部容器会被
                 * 一遍遍重新确认，造成扫描永远停不下来。
                 */
                const probed = await probeTrueScrollBox(
                    context,
                    true,
                    { forwardOnly: true }
                );

                if (probed) {
                    context.scrollBox = probed;
                    bottomStable = 0;
                    sameView = 0;
                    noNewSteps = 0;
                    lastProgressAt = Date.now();
                    continue;
                }

                /*
                 * 页面已经真实到底、后台也没有继续返回新专辑时，必须结束扫描。
                 * expectedCount 有时来自不完全可靠的 total 字段，可能比实际可用专辑数大；
                 * 此时记录警告，但不能为了追一个错误总数无限滚动。
                 */
                console.warn(
                    '[TME] 已到专辑列表真实底部，但识别数量少于总数提示，按实际可见目录结束扫描',
                    {
                        expectedCount,
                        scanned: task.albums.length
                    }
                );

                setStatus(
                    `已到专辑列表底部，连续确认无新增，扫描结束：${task.albums.length} 个` +
                    (expectedCount > task.albums.length
                        ? `（页面/接口曾提示 ${expectedCount} 个，已记录差异）`
                        : '')
                );
                break;
            }

            /*
             * 连续多步没有新增专辑时，主动重新寻找“可见专辑行的最近滚动祖先”。
             * 这就是针对“大量专辑账号卡在下拉框顶部”的核心修复。
             */
            if (noNewSteps === CFG.scanStallSteps) {
                setStatus(
                    `扫描已识别 ${task.albums.length} 个专辑，连续 ${noNewSteps} 次没有新增，正在重新定位专辑列表滚动层……`
                );

                const recovered = await probeTrueScrollBox(context, true);

                if (recovered) {
                    context.scrollBox = recovered;
                    noNewSteps = 0;
                    await sleep(180);
                    continue;
                }
            }

            if (
                noNewSteps >= CFG.scanHardStallSteps ||
                Date.now() - lastProgressAt > 45000
            ) {
                const currentBox = context.scrollBox;
                const currentMaxTop = currentBox
                    ? Math.max(0, currentBox.scrollHeight - currentBox.clientHeight)
                    : 0;
                const currentlyAtBottom = Boolean(
                    currentBox &&
                    Number(currentBox.scrollTop || 0) >= currentMaxTop - 3
                );
                const recovered = await probeTrueScrollBox(
                    context,
                    true,
                    currentlyAtBottom
                        ? { forwardOnly: true }
                        : {}
                );

                if (recovered) {
                    context.scrollBox = recovered;
                    noNewSteps = 0;
                    lastProgressAt = Date.now();
                    continue;
                }

                if (currentlyAtBottom) {
                    setStatus(
                        `已到专辑列表底部，长时间无新增，扫描结束，共 ${task.albums.length} 个专辑。`
                    );
                    break;
                }

                throw new Error(
                    `专辑目录连续滚动仍无新增，已识别 ${task.albums.length} 个。` +
                    '已停止无效滚动，避免在大量专辑账号中长时间假死。'
                );
            }

            const result = await scrollList(context);

            if (result.boxChanged) {
                noNewSteps = 0;
                lastProgressAt = Date.now();
            }

            if (
                !result.changed &&
                result.newTop === result.oldTop &&
                result.newSignature === result.oldSignature
            ) {
                const refreshed = await probeTrueScrollBox(context, true) ||
                    findRowScrollBox(context) ||
                    findScrollBox(
                        context.root,
                        context.input,
                        context.createRow
                    );

                if (
                    refreshed &&
                    refreshed !== context.scrollBox
                ) {
                    context.scrollBox = refreshed;
                    noNewSteps = 0;
                } else if (atBottom) {
                    bottomStable += 1;
                }
            }

            /*
             * 主动把执行权还给浏览器，让页面渲染、输入和系统消息都有机会处理。
             * 这对几百本专辑的账号很重要，可避免长任务被浏览器判定为“页面无响应”。
             */
            await sleep(24);
        }

        const selected =
            currentAlbum();

        if (
            selected !==
            '未识别专辑'
        ) {
            appendAlbums([
                selected
            ]);
        }

        task.scanCompleted = true;

        saveTask();

        await closeDropdown();

        if (!task.albums.length) {
            throw new Error(
                '下拉列表已经滚动，但没有成功记录到专辑名称。'
            );
        }

        if (
            expectedCount > 0 &&
            task.albums.length <
                expectedCount
        ) {
            console.warn(
                '[TME] 专辑扫描数量少于页面总数',
                {
                    expectedCount,
                    scanned:
                        task.albums.length
                }
            );
        }

        return [
            ...task.albums
        ];
    }

    function matchItem(
        items,
        target
    ) {
        const key =
            compact(target);

        return (
            items.find(item => {
                return (
                    compact(item.name) ===
                    key
                );
            }) ||

            items.find(item => {
                return (
                    compact(item.name)
                        .includes(key) ||
                    key.includes(
                        compact(item.name)
                    )
                );
            }) ||

            null
        );
    }

    class AlbumUnavailableError extends Error {
        constructor(album, reason) {
            super(`专辑不可进入：${album}；${reason}`);
            this.name = 'AlbumUnavailableError';
            this.code = 'ALBUM_UNAVAILABLE';
            this.album = album;
            this.reason = reason;
        }
    }

    function isAlbumUnavailableError(error) {
        return Boolean(
            error &&
            (
                error.code === 'ALBUM_UNAVAILABLE' ||
                error.name === 'AlbumUnavailableError'
            )
        );
    }

    function parseCssRgb(value) {
        const match = String(value || '').match(
            /rgba?\(\s*(\d+(?:\.\d+)?)\s*,\s*(\d+(?:\.\d+)?)\s*,\s*(\d+(?:\.\d+)?)/i
        );

        if (!match) {
            return null;
        }

        return [
            Number(match[1]),
            Number(match[2]),
            Number(match[3])
        ];
    }

    function isDisabledGrayColor(value) {
        const rgb = parseCssRgb(value);

        if (!rgb) {
            return false;
        }

        const max = Math.max(...rgb);
        const min = Math.min(...rgb);
        const average = (rgb[0] + rgb[1] + rgb[2]) / 3;

        return (
            max - min <= 12 &&
            average >= 120 &&
            average <= 225
        );
    }

    function noApprovedAlbumSignal() {
        const selector = [
            'div',
            'span',
            'p',
            'strong',
            'b',
            'h1',
            'h2',
            'h3',
            'h4'
        ].join(',');

        const patterns = new Set([
            '暂无专辑通过审核',
            '暂无已通过审核专辑',
            '暂无已通过审核的专辑',
            '暂无审核通过专辑',
            '暂无审核通过的专辑'
        ]);

        const matches = all(selector)
            .filter(element => {
                return patterns.has(
                    compact(
                        element.innerText ||
                        element.textContent
                    )
                );
            })
            .sort((a, b) => {
                return (
                    a.getBoundingClientRect().top -
                    b.getBoundingClientRect().top
                );
            });

        const element = matches[0] || null;

        return {
            active: Boolean(element),
            text: element
                ? clean(
                    element.innerText ||
                    element.textContent
                )
                : '',
            element
        };
    }

    function inspectAlbumItemState(item, context) {
        if (!item?.element) {
            return {
                unavailable: false,
                grayHint: false,
                interactiveHint: false,
                reason: ''
            };
        }

        const reasons = [];
        let grayHint = false;
        let interactiveHint = false;

        /*
         * 旧版只检查搜索结果行本身和祖先节点。
         * TME下架结果常常是“行本身正常、标题文字和封面变灰”，
         * 因此连续下架时会漏判。9.5.4同时检查行内标题、图片、按钮等后代节点。
         */
        const targetName = compact(item.name);
        const visualNodes = [
            item.element,
            ...Array.from(
                item.element.querySelectorAll(
                    'span,p,strong,b,a,button,[role="button"],[role="option"],[title],img'
                )
            ).slice(0, 160)
        ];

        for (const node of visualNodes) {
            if (!(node instanceof Element) || !visible(node)) {
                continue;
            }

            const text = compact(
                node.innerText ||
                node.textContent ||
                node.getAttribute('title')
            );
            const className = String(node.className || '');
            const style = getComputedStyle(node);
            const ariaDisabled = clean(
                node.getAttribute('aria-disabled')
            ).toLowerCase();
            const dataDisabled = clean(
                node.getAttribute('data-disabled')
            ).toLowerCase();
            const disabledAttribute =
                node.hasAttribute('disabled') ||
                node.disabled === true;

            if (
                /已下架|下架|已删除|不可用|不可进入|已失效|已封禁|封禁|已停用|停止分发|审核不通过|暂无专辑通过审核/.test(text)
            ) {
                reasons.push('搜索结果标记为已下架或不可用');
            }

            if (
                disabledAttribute ||
                ariaDisabled === 'true' ||
                dataDisabled === 'true' ||
                dataDisabled === '1'
            ) {
                reasons.push('搜索结果带有禁用属性');
            }

            if (
                /(?:^|[-_\s])(disabled|disable|inactive|unavailable|offshelf|off-shelf|removed|invalid|gray|grey)(?:$|[-_\s])/i.test(
                    className
                )
            ) {
                reasons.push('搜索结果使用禁用/下架样式');
            }

            if (style.pointerEvents === 'none') {
                reasons.push('搜索结果禁止点击');
            }

            if (style.cursor === 'not-allowed') {
                reasons.push('搜索结果鼠标状态为不可点击');
            }

            const opacity = Number(style.opacity || 1);
            const nodeText = compact(
                node.innerText ||
                node.textContent ||
                node.getAttribute('title')
            );
            const isPrimaryText = Boolean(
                targetName &&
                nodeText &&
                (
                    nodeText === targetName ||
                    (
                        nodeText.length <= targetName.length + 12 &&
                        (
                            nodeText.includes(targetName) ||
                            targetName.includes(nodeText)
                        )
                    )
                )
            );
            const isImage = node.matches('img');
            const rowLevel = node === item.element;

            /*
             * 只把“专辑主标题、封面或整行本身”变灰当作下架视觉信号。
             * 普通专辑里的“完结”等灰色角标不能触发下架判断。
             */
            if (
                Number.isFinite(opacity) &&
                opacity > 0 &&
                opacity <= 0.72 &&
                (rowLevel || isPrimaryText || isImage)
            ) {
                grayHint = true;
            }

            if (
                isPrimaryText &&
                isDisabledGrayColor(style.color)
            ) {
                grayHint = true;
            }

            if (
                isImage &&
                /grayscale|gray/i.test(String(style.filter || ''))
            ) {
                grayHint = true;
            }

            if (
                node.matches('button,a,[role="button"],[role="option"],li') ||
                style.cursor === 'pointer'
            ) {
                if (
                    style.pointerEvents !== 'none' &&
                    !disabledAttribute &&
                    ariaDisabled !== 'true' &&
                    dataDisabled !== 'true' &&
                    dataDisabled !== '1'
                ) {
                    interactiveHint = true;
                }
            }
        }

        /* 继续检查点击目标到下拉根节点之间的祖先，捕获组件级禁用属性。 */
        let current = item.element;

        for (
            let level = 0;
            level < 7 &&
            current &&
            current !== context?.root?.parentElement;
            level += 1
        ) {
            if (!(current instanceof Element)) {
                break;
            }

            const className = String(current.className || '');
            const style = getComputedStyle(current);
            const ariaDisabled = clean(
                current.getAttribute('aria-disabled')
            ).toLowerCase();
            const dataDisabled = clean(
                current.getAttribute('data-disabled')
            ).toLowerCase();
            const disabledAttribute =
                current.hasAttribute('disabled') ||
                current.disabled === true;

            if (
                disabledAttribute ||
                ariaDisabled === 'true' ||
                dataDisabled === 'true' ||
                dataDisabled === '1'
            ) {
                reasons.push('搜索结果带有禁用属性');
            }

            if (
                /(?:^|[-_\s])(disabled|disable|inactive|unavailable|offshelf|off-shelf|removed|invalid|gray|grey)(?:$|[-_\s])/i.test(
                    className
                )
            ) {
                reasons.push('搜索结果使用禁用/下架样式');
            }

            if (
                current.matches('button,a,[role="button"],[role="option"],li') ||
                style.cursor === 'pointer'
            ) {
                if (
                    style.pointerEvents !== 'none' &&
                    !disabledAttribute &&
                    ariaDisabled !== 'true' &&
                    dataDisabled !== 'true' &&
                    dataDisabled !== '1'
                ) {
                    interactiveHint = true;
                }
            }

            if (current === context?.root) {
                break;
            }

            current = current.parentElement;
        }

        const uniqueReasons = Array.from(new Set(reasons));

        return {
            unavailable: uniqueReasons.length > 0,
            grayHint,
            interactiveHint,
            reason:
                uniqueReasons.join('；') ||
                (grayHint ? '搜索结果呈灰色不可用样式' : '')
        };
    }

    function markAlbumUnavailable(album, reason) {
        task.unavailable = Array.isArray(task.unavailable)
            ? task.unavailable
            : [];

        const key = compact(album);
        const existing = task.unavailable.find(item => {
            return compact(item?.album) === key;
        });
        const entry = {
            album,
            reason:
                clean(reason) ||
                '搜索结果不可点击，疑似已下架或不可进入',
            detectedAt: new Date().toISOString()
        };

        if (existing) {
            Object.assign(existing, entry);
        } else {
            task.unavailable.push(entry);
        }

        task.failed = (task.failed || []).filter(item => {
            return compact(item?.album) !== key;
        });

        task.done = (task.done || []).filter(name => {
            return compact(name) !== key;
        });
    }

    function albumAlreadyUnavailable(album) {
        const key = compact(album);

        return (task.unavailable || []).some(item => {
            return compact(item?.album) === key;
        });
    }

    function albumSearchContextUsable(context) {
        return Boolean(
            context &&
            context.input &&
            context.root &&
            context.input.isConnected &&
            context.root.isConnected &&
            visible(context.input) &&
            visible(context.root)
        );
    }

    function discoverOpenAlbumSearchContext() {
        const inputs = all('input').filter(input => {
            const placeholder = clean(input.getAttribute('placeholder'));
            return /(?:\u8f93\u5165.*\u4e13\u8f91.*\u641c\u7d22|\u4e13\u8f91.*\u641c\u7d22|\u641c\u7d22.*\u4e13\u8f91)/.test(placeholder);
        });

        for (const input of inputs) {
            const inputRect = input.getBoundingClientRect();
            const pseudoSelectorRect = {
                left: inputRect.left,
                right: inputRect.right,
                width: Math.max(inputRect.width, 360),
                top: Math.max(0, inputRect.top - 90),
                bottom: inputRect.top + 8
            };

            let root = findDropdownRoot(input, pseudoSelectorRect);

            if (!root) {
                let current = input.parentElement;
                let best = null;

                for (
                    let level = 0;
                    level < 12 &&
                    current &&
                    current !== document.body;
                    level += 1
                ) {
                    if (visible(current) && !inPanel(current)) {
                        const rect = current.getBoundingClientRect();
                        const hasSearch = Boolean(searchButton(current));
                        const hasRows = all('img,[role="option"],li', current).length > 0;
                        const hasCreate = Boolean(findCreateText(current));

                        if (
                            rect.width >= Math.max(320, inputRect.width * 0.9) &&
                            rect.height >= 110 &&
                            (hasSearch || hasRows || hasCreate)
                        ) {
                            best = current;
                        }
                    }

                    current = current.parentElement;
                }

                root = best;
            }

            if (!root) {
                continue;
            }

            const createRow = closestWideRow(
                findCreateText(root),
                root
            );
            const scrollBox = findScrollBox(
                root,
                input,
                createRow
            );

            return {
                selectorElement: null,
                input,
                root,
                createRow,
                scrollBox
            };
        }

        return null;
    }

    function preserveAlbumSearchContext(context) {
        /*
         * 9.5.8：优先直接保存当前仍有效的搜索上下文。
         * 旧版每次保存都会重新扫描整块下拉DOM，是批量切书时的重要耗时点。
         */
        if (albumSearchContextUsable(context)) {
            preservedAlbumSearchContext = context;
            return;
        }

        const liveContext = discoverOpenAlbumSearchContext();

        if (albumSearchContextUsable(liveContext)) {
            preservedAlbumSearchContext = liveContext;
        }
    }

    function clearPreservedAlbumSearchContext() {
        preservedAlbumSearchContext = null;
    }

    async function getAlbumSearchContext() {
        /*
         * 9.5.8：连续下架时搜索框本来就保持展开，优先复用已验证仍连接的输入框。
         * 只有缓存DOM失效时才重新扫描页面，避免每本书都做一次重型DOM发现。
         */
        if (albumSearchContextUsable(preservedAlbumSearchContext)) {
            return preservedAlbumSearchContext;
        }

        const liveContext = discoverOpenAlbumSearchContext();

        if (albumSearchContextUsable(liveContext)) {
            preservedAlbumSearchContext = liveContext;
            return liveContext;
        }

        preservedAlbumSearchContext = null;

        const context = await openDropdown();
        preserveAlbumSearchContext(context);
        return context;
    }

    function fastVisibleElement(element) {
        if (!element || !(element instanceof Element) || !element.isConnected || inPanel(element)) {
            return false;
        }

        const rect = element.getBoundingClientRect();

        if (rect.width <= 0 || rect.height <= 0) {
            return false;
        }

        const style = getComputedStyle(element);
        return style.display !== 'none' && style.visibility !== 'hidden' && Number(style.opacity || 1) !== 0;
    }

    function visibleAlbumSearchInputFast() {
        const inputs = document.querySelectorAll('input');

        for (const input of inputs) {
            const placeholder = clean(input.getAttribute('placeholder'));

            if (
                /输入.*专辑.*搜索|专辑.*搜索|搜索.*专辑/.test(placeholder) &&
                fastVisibleElement(input)
            ) {
                return input;
            }
        }

        return null;
    }

    function fastAlbumResult(context, name) {
        if (!albumSearchContextUsable(context)) {
            return null;
        }

        const root = context.root;
        const key = compact(name);

        if (!key) {
            return null;
        }

        /*
         * 搜索结果基本都会带封面。先从少量img向上找所属结果行，
         * 避免 collectRows 对整块下拉DOM做多轮全量扫描。
         */
        const images = Array.from(root.querySelectorAll('img')).slice(0, 36);

        for (const image of images) {
            let current = image.parentElement;

            for (let level = 0; level < 8 && current && current !== root; level += 1) {
                const text = compact(current.innerText || current.textContent);

                if (text && (text.includes(key) || key.includes(text))) {
                    const rect = current.getBoundingClientRect();

                    if (rect.width >= 180 && rect.height >= 34 && rect.height <= 220) {
                        return {
                            name,
                            element: rowClickTarget(current, context)
                        };
                    }
                }

                current = current.parentElement;
            }
        }

        /* 无封面或结构变化时，仅扫描标题类节点作为轻量兜底。 */
        const nodes = Array.from(
            root.querySelectorAll('[title],a,span,p,strong,b')
        ).slice(0, 160);

        for (const node of nodes) {
            const value = compact(
                node.getAttribute('title') ||
                node.innerText ||
                node.textContent
            );

            if (!value || !(value === key || value.includes(key) || key.includes(value))) {
                continue;
            }

            let current = node;

            for (let level = 0; level < 7 && current && current !== root; level += 1) {
                const rect = current.getBoundingClientRect();

                if (rect.width >= 180 && rect.height >= 34 && rect.height <= 220) {
                    return {
                        name,
                        element: rowClickTarget(current, context)
                    };
                }

                current = current.parentElement;
            }
        }

        return null;
    }

    function fastAlbumItemState(item, context) {
        if (!item?.element) {
            return {
                unavailable: false,
                grayHint: false,
                interactiveHint: false,
                confidentAvailable: false,
                reason: ''
            };
        }

        const element = item.element;
        const targetName = compact(item.name);
        const text = compact(element.innerText || element.textContent);
        const reasons = [];
        let grayHint = false;
        let interactiveHint = false;

        const checkNode = (node, primary = false) => {
            if (!node || !(node instanceof Element) || !node.isConnected) {
                return;
            }

            const nodeText = compact(
                node.getAttribute('title') ||
                node.innerText ||
                node.textContent
            );
            const className = String(node.className || '');
            const ariaDisabled = clean(node.getAttribute('aria-disabled')).toLowerCase();
            const dataDisabled = clean(node.getAttribute('data-disabled')).toLowerCase();
            const disabled = node.hasAttribute('disabled') || node.disabled === true;
            const style = getComputedStyle(node);

            if (/已下架|下架|已删除|不可用|不可进入|已失效|已封禁|封禁|已停用|停止分发|审核不通过/.test(nodeText)) {
                reasons.push('搜索结果标记为已下架或不可用');
            }

            if (disabled || ariaDisabled === 'true' || dataDisabled === 'true' || dataDisabled === '1') {
                reasons.push('搜索结果带有禁用属性');
            }

            if (/(?:^|[-_\s])(disabled|disable|inactive|unavailable|offshelf|off-shelf|removed|invalid|gray|grey)(?:$|[-_\s])/i.test(className)) {
                reasons.push('搜索结果使用禁用/下架样式');
            }

            if (style.pointerEvents === 'none' || style.cursor === 'not-allowed') {
                reasons.push('搜索结果不可点击');
            }

            const opacity = Number(style.opacity || 1);

            if (Number.isFinite(opacity) && opacity > 0 && opacity <= 0.72 && primary) {
                grayHint = true;
            }

            if (primary && isDisabledGrayColor(style.color)) {
                grayHint = true;
            }

            if (node.matches('button,a,[role="button"],[role="option"],li') || style.cursor === 'pointer') {
                if (!disabled && ariaDisabled !== 'true' && dataDisabled !== 'true' && dataDisabled !== '1' && style.pointerEvents !== 'none') {
                    interactiveHint = true;
                }
            }
        };

        checkNode(element, true);

        let titleNode = null;
        const titleCandidates = Array.from(
            element.querySelectorAll('[title],a,span,p,strong,b')
        ).slice(0, 36);

        for (const node of titleCandidates) {
            const nodeText = compact(
                node.getAttribute('title') ||
                node.innerText ||
                node.textContent
            );

            if (
                targetName &&
                nodeText &&
                (nodeText === targetName || nodeText.includes(targetName) || targetName.includes(nodeText))
            ) {
                titleNode = node;
                break;
            }
        }

        if (titleNode && titleNode !== element) {
            checkNode(titleNode, true);
        }

        const image = element.querySelector('img');

        if (image) {
            const style = getComputedStyle(image);
            const opacity = Number(style.opacity || 1);

            if (
                (Number.isFinite(opacity) && opacity > 0 && opacity <= 0.72) ||
                /grayscale|gray/i.test(String(style.filter || ''))
            ) {
                grayHint = true;
            }
        }

        if (/已下架|下架|不可用|不可进入|审核不通过/.test(text)) {
            reasons.push('搜索结果标记为已下架或不可用');
        }

        const uniqueReasons = Array.from(new Set(reasons));
        const unavailable = uniqueReasons.length > 0;

        return {
            unavailable,
            grayHint,
            interactiveHint,
            confidentAvailable: !unavailable && !grayHint && interactiveHint,
            reason:
                uniqueReasons.join('；') ||
                (grayHint ? '搜索结果呈灰色不可用样式' : '')
        };
    }

    function explicitAlbumUnavailableFast(item, context) {
        /*
         * 9.5.9极速预判：正常专辑不再执行 getComputedStyle 扫描。
         * 这里只读取显式禁用属性、禁用类名和短状态标签。
         * 灰色但没有显式属性的下架专辑，点击后搜索框不会关闭，
         * 再由 waitAlbum 的一次性详细诊断识别。
         */
        if (!item?.element) {
            return {
                unavailable: false,
                reason: ''
            };
        }

        const reasons = [];
        const statusPattern = /^(?:已下架|下架|已删除|不可用|不可进入|已失效|已封禁|封禁|已停用|停止分发|审核不通过)$/;
        const classPattern = /(?:^|[-_\s])(disabled|disable|inactive|unavailable|offshelf|off-shelf|removed|invalid)(?:$|[-_\s])/i;

        const checkAttributes = node => {
            if (!node || !(node instanceof Element) || !node.isConnected) {
                return;
            }

            const ariaDisabled = clean(node.getAttribute('aria-disabled')).toLowerCase();
            const dataDisabled = clean(node.getAttribute('data-disabled')).toLowerCase();
            const className = String(node.className || '');

            if (
                node.hasAttribute('disabled') ||
                node.disabled === true ||
                ariaDisabled === 'true' ||
                dataDisabled === 'true' ||
                dataDisabled === '1'
            ) {
                reasons.push('搜索结果带有禁用属性');
            }

            if (classPattern.test(className)) {
                reasons.push('搜索结果使用禁用/下架样式');
            }
        };

        checkAttributes(item.element);

        let ancestor = item.element.parentElement;

        for (
            let level = 0;
            level < 4 &&
            ancestor &&
            ancestor !== context?.root?.parentElement;
            level += 1
        ) {
            checkAttributes(ancestor);

            if (ancestor === context?.root) {
                break;
            }

            ancestor = ancestor.parentElement;
        }

        const statusNodes = Array.from(
            item.element.querySelectorAll('span,p,strong,b,em,i,label')
        ).slice(0, 24);

        for (const node of statusNodes) {
            const label = compact(node.textContent || '');

            if (label && label.length <= 12 && statusPattern.test(label)) {
                reasons.push(`搜索结果状态：${clean(node.textContent || label)}`);
                break;
            }
        }

        const uniqueReasons = Array.from(new Set(reasons));

        return {
            unavailable: uniqueReasons.length > 0,
            reason: uniqueReasons.join('；')
        };
    }

    async function waitFastSearchResult(context, name, timeout = 2600) {
        const start = Date.now();
        let liveContext = context;

        while (Date.now() - start < timeout) {
            if (!albumSearchContextUsable(liveContext)) {
                const refreshed = discoverOpenAlbumSearchContext();

                if (albumSearchContextUsable(refreshed)) {
                    liveContext = refreshed;
                    preservedAlbumSearchContext = refreshed;
                }
            }

            const item = fastAlbumResult(liveContext, name);

            if (item) {
                return {
                    context: liveContext,
                    item
                };
            }

            await sleep(90);
        }

        return {
            context: liveContext,
            item: null
        };
    }

    function throwAlbumUnavailableKeepSearch(context, name, reason) {
        /*
         * Keep the search UI open. The site may replace the popup DOM node
         * after every search, so preserveAlbumSearchContext reacquires the
         * newest live input/root before the batch loop advances.
         */
        preserveAlbumSearchContext(context);

        throw new AlbumUnavailableError(
            name,
            reason
        );
    }

    async function selectAlbum(name) {
        /*
         * 9.5.9极速进入：
         * 1. 搜索结果一旦出现，只做“显式禁用”轻判断；
         * 2. 没有明确禁用信号就马上点击，不再先扫描颜色/透明度/鼠标样式；
         * 3. 能否进入以“该搜索框是否真正关闭”为主判断；
         * 4. 只有点击后搜索框仍然存在，才执行一次完整的灰色/下架诊断。
         */
        let context = await getAlbumSearchContext();

        setStatus(`快速搜索：${name}`);

        context = await searchAlbum(
            context,
            name
        );

        const searchResult = await waitFastSearchResult(
            context,
            name,
            2200
        );

        context = searchResult.context;
        let item = searchResult.item;

        if (!item) {
            const noApprovedState = noApprovedAlbumSignal();

            if (noApprovedState.active) {
                throwAlbumUnavailableKeepSearch(
                    context,
                    name,
                    `本次搜索后页面显示“${noApprovedState.text || '暂无专辑通过审核'}”，且没有找到可进入的数据页搜索结果`
                );
            }

            /* 只在快速定位失败时做一次完整结果识别。 */
            item = matchItem(
                collectRows(context),
                name
            );

            if (!item) {
                throw new Error(
                    `已搜索专辑，但没有识别到搜索结果：${name}`
                );
            }
        }

        const explicitState = explicitAlbumUnavailableFast(
            item,
            context
        );

        if (explicitState.unavailable) {
            throwAlbumUnavailableKeepSearch(
                context,
                name,
                explicitState.reason || '搜索结果明确处于禁用状态'
            );
        }

        setStatus(`搜索结果已出现，立即尝试进入：${name}`);

        item.element.scrollIntoView({
            behavior: 'auto',
            block: 'center',
            inline: 'nearest'
        });

        /* 记录原搜索框节点，后面直接观察它是否关闭。 */
        pendingAlbumNavigation = {
            name,
            input: context.input,
            root: context.root,
            context,
            itemElement: item.element,
            clickedAt: Date.now()
        };

        await sleep(15);

        const clicked = humanClick(item.element);

        if (!clicked) {
            pendingAlbumNavigation = null;
            throw new Error(`搜索结果点击被拦截：${name}`);
        }

        /* 不在这里做任何额外“可进入”扫描，立即交给 waitAlbum。 */
        await sleep(20);
    }

    function loading() {
        const selector = [
            '.ant-spin-spinning',
            '.el-loading-mask',
            '[class*="loading"]',
            '[class*="Loading"]',
            '[class*="spinner"]',
            '[class*="Spinner"]'
        ].join(',');

        for (const element of document.querySelectorAll(selector)) {
            if (!inPanel(element) && visible(element)) {
                return true;
            }
        }

        return false;
    }

    async function waitAlbum(name) {
        /*
         * 9.5.9：正常专辑的“可进入判断”只观察刚才那个搜索框是否关闭。
         * 这是站点自身最直接的成功信号，而且不需要反复调用 currentAlbum()。
         * 原 currentAlbum() 会扫描大量页面DOM，在专辑很多时非常慢，甚至会误识别短文字。
         */
        const navigation =
            pendingAlbumNavigation &&
            compact(pendingAlbumNavigation.name) === compact(name)
                ? pendingAlbumNavigation
                : null;

        const start = Date.now();
        const fastDecisionTimeout = 850;
        let closedSince = 0;

        const searchInputStillOpen = () => {
            if (navigation?.input?.isConnected) {
                return fastVisibleElement(navigation.input);
            }

            /* 缓存节点被站点替换时，才做一次轻量全局input查找。 */
            return Boolean(visibleAlbumSearchInputFast());
        };

        while (Date.now() - start < fastDecisionTimeout) {
            const searchOpen = searchInputStillOpen();

            if (!searchOpen) {
                if (!closedSince) {
                    closedSince = Date.now();
                }

                if (Date.now() - closedSince >= 100) {
                    pendingAlbumNavigation = null;
                    clearPreservedAlbumSearchContext();
                    controlCache.clear();

                    const panelInput = document.getElementById('tme-current');

                    if (panelInput) {
                        panelInput.value = name;
                        panelInput.title = name;
                    }

                    /* 页面后续fanChart接口还会严格校验albumId，这里无需再扫整页标题。 */
                    await sleep(60);
                    return;
                }
            } else {
                closedSince = 0;
            }

            await sleep(45);
        }

        /*
         * 850ms后搜索框还在，才认为“可能下架/不可进入”。
         * 此时只做一次完整诊断，而不是正常专辑每本都做。
         */
        let openSearch = navigation?.context;

        if (!albumSearchContextUsable(openSearch)) {
            openSearch = discoverOpenAlbumSearchContext();
        }

        if (albumSearchContextUsable(openSearch)) {
            const item =
                fastAlbumResult(openSearch, name) ||
                matchItem(
                    collectRows(openSearch),
                    name
                );
            const state = item
                ? inspectAlbumItemState(item, openSearch)
                : {
                    unavailable: false,
                    grayHint: false,
                    reason: ''
                };
            const noApproved = noApprovedAlbumSignal();

            if (
                noApproved.active ||
                state.unavailable ||
                state.grayHint
            ) {
                pendingAlbumNavigation = null;

                throwAlbumUnavailableKeepSearch(
                    openSearch,
                    name,
                    state.reason ||
                    `点击后搜索框仍保持展开${noApproved.active ? `，页面显示“${noApproved.text || '暂无专辑通过审核'}”` : ''}`
                );
            }
        }

        /*
         * 极少数页面动画较慢但没有任何下架信号，再给一个很短的宽限期。
         * 仍只观察输入框，不执行 currentAlbum()/全页标题扫描。
         */
        const graceStart = Date.now();

        while (Date.now() - graceStart < 650) {
            if (!searchInputStillOpen()) {
                pendingAlbumNavigation = null;
                clearPreservedAlbumSearchContext();
                controlCache.clear();

                const panelInput = document.getElementById('tme-current');

                if (panelInput) {
                    panelInput.value = name;
                    panelInput.title = name;
                }

                await sleep(60);
                return;
            }

            await sleep(55);
        }

        pendingAlbumNavigation = null;
        throw new Error(`专辑点击后搜索框未关闭，无法确认进入数据页：${name}`);
    }

    function clickable(values) {
        const keys = values.map(compact);
        const cacheKey = keys.join('|');
        const cached = controlCache.get(cacheKey);

        if (
            cached &&
            cached.isConnected &&
            !inPanel(cached) &&
            visible(cached) &&
            keys.some(key => compact(cached.textContent || '').includes(key))
        ) {
            return cached;
        }

        const selector = [
            'button',
            'a',
            '[role="button"]',
            '[role="tab"]',
            '[role="radio"]',
            'li',
            'label',
            'span',
            'p',
            'strong',
            'b',
            'div'
        ].join(',');

        const seen = new Set();
        const result = [];

        for (const raw of document.querySelectorAll(selector)) {
            if (inPanel(raw)) {
                continue;
            }

            const rawText = compact(raw.textContent || '');

            if (!keys.includes(rawText)) {
                continue;
            }

            const target = raw.closest(
                'button,a,[role="button"],[role="tab"],[role="radio"],li,label'
            ) || raw;

            if (seen.has(target) || inPanel(target) || !visible(target)) {
                continue;
            }

            seen.add(target);

            const role = target.getAttribute('role') || '';
            const className = String(target.className || '').toLowerCase();
            const rect = target.getBoundingClientRect();
            let score = 0;

            if (role === 'tab') {
                score += 40;
            }

            if (role === 'button' || role === 'radio') {
                score += 28;
            }

            if (target.matches('button,a')) {
                score += 20;
            }

            if (/tab|radio|button|range|quick|date/.test(className)) {
                score += 15;
            }

            if (rect.top >= 0 && rect.top <= 700) {
                score += 10;
            }

            score -= Math.min(rawText.length, 100) / 100;
            result.push({ target, score });
        }

        result.sort((a, b) => b.score - a.score);

        const target = result[0]?.target || null;

        if (target) {
            controlCache.set(cacheKey, target);
        }

        return target;
    }

    async function clickText(values, typeName) {
        const target = clickable(values);

        if (!target) {
            throw new Error(`没有找到${typeName}：${values.join('、')}`);
        }

        const rect = target.getBoundingClientRect();

        if (
            rect.top < 0 ||
            rect.bottom > window.innerHeight ||
            rect.left < 0 ||
            rect.right > window.innerWidth
        ) {
            target.scrollIntoView({
                behavior: 'auto',
                block: 'center',
                inline: 'center'
            });
            await sleep(70);
        }

        humanClick(target);
        await sleep(70);

        return target;
    }

    function parsePercentValue(value) {
        const text =
            compact(value);

        const strictlyLessThan =
            text.match(
                /占比[：:]?(?:(?:<|＜|小于|少于|低于|不足)([\d.]+)%|([\d.]+)%(?:以下|以内))/
            );

        if (strictlyLessThan) {
            const boundary =
                Number(
                    strictlyLessThan[1] ??
                    strictlyLessThan[2]
                );

            if (
                Number.isFinite(boundary) &&
                boundary <=
                CFG.percentZeroThreshold
            ) {
                return {
                    ready: true,
                    value: 0,
                    raw:
                        strictlyLessThan[0],
                    source:
                        'below-zero-threshold'
                };
            }

            /*
             * 例如页面若显示“低于0.5%”，并不能确定实际值
             * 一定低于0.1%，因此不能擅自写成0。
             */
            return {
                ready: false,
                value: null,
                raw:
                    strictlyLessThan[0],
                source:
                    'unknown-less-than-range'
            };
        }

        const atMost =
            text.match(
                /占比[：:]?(?:≤|不超过)([\d.]+)%/
            );

        if (atMost) {
            const boundary = Number(atMost[1]);

            if (
                Number.isFinite(boundary) &&
                boundary < CFG.percentZeroThreshold
            ) {
                return {
                    ready: true,
                    value: 0,
                    raw: atMost[0],
                    source:
                        'below-zero-threshold'
                };
            }

            return {
                ready: false,
                value: null,
                raw: atMost[0],
                source:
                    'unknown-at-most-range'
            };
        }

        const normal =
            text.match(
                /占比[：:]?([\d.]+)%/
            );

        if (normal) {
            const number =
                Number(normal[1]);

            const ready =
                Number.isFinite(number);

            const belowThreshold =
                ready &&
                number <
                CFG.percentZeroThreshold;

            return {
                ready,

                value:
                    ready
                        ? (
                            belowThreshold
                                ? 0
                                : number
                        )
                        : null,

                raw:
                    normal[1],

                source:
                    belowThreshold
                        ? 'below-zero-threshold'
                        : 'number'
            };
        }

        if (
            /占比[：:]?[-‐‑‒–—―﹘﹣－−]+/.test(
                text
            )
        ) {
            return {
                ready: true,
                value: 0,
                raw: '0',
                source:
                    'dash-placeholder'
            };
        }

        if (
            /占比[：:]?(?:暂无数据|无数据|暂无)/.test(
                text
            )
        ) {
            return {
                ready: true,
                value: 0,
                raw: '0',
                source:
                    'explicit-no-data'
            };
        }

        return {
            ready: false,
            value: null,
            raw: '',
            source:
                'not-ready'
        };
    }

    function normalizeApiPlatform(value) {
        const text = compact(value)
            .toLowerCase()
            .replace(/[_.\-/]/g, '');

        if (!text) {
            return '';
        }

        if (
            text.includes('酷狗') ||
            /^(?:kg|kgmusic|kugou|kugoumusic)$/.test(text)
        ) {
            return '酷狗音乐';
        }

        if (
            text.includes('酷我') ||
            /^(?:kw|kwmusic|kuwo|kuwomusic)$/.test(text)
        ) {
            return '酷我音乐';
        }

        if (
            text.includes('懒人') ||
            /^(?:lr|lrts|lanren|lanrents|lazyaudio|lazylisten)$/.test(text)
        ) {
            return '懒人听书';
        }

        if (
            text.includes('qq音乐') ||
            /^(?:qq|qqmusic)$/.test(text)
        ) {
            return 'QQ音乐';
        }

        return '';
    }

    function collectApiObjects(value) {
        const result = [];
        const seen = new Set();

        function visit(node, path) {
            if (!node || typeof node !== 'object' || seen.has(node)) {
                return;
            }

            seen.add(node);

            if (!Array.isArray(node)) {
                result.push({
                    node,
                    path
                });
            }

            for (const [key, child] of Object.entries(node)) {
                if (child && typeof child === 'object') {
                    visit(child, path.concat(key));
                }
            }
        }

        visit(value, []);
        return result;
    }

    function collectApiLeaves(value, maxDepth = 7) {
        const result = [];
        const seen = new Set();

        function visit(node, path, depth) {
            if (depth > maxDepth) {
                return;
            }

            if (node && typeof node === 'object') {
                if (seen.has(node)) {
                    return;
                }

                seen.add(node);

                for (const [key, child] of Object.entries(node)) {
                    visit(child, path.concat(key), depth + 1);
                }

                return;
            }

            result.push({
                path,
                key: String(path[path.length - 1] || ''),
                value: node
            });
        }

        visit(value, [], 0);
        return result;
    }

    function apiPlatformIdentity(node, path) {
        const matches = [];
        const pathPlatform = normalizeApiPlatform(path[path.length - 1]);

        if (pathPlatform) {
            matches.push({
                platform: pathPlatform,
                score: 110,
                source: 'object-key'
            });
        }

        const preferredKey =
            /(?:platform|source|channel|app)(?:name|title|label)?$|^(?:name|label|title)$/i;

        for (const [key, value] of Object.entries(node)) {
            if (
                value !== null &&
                typeof value !== 'object'
            ) {
                const platform = normalizeApiPlatform(value);

                if (!platform) {
                    continue;
                }

                matches.push({
                    platform,
                    score: preferredKey.test(key) ? 180 : 85,
                    source: key
                });
            }
        }

        const platforms = new Set(matches.map(item => item.platform));

        if (platforms.size !== 1) {
            return null;
        }

        matches.sort((a, b) => b.score - a.score);
        return matches[0];
    }

    function apiNumberValue(value) {
        if (typeof value === 'number') {
            return Number.isFinite(value)
                ? {
                    number: value,
                    source: value === 0 ? 'zero-number' : 'number',
                    raw: String(value)
                }
                : null;
        }

        const text = compact(value);

        if (!text) {
            return null;
        }

        if (isDashPlaceholder(text)) {
            return {
                number: 0,
                source: 'dash-placeholder',
                raw: text
            };
        }

        if (/^(?:暂无数据|无数据|暂无|null)$/i.test(text)) {
            return {
                number: 0,
                source: 'explicit-zero',
                raw: text
            };
        }

        const number = parseNumber(text);

        if (number === null || !Number.isFinite(number)) {
            return null;
        }

        return {
            number,
            source: number === 0 ? 'zero-number' : 'number',
            raw: text
        };
    }

    function apiMetricTokens(metric) {
        if (metric.name === '播放时长') {
            return /playduration|duration|playtime|minutes?|minute|时长/i;
        }

        if (metric.name === '播放人数') {
            return /peoplecnt|peoplecount|personcnt|listener|usercnt|persons?|人数/i;
        }

        return /playcnt|playcount|playnum|playvolume|plays?|播放量/i;
    }

    function apiPercentField(pathText) {
        return String(pathText || '')
            .split('.')
            .some(part => {
                const key = part.replace(/duration/ig, '');
                return /percent|percentage|ratio|rate|proportion|share|scale|占比|比例|百分比/i.test(
                    key
                );
            });
    }

    function apiValueCandidate(leaves, metric) {
        const metricTokens = apiMetricTokens(metric);
        const ignoredTokens =
            /(?:^|\.)(?:id|code|index|order|sort|rank|timestamp|time|date|startdate|enddate|albumid|platformid|sourceid)$/i;
        const candidates = [];

        for (const leaf of leaves) {
            const pathText = leaf.path.join('.');
            const key = leaf.key;

            if (
                apiPercentField(pathText) ||
                ignoredTokens.test(pathText) ||
                typeof leaf.value === 'boolean'
            ) {
                continue;
            }

            const parsed = apiNumberValue(leaf.value);

            if (!parsed || parsed.number < 0) {
                continue;
            }

            let score = 5;

            if (metricTokens.test(pathText)) {
                score += 150;
            }

            if (
                new RegExp(`^${esc(metric.apiType)}$`, 'i').test(key)
            ) {
                score += 90;
            }

            if (/^(?:value|val|count|cnt|num|number|total|amount)$/i.test(key)) {
                score += 65;
            }

            if (typeof leaf.value === 'number') {
                score += 12;
            }

            candidates.push({
                ...parsed,
                key,
                path: pathText,
                score
            });
        }

        candidates.sort((a, b) => b.score - a.score);
        const selected = candidates[0] || null;

        if (!selected || metric.name !== '播放时长') {
            return selected;
        }

        const unitText = leaves
            .filter(leaf => {
                return /unit|单位/i.test(leaf.key);
            })
            .map(leaf => clean(leaf.value).toLowerCase())
            .join('|');
        const valuePath = selected.path.toLowerCase();
        let multiplier = 1;

        if (/millisecond|毫秒|(?:^|[^a-z])ms(?:$|[^a-z])/.test(`${valuePath}|${unitText}`)) {
            multiplier = 1 / 60000;
        } else if (/seconds?|秒|(?:^|[^a-z])sec(?:$|[^a-z])/.test(`${valuePath}|${unitText}`)) {
            multiplier = 1 / 60;
        } else if (/hours?|小时|(?:^|[^a-z])hour(?:$|[^a-z])/.test(`${valuePath}|${unitText}`)) {
            multiplier = 60;
        }

        if (multiplier !== 1) {
            selected.number = roundApiNumber(selected.number * multiplier);
            selected.source = `${selected.source}-converted-to-minutes`;
        }

        return selected;
    }

    function apiPercentCandidate(leaves) {
        const candidates = [];

        for (const leaf of leaves) {
            const pathText = leaf.path.join('.');
            const text = compact(leaf.value).replace(/％/g, '%');
            const pathLooksPercent = apiPercentField(pathText);
            const textLooksPercent = /%|<|＜|低于|小于|少于|以下/.test(text);

            if (!pathLooksPercent && !textLooksPercent) {
                continue;
            }

            if (pathLooksPercent && isDashPlaceholder(text)) {
                candidates.push({
                    value: 0,
                    raw: text,
                    scale: 'percent',
                    source: 'dash-placeholder',
                    score: 160
                });
                continue;
            }

            if (textLooksPercent) {
                const parsed = parsePercentValue(`占比:${text}`);

                if (parsed.ready) {
                    candidates.push({
                        value: parsed.value,
                        raw: parsed.raw,
                        scale: 'percent',
                        source: parsed.source,
                        score: 190
                    });
                    continue;
                }
            }

            const numeric =
                typeof leaf.value === 'number'
                    ? leaf.value
                    : (
                        /^-?\d+(?:\.\d+)?$/.test(text)
                            ? Number(text)
                            : NaN
                    );

            if (!pathLooksPercent || !Number.isFinite(numeric) || numeric < 0) {
                continue;
            }

            candidates.push({
                value: numeric,
                raw: text,
                scale:
                    /ratio|rate|proportion|share|scale/i.test(pathText) && numeric <= 1
                        ? 'fraction'
                        : 'unknown',
                source: 'api-number',
                score: 125
            });
        }

        candidates.sort((a, b) => b.score - a.score);
        return candidates[0] || null;
    }

    function apiBusinessError(payload) {
        if (!payload || typeof payload !== 'object') {
            return '接口没有返回JSON对象';
        }

        if (payload.success === false) {
            return clean(payload.message || payload.msg || '接口返回失败');
        }

        const codeKeys = ['code', 'status', 'ret', 'retCode', 'errno', 'errCode'];

        for (const key of codeKeys) {
            if (!(key in payload)) {
                continue;
            }

            const value = payload[key];
            const normalized = clean(value).toLowerCase();
            const success =
                value === 0 ||
                value === 200 ||
                value === true ||
                ['0', '200', 'ok', 'success', 'true'].includes(normalized);

            if (!success) {
                return (
                    clean(payload.message || payload.msg || payload.errMsg) ||
                    `接口业务状态异常：${key}=${clean(value)}`
                );
            }

            break;
        }

        return '';
    }

    function payloadShape(payload) {
        if (!payload || typeof payload !== 'object') {
            return typeof payload;
        }

        const keys = Object.keys(payload).slice(0, 12);
        const data = payload.data;

        return (
            `顶层字段=${keys.join(',') || '无'}；` +
            `data=${Array.isArray(data) ? `数组(${data.length})` : typeof data}`
        );
    }

    function roundApiNumber(value, digits = 6) {
        const multiplier = 10 ** digits;
        return Math.round((Number(value) + Number.EPSILON) * multiplier) / multiplier;
    }

    function parseFanChartPayload(payload, metric) {
        if (typeof payload === 'string') {
            const decoded = decodeJsonText(payload);

            if (decoded.parseError) {
                throw new Error(decoded.parseError);
            }

            payload = decoded.payload;
        }

        const businessError = apiBusinessError(payload);

        if (businessError) {
            throw new Error(businessError);
        }

        const best = new Map();

        for (const item of collectApiObjects(payload)) {
            const identity = apiPlatformIdentity(item.node, item.path);

            if (!identity) {
                continue;
            }

            const leaves = collectApiLeaves(item.node);
            const value = apiValueCandidate(leaves, metric);

            if (!value) {
                continue;
            }

            const percent = apiPercentCandidate(leaves);
            const score =
                identity.score +
                value.score +
                (percent ? percent.score : 0) -
                Math.min(leaves.length, 80) * 0.3;
            const old = best.get(identity.platform);

            if (!old || score > old.score) {
                best.set(identity.platform, {
                    platform: identity.platform,
                    number: value.number,
                    raw: value.raw,
                    numberSource: value.source,
                    percent,
                    score
                });
            }
        }

        const missing = CFG.platforms
            .map(item => item.page)
            .filter(name => !best.has(name));

        if (missing.length) {
            throw new Error(
                `fanChart响应缺少平台：${missing.join('、')}。${payloadShape(payload)}`
            );
        }

        const rows = CFG.platforms.map(platform => ({
            ...best.get(platform.page)
        }));
        const presentPercents = rows
            .map(item => item.percent)
            .filter(Boolean);
        const unknownSum = presentPercents
            .filter(item => item.scale === 'unknown')
            .reduce((sum, item) => sum + Number(item.value || 0), 0);
        const hasExplicitPercent = presentPercents.some(item => {
            return item.scale === 'percent';
        });
        const hasExplicitFraction = presentPercents.some(item => {
            return item.scale === 'fraction';
        });
        const unknownAsFraction =
            !hasExplicitPercent &&
            (
                hasExplicitFraction ||
                (
                    presentPercents.length === rows.length &&
                    unknownSum > 0 &&
                    unknownSum <= 1.0005
                )
            );
        const total = rows.reduce((sum, item) => sum + item.number, 0);

        for (const row of rows) {
            let percent = null;
            let percentSource = '';

            if (row.percent) {
                const multiplier =
                    row.percent.scale === 'fraction' ||
                    (row.percent.scale === 'unknown' && unknownAsFraction)
                        ? 100
                        : 1;
                percent = Number(row.percent.value) * multiplier;
                percentSource = row.percent.source;
            }

            if (!Number.isFinite(percent)) {
                percent = total > 0
                    ? row.number / total * 100
                    : 0;
                percentSource = total > 0
                    ? 'computed-from-api-values'
                    : 'zero-total';
            }

            percent = roundApiNumber(percent);

            if (percent < CFG.percentZeroThreshold) {
                percent = 0;
                percentSource = 'below-zero-threshold';
            }

            row.percent = percent;
            row.percentReady = true;
            row.percentSource = percentSource;
        }

        const percentSum = rows.reduce((sum, item) => sum + item.percent, 0);

        if (
            total > 0 &&
            (percentSum < 98.5 || percentSum > 101.5)
        ) {
            for (const row of rows) {
                const computed = roundApiNumber(row.number / total * 100);
                row.percent = computed < CFG.percentZeroThreshold ? 0 : computed;
                row.percentSource =
                    row.percent === 0 && computed < CFG.percentZeroThreshold
                        ? 'below-zero-threshold'
                        : 'computed-from-api-values';
            }
        }

        return rows;
    }

    function addIsoDays(value, days) {
        const time = parseIsoDate(value);

        if (time === null) {
            throw new Error(`接口日期格式无效：${value}`);
        }

        return new Date(time + days * 86400000)
            .toISOString()
            .slice(0, 10);
    }

    function rangeForPeriod(period, endDate) {
        if (period.cumulative) {
            const startDate = clean(period.startDate) || '2000-01-01';

            if (parseIsoDate(startDate) === null) {
                throw new Error(`累计起始日期格式无效：${startDate}`);
            }

            if (parseIsoDate(endDate) === null) {
                throw new Error(`累计截止日期格式无效：${endDate}`);
            }

            if (parseIsoDate(startDate) > parseIsoDate(endDate)) {
                throw new Error(`累计起始日期晚于截止日期：${startDate} > ${endDate}`);
            }

            return {
                startDate,
                endDate
            };
        }

        return {
            startDate: addIsoDays(endDate, -(period.days - 1)),
            endDate
        };
    }

    function randomNonce(length = 6) {
        const alphabet = 'abcdefghijklmnopqrstuvwxyz0123456789';
        const size = Math.max(4, Math.min(24, Number(length) || 6));
        const bytes = new Uint8Array(size);

        if (window.crypto?.getRandomValues) {
            window.crypto.getRandomValues(bytes);
        } else {
            for (let index = 0; index < bytes.length; index += 1) {
                bytes[index] = Math.floor(Math.random() * 256);
            }
        }

        return Array.from(bytes, value => {
            return alphabet[value % alphabet.length];
        }).join('');
    }

    function replayHeaders(seed) {
        const result = {
            Accept: 'application/json, text/plain, */*'
        };
        const blocked =
            /^(?:cookie|host|origin|referer|content-length|user-agent|connection|sec-|accept-encoding)/i;

        for (const [name, value] of Object.entries(seed.requestHeaders || {})) {
            if (!blocked.test(name)) {
                result[name] = value;
            }
        }

        return result;
    }

    function buildFanChartRequest(session, period, metric) {
        const url = new URL(session.seed.url, location.href);

        if (url.origin !== location.origin) {
            throw new Error('fanChart接口不是当前后台的同源地址，已拒绝请求');
        }

        const range = rangeForPeriod(period, session.endDate);
        url.searchParams.set('albumId', session.albumId);
        url.searchParams.set('type', metric.apiType);
        url.searchParams.set('startDate', range.startDate);
        url.searchParams.set('endDate', range.endDate);

        if (url.searchParams.has('timestamp')) {
            const old = url.searchParams.get('timestamp') || '';
            const timestamp = old.length >= 13
                ? Date.now()
                : Math.floor(Date.now() / 1000);
            url.searchParams.set('timestamp', String(timestamp));
        }

        if (url.searchParams.has('nonceStr')) {
            const old = url.searchParams.get('nonceStr') || '';
            url.searchParams.set('nonceStr', randomNonce(old.length || 6));
        }

        if (url.searchParams.has('_')) {
            url.searchParams.set('_', String(Date.now()));
        }

        const meta = requestMeta(url.href);

        if (
            !meta ||
            meta.albumId !== session.albumId ||
            meta.metric !== metric.name ||
            meta.period !== period.name ||
            meta.startDate !== range.startDate ||
            meta.endDate !== range.endDate
        ) {
            throw new Error('生成的fanChart请求未通过参数自检');
        }

        return {
            url,
            meta,
            range
        };
    }

    async function fetchFanChartDirect(session, period, metric) {
        if (!nativeFetch) {
            throw new Error('当前浏览器不支持接口直读');
        }

        const built = buildFanChartRequest(session, period, metric);
        let lastError = null;

        for (let attempt = 1; attempt <= CFG.directApiRetries; attempt += 1) {
            const controller = new AbortController();
            const timer = setTimeout(() => controller.abort(), CFG.apiTimeout);

            try {
                const response = await nativeFetch(built.url.href, {
                    method: 'GET',
                    credentials: 'include',
                    cache: 'no-store',
                    headers: replayHeaders(session.seed),
                    signal: controller.signal
                });

                const text = await response.text();
                const decoded = decodeJsonText(text);

                if (!response.ok) {
                    throw new Error(`fanChart接口HTTP ${response.status}`);
                }

                if (decoded.parseError) {
                    throw new Error(decoded.parseError);
                }

                const businessError = apiBusinessError(decoded.payload);

                if (businessError) {
                    throw new Error(businessError);
                }

                return {
                    id: 0,
                    ...built.meta,
                    ok: true,
                    status: response.status,
                    payload: decoded.payload,
                    requestHeaders: session.seed.requestHeaders || {},
                    source: 'direct-api',
                    completedAt: Date.now()
                };
            } catch (error) {
                lastError = error;

                if (attempt < CFG.directApiRetries) {
                    await sleep(280 * attempt);
                }
            } finally {
                clearTimeout(timer);
            }
        }

        throw lastError || new Error('fanChart接口直读失败');
    }

    async function waitTrackedFanChart(
        afterId,
        album,
        period,
        metric,
        albumId = '',
        timeout = CFG.apiSeedTimeout
    ) {
        const start = Date.now();
        let lastFailure = null;

        while (Date.now() - start < timeout) {
            const candidates = requestTracker.completed
                .filter(item => {
                    return (
                        item.id > afterId &&
                        item.period === period.name &&
                        item.metric === metric.name &&
                        (!albumId || item.albumId === String(albumId)) &&
                        (
                            !item.albumName ||
                            compact(item.albumName) === compact(album)
                        )
                    );
                })
                .sort((a, b) => b.id - a.id);
            const success = candidates.find(item => {
                return item.ok && item.payload !== null && !item.parseError;
            });

            if (success) {
                const businessError = apiBusinessError(success.payload);

                if (!businessError) {
                    return success;
                }

                lastFailure = new Error(businessError);
            } else if (candidates.length) {
                const item = candidates[0];
                lastFailure = new Error(
                    item.parseError ||
                    `fanChart接口请求失败，HTTP ${item.status || 0}`
                );
            }

            await sleep(CFG.pollInterval);
        }

        throw lastFailure || new Error(
            `未捕获到${album}/${period.name}/${metric.name}的fanChart响应`
        );
    }

    async function triggerFanChartFromPage(
        album,
        period,
        metric,
        albumId = ''
    ) {
        controlCache.clear();
        const marker = requestTracker.seq;

        await clickText(period.buttons, '时间周期');
        await sleep(120);
        await clickText([metric.name], '指标标签');

        try {
            return await waitTrackedFanChart(
                marker,
                album,
                period,
                metric,
                albumId,
                Math.min(4500, CFG.apiSeedTimeout)
            );
        } catch (firstError) {
            const alternatePeriod = CFG.periods.find(item => {
                return item.name !== period.name;
            });
            const alternateMetric = CFG.metrics.find(item => {
                return item.name !== metric.name;
            });

            if (alternatePeriod) {
                await clickText(alternatePeriod.buttons, '备用时间周期');
                await sleep(220);
                await clickText(period.buttons, '目标时间周期');
            }

            if (alternateMetric) {
                await sleep(120);
                await clickText([alternateMetric.name], '备用指标标签');
                await sleep(220);
                await clickText([metric.name], '目标指标标签');
            }

            return waitTrackedFanChart(
                marker,
                album,
                period,
                metric,
                albumId,
                CFG.apiSeedTimeout
            );
        }
    }

    async function prepareAlbumApiSession(album) {
        const seedPeriod = CFG.periods.find(item => item.name === '近七天');
        const seedMetric = CFG.metrics.find(item => item.name === '播放量');

        setStatus(`正在建立接口会话：${album}`);

        const seed = await triggerFanChartFromPage(
            album,
            seedPeriod,
            seedMetric
        );

        if (!seed.albumId || !seed.endDate) {
            throw new Error('fanChart种子响应缺少专辑ID或截止日期');
        }

        const rows = parseFanChartPayload(seed.payload, seedMetric);
        const cache = new Map();
        cache.set(`${seedPeriod.name}||${seedMetric.name}`, {
            request: seed,
            rows
        });

        return {
            album,
            albumId: seed.albumId,
            endDate: seed.endDate,
            seed,
            cache
        };
    }

    function apiMetricRecords(album, period, metric, request, rows) {
        const map = new Map(rows.map(item => [item.platform, item]));

        return CFG.platforms.map(platform => {
            const data = map.get(platform.page);

            if (!data) {
                return {
                    album,
                    period: period.name,
                    platform: platform.page,
                    metric: metric.name,
                    unit: metric.unit,
                    value: '',
                    percent: '',
                    status: '未识别',
                    note: 'fanChart接口响应缺少该平台'
                };
            }

            let zeroNote = '';

            if (data.numberSource === 'dash-placeholder') {
                zeroNote = '接口数值为横杠，按0记录';
            } else if (data.percentSource === 'dash-placeholder') {
                zeroNote = '接口占比为横杠，按0记录';
            } else if (data.percentSource === 'below-zero-threshold') {
                zeroNote = '占比低于0.1%，按0记录';
            }

            return {
                album,
                period: period.name,
                platform: platform.page,
                metric: metric.name,
                unit: metric.unit,
                value: data.number,
                percent: data.percent,
                status: '成功',
                note:
                    `fanChart接口 ${request.startDate}至${request.endDate}` +
                    (zeroNote ? `；${zeroNote}` : '')
            };
        });
    }

    async function captureMetricFromApi(album, period, metric, session) {
        const cacheKey = `${period.name}||${metric.name}`;
        let cached = session.cache.get(cacheKey);
        let directError = null;

        if (!cached) {
            try {
                const request = await fetchFanChartDirect(session, period, metric);
                const rows = parseFanChartPayload(request.payload, metric);
                cached = {
                    request,
                    rows
                };
            } catch (error) {
                directError = error;

                console.warn(
                    '[TME] 接口主动请求失败，准备改用页面触发的真实接口响应',
                    album,
                    period.name,
                    metric.name,
                    error.message
                );

                /*
                 * “累计”在当前页面并不是固定快捷按钮，优先依赖fanChart全历史
                 * 区间直读。如果站点未来增加真正的“累计”按钮，下面仍可自动回退。
                 */
                if (
                    period.cumulative &&
                    !clickable(period.buttons || [])
                ) {
                    throw new Error(
                        `累计fanChart全历史直读失败：${error.message}。` +
                        '当前页面没有可点击的“累计”快捷按钮，无法安全回退到页面触发。'
                    );
                }

                const request = await triggerFanChartFromPage(
                    album,
                    period,
                    metric,
                    session.albumId
                );

                if (request.albumId !== session.albumId) {
                    throw new Error(
                        `专辑ID不一致：目标${session.albumId}，响应${request.albumId}`
                    );
                }

                const expected = rangeForPeriod(period, session.endDate);

                if (
                    request.startDate !== expected.startDate ||
                    request.endDate !== expected.endDate ||
                    request.metric !== metric.name
                ) {
                    throw new Error(
                        '页面触发的fanChart响应未通过日期或指标校验'
                    );
                }

                const rows = parseFanChartPayload(request.payload, metric);
                cached = {
                    request,
                    rows
                };

                session.seed = request;
            }

            session.cache.set(cacheKey, cached);
        }

        const list = apiMetricRecords(
            album,
            period,
            metric,
            cached.request,
            cached.rows
        );
        const issues = metricIssues(list);

        if (issues.length) {
            throw new Error(
                `${period.name}/${metric.name}仍有${issues.length}个平台接口数据不完整` +
                (directError ? `；主动请求错误：${directError.message}` : '')
            );
        }

        upsert(list, false);
        return [];
    }

    function summaryNumericText(value) {
        const text = compact(value)
            .replace(/[％]/g, '%');

        if (!text || /%/.test(text)) {
            return false;
        }

        if (isDashPlaceholder(text)) {
            return true;
        }

        return /^(?:0|\d[\d,]*(?:\.\d+)?(?:万|亿)?(?:分钟|min)?)$/i.test(
            text
        );
    }

    function summaryLabelElement(labels) {
        const normalized = labels.map(compact);
        const candidates = [];

        for (const element of all('div,span,p,strong,b,label')) {
            const text = compact(
                element.innerText ||
                element.textContent
            );

            if (!text) {
                continue;
            }

            let matched = '';

            for (const label of normalized) {
                if (
                    text === label ||
                    (
                        text.startsWith(label) &&
                        text.length <= label.length + 18
                    )
                ) {
                    matched = label;
                    break;
                }
            }

            if (!matched) {
                continue;
            }

            const childHasSameText = Array.from(element.children).some(child => {
                return (
                    visible(child) &&
                    compact(child.innerText || child.textContent) === text
                );
            });

            if (childHasSameText) {
                continue;
            }

            const rect = element.getBoundingClientRect();

            candidates.push({
                element,
                text,
                matched,
                area: rect.width * rect.height,
                top: rect.top
            });
        }

        candidates.sort((a, b) => {
            return (
                a.area - b.area ||
                a.top - b.top
            );
        });

        return candidates[0] || null;
    }

    function summaryValueFromCombinedText(labelItem) {
        if (!labelItem) {
            return null;
        }

        const text = labelItem.text;
        const after = text.slice(labelItem.matched.length);
        const match = after.match(
            /^[:：]?((?:0|\d[\d,]*(?:\.\d+)?(?:万|亿)?(?:分钟|min)?|[-‐‑‒–—―﹘﹣－−]+))/i
        );

        if (!match || !summaryNumericText(match[1])) {
            return null;
        }

        const number = parseNumber(match[1]);

        if (number === null) {
            return null;
        }

        return {
            value: number,
            raw: clean(match[1]),
            source: 'combined-label-text'
        };
    }

    function summaryNearbyValue(labelItem) {
        if (!labelItem?.element) {
            return null;
        }

        const labelRect = labelItem.element.getBoundingClientRect();
        const candidates = [];
        const seen = new Set();
        let root = labelItem.element.parentElement;

        for (
            let level = 0;
            level < 6 &&
            root &&
            root !== document.body;
            level += 1
        ) {
            const rootRect = root.getBoundingClientRect();

            if (
                rootRect.width >= 100 &&
                rootRect.width <= 620 &&
                rootRect.height >= 45 &&
                rootRect.height <= 260
            ) {
                for (const element of all('div,span,p,strong,b', root)) {
                    if (
                        element === labelItem.element ||
                        seen.has(element)
                    ) {
                        continue;
                    }

                    seen.add(element);

                    const text = clean(
                        element.innerText ||
                        element.textContent
                    );

                    if (!summaryNumericText(text)) {
                        continue;
                    }

                    const childHasSameText = Array.from(element.children).some(child => {
                        return (
                            visible(child) &&
                            clean(child.innerText || child.textContent) === text
                        );
                    });

                    if (childHasSameText) {
                        continue;
                    }

                    const number = parseNumber(text);

                    if (number === null) {
                        continue;
                    }

                    const rect = element.getBoundingClientRect();
                    const dy = rect.top - labelRect.bottom;
                    const dx = Math.abs(
                        (rect.left + rect.width / 2) -
                        (labelRect.left + labelRect.width / 2)
                    );
                    const style = getComputedStyle(element);
                    const fontSize = parseFloat(style.fontSize) || 0;
                    const weight = parseInt(style.fontWeight, 10) || 400;
                    let score = 0;

                    if (dy >= -8 && dy <= 95) {
                        score += 180 - Math.abs(dy) * 1.6;
                    } else if (dy >= -20 && dy <= 145) {
                        score += 70 - Math.abs(dy) * 0.35;
                    } else {
                        continue;
                    }

                    score += Math.max(0, 70 - dx * 0.25);
                    score += Math.min(fontSize, 40) * 2;

                    if (fontSize >= 20) {
                        score += 55;
                    } else if (fontSize >= 16) {
                        score += 25;
                    }

                    if (weight >= 600) {
                        score += 20;
                    }

                    candidates.push({
                        value: number,
                        raw: text,
                        source: 'nearby-element',
                        score,
                        area: rect.width * rect.height
                    });
                }
            }

            root = root.parentElement;
        }

        candidates.sort((a, b) => {
            return (
                b.score - a.score ||
                a.area - b.area
            );
        });

        return candidates[0] || null;
    }

    }

    function recordKey(item) {
        return [
            item.album,
            item.period,
            item.metric,
            item.platform
        ].join('||');
    }

    function upsert(list, persist = true) {
        const map = new Map(records.map(item => [recordKey(item), item]));

        for (const item of list) {
            const key = recordKey(item);
            map.set(key, item);
            dirtyRecords.set(key, item);
        }

        records = Array.from(map.values());

        if (persist) {
            saveRecords();
        } else {
            updateCount();
        }
    }

                item.percent === null
            );
        });
    }

    function validatePeriodOrder(album) {
        const map =
            new Map(
                records.map(item => {
                    return [
                        recordKey(item),
                        item
                    ];
                })
            );

        const issues = [];
        const names =
            CFG.periods.map(item => {
                return item.name;
            });

        for (const metric of CFG.metrics) {
            for (
                const platform of
                CFG.platforms
            ) {
                const values =
                    names.map(periodName => {
                        const item =
                            map.get(
                                [
                                    album,
                                    periodName,
                                    metric.name,
                                    platform.page
                                ].join('||')
                            );

                        if (
                            !item ||
                            item.status !==
                            '成功' ||
                            item.value === '' ||
                            item.value === null
                        ) {
                            return null;
                        }

                        const number =
                            Number(item.value);

                        return Number.isFinite(
                            number
                        )
                            ? number
                            : null;
                    });

                if (
                    values.some(value => {
                        return value === null;
                    })
                ) {
                    continue;
                }

                for (
                    let index = 0;
                    index <
                    values.length - 1;
                    index += 1
                ) {
                    const earlier =
                        values[index];

                    const later =
                        values[index + 1];

                    const tolerance =
                        Math.max(
                            0.000001,
                            Math.abs(later) *
                            0.0000001
                        );

                    if (
                        earlier >
                        later +
                        tolerance
                    ) {
                        issues.push({
                            period:
                                `${names[index]}→${names[index + 1]}`,

                            metric:
                                metric.name,

                            platform:
                                platform.page,

                            reason:
                                `周期累计值倒挂：${earlier} > ${later}`
                        });
                    }
                }
            }
        }

        return issues;
    }

    async function captureAlbum(album) {
        const issues = [];
        let session;

        try {
            session = await prepareAlbumApiSession(album);
        } catch (error) {
            for (const period of CFG.periods) {
                markPeriodFailure(album, period, error.message);
            }

            saveRecords();
            throw new Error(`无法建立fanChart接口会话：${error.message}`);
        }

        try {
            setStatus(`读取页面专辑总计：${album}`);
            const totalIssues = capturePageAlbumTotals(album);
            issues.push(...totalIssues);
            saveRecords();
        } catch (error) {
            const reason = `读取页面专辑总计失败：${error.message}`;

            for (const metric of CFG.metrics) {
                upsert([
                    {
                        album,
                        period: '累计',
                        platform: CFG.totalPlatform.page,
                        metric: metric.name,
                        unit: metric.unit,
                        value: '',
                        percent: 100,
                        status: '未识别',
                        note: reason
                    }
                ], false);

                issues.push({
                    period: '累计',
                    metric: metric.name,
                    platform: CFG.totalPlatform.page,
                    reason
                });
            }

            saveRecords();
        }

        for (const period of CFG.periods) {
            if (stopFlag || !task.running) {
                throw new Error('任务已停止。');
            }

            for (const metric of CFG.metrics) {
                try {
                    setStatus(
                        `接口直读：${album} / ${period.name} / ${metric.name}`
                    );

                    const metricIssueList = await captureMetricFromApi(
                        album,
                        period,
                        metric,
                        session
                    );
                    issues.push(...metricIssueList);
                } catch (error) {
                    if (stopFlag || !task.running) {
                        throw error;
                    }

                    markMetricFailure(
                        album,
                        period,
                        metric,
                        error.message
                    );
                    issues.push({
                        period: period.name,
                        metric: metric.name,
                        platform: '全部平台',
                        reason: error.message
                    });

                    console.error(
                        '[TME] 单个fanChart接口抓取失败，仅记录本指标异常，继续后续指标',
                        album,
                        period.name,
                        metric.name,
                        error
                    );
                }
            }

            /* 每个周期保存一次，兼顾断点恢复和写入速度。 */
            saveRecords();
        }

        const orderIssues = validatePeriodOrder(album);
        issues.push(...orderIssues);

        if (orderIssues.length) {
            console.warn(
                '[TME] 发现周期累计值倒挂，将只在这种关键异常时重试整张专辑',
                album,
                orderIssues
            );
        }

        const unique = new Map();

        for (const issue of issues) {
            const key = [
                issue.period,
                issue.metric,
                issue.platform,
                issue.reason
            ].join('||');
            unique.set(key, issue);
        }

        saveRecords();
        return Array.from(unique.values());
    }

    function requiredDataIssues(album) {
        const map = new Map(
            records.map(item => [recordKey(item), item])
        );
        const issues = [];

        /*
         * “真实数据异常”只看最终落盘的数据是否完整。
         * 四个周期 × 三个指标 × 四个平台 = 48条接口数据；
         * 再加页面专辑总计3条，共51条。
         */
        for (const period of CFG.periods) {
            for (const metric of CFG.metrics) {
                for (const platform of CFG.platforms) {
                    const item = map.get([
                        album,
                        period.name,
                        metric.name,
                        platform.page
                    ].join('||'));

                    if (
                        !item ||
                        item.status !== '成功' ||
                        item.value === '' ||
                        item.value === null ||
                        item.percent === '' ||
                        item.percent === null
                    ) {
                        issues.push({
                            period: period.name,
                            metric: metric.name,
                            platform: platform.page,
                            reason: !item
                                ? '缺少最终记录'
                                : clean(item.note) || '最终记录未成功'
                        });
                    }
                }
            }
        }

        for (const metric of CFG.metrics) {
            const item = map.get([
                album,
                '累计',
                metric.name,
                CFG.totalPlatform.page
            ].join('||'));

            if (
                !item ||
                item.status !== '成功' ||
                item.value === '' ||
                item.value === null
            ) {
                issues.push({
                    period: '累计',
                    metric: metric.name,
                    platform: CFG.totalPlatform.page,
                    reason: !item
                        ? '缺少页面专辑总计记录'
                        : clean(item.note) || '页面专辑总计未成功'
                });
            }
        }

        return issues;
    }

    function albumComplete(album) {
        /*
         * 9.5.5：完整性与“周期累计值倒挂”彻底分离。
         * 周期关系校验只是提醒，不再让已完整抓取的专辑被判成异常，
         * 也不会导致“跳过已完整专辑”失效、反复重抓。
         */
        return requiredDataIssues(album).length === 0;
    }

    function setAlbumWarning(album, warningIssues) {
        task.warnings = (task.warnings || []).filter(item => item.album !== album);

        if (!warningIssues.length) {
            return;
        }

        task.warnings.push({
            album,
            warning:
                `发现 ${warningIssues.length} 个周期关系校验提醒；` +
                warningIssues
                    .slice(0, 3)
                    .map(item => `${item.period}/${item.metric}/${item.platform}：${item.reason}`)
                    .join('；')
        });
    }

    function reconcileTaskDiagnostics() {
        /*
         * 从9.5.4升级时，旧task.failed里可能混入“数据完整但有周期校验提醒”的专辑。
         * 这里按当前最终51条记录重新核算，自动清掉历史误报。
         */
        task.failed = (task.failed || []).filter(item => {
            return item?.album && requiredDataIssues(item.album).length > 0;
        });

        const albumNames = Array.from(new Set(
            records.map(item => item.album).filter(Boolean)
        ));

        task.warnings = [];
        for (const album of albumNames) {
            if (requiredDataIssues(album).length) {
                continue;
            }

            const warnings = validatePeriodOrder(album);
            if (warnings.length) {
                setAlbumWarning(album, warnings);
            }
        }
    }

    async function runTask() {
        if (processing || !task.running) {
            return;
        }

        processing = true;
        stopFlag = false;
        setBusy(true);

        let consecutiveFails = 0;

        try {
            while (
                task.running &&
                !stopFlag &&
                task.index < task.albums.length
            ) {
                const album = task.albums[task.index];
                task.current = album;
                saveTask();

                if (albumAlreadyUnavailable(album)) {
                    task.index += 1;
                    task.error = '';
                    saveTask();
                    continue;
                }

                if (task.skipDone && albumComplete(album)) {
                    if (!task.done.includes(album)) {
                        task.done.push(album);
                    }

                    task.index += 1;
                    saveTask();
                    continue;
                }

                let success = false;
                let lastError = null;
                let unavailableReason = '';

                for (let attempt = 1; attempt <= 2; attempt += 1) {
                    try {
                        setStatus(
                            `进度 ${task.index + 1}/${task.albums.length}：` +
                            `${album}，第 ${attempt} 次尝试`
                        );

                        if (currentAlbum() !== album) {
                            await selectAlbum(album);
                            await waitAlbum(album);
                        }

                        const issueList = await captureAlbum(album);
                        const validationWarnings = issueList.filter(item => {
                            return /周期累计值倒挂/.test(item.reason);
                        });
                        const criticalIssues = issueList.filter(item => {
                            return (
                                item.metric === '全部指标' ||
                                /周期切换后页面一直未稳定/.test(item.reason)
                            );
                        });

                        /*
                         * 9.5.5：周期累计值倒挂不再视为抓取失败。
                         * 数据来自后台且最终记录完整时，它只作为“校验提醒”保存。
                         * 只有真正的页面/接口切换关键错误才允许整本重试。
                         */
                        if (criticalIssues.length && attempt < 2) {
                            const preview = criticalIssues
                                .slice(0, 3)
                                .map(item => {
                                    return (
                                        `${item.period}/${item.metric}/${item.platform}：` +
                                        item.reason
                                    );
                                })
                                .join('；');

                            throw new Error(
                                `发现 ${criticalIssues.length} 个关键异常，准备重试。${preview}`
                            );
                        }

                        success = true;

                        const finalDataIssues = requiredDataIssues(album);
                        setAlbumWarning(album, validationWarnings);

                        if (finalDataIssues.length) {
                            lastError = new Error(
                                `最终数据仍有 ${finalDataIssues.length} 项未完整；` +
                                finalDataIssues
                                    .slice(0, 3)
                                    .map(item => `${item.period}/${item.metric}/${item.platform}`)
                                    .join('、')
                            );
                        } else {
                            /* 最终51条数据完整，绝不再因为校验提醒计入“异常”。 */
                            lastError = null;
                        }

                        break;
                    } catch (error) {
                        lastError = error;

                        if (stopFlag || !task.running) {
                            throw error;
                        }

                        if (isAlbumUnavailableError(error)) {
                            unavailableReason =
                                error.reason ||
                                error.message ||
                                '搜索结果不可点击，疑似已下架或不可进入';

                            console.warn(
                                '[TME] 专辑已下架/不可进入，保持搜索框展开并直接搜索下一本',
                                album,
                                unavailableReason
                            );

                            /*
                             * 9.5.5核心：这里绝不调用 closeDropdown。
                             * 当前搜索框保留，下一轮 selectAlbum 会直接复用该输入框搜索下一本。
                             */
                            controlCache.clear();
                            break;
                        }

                        console.error('[TME] 抓取失败', album, error);
                        await closeDropdown();
                        clearPreservedAlbumSearchContext();
                        controlCache.clear();
                        await sleep(650);
                    }
                }

                task.failed = task.failed.filter(item => {
                    return item.album !== album;
                });

                if (unavailableReason || !success) {
                    task.warnings = (task.warnings || []).filter(item => item.album !== album);
                }

                if (unavailableReason) {
                    consecutiveFails = 0;
                    markAlbumUnavailable(
                        album,
                        unavailableReason
                    );
                    lastError = null;

                    setStatus(
                        `进度 ${task.index + 1}/${task.albums.length}：` +
                        `${album} 已下架/不可进入，已跳过，不计异常。`
                    );
                } else if (success) {
                    consecutiveFails = 0;

                    if (!task.done.includes(album)) {
                        task.done.push(album);
                    }

                    if (lastError) {
                        task.failed.push({
                            album,
                            error: lastError.message
                        });
                    }
                } else {
                    consecutiveFails += 1;
                    task.failed.push({
                        album,
                        error: lastError?.message || '未知错误'
                    });
                }

                task.index += 1;
                task.error = lastError?.message || '';
                saveRecords();
                saveTask();

                if (consecutiveFails >= 3) {
                    task.running = false;
                    task.error = '连续三个专辑失败，任务已暂停。';
                    saveTask();
                    throw new Error(task.error);
                }
            }

            if (task.index >= task.albums.length) {
                task.running = false;
                task.current = '';
                saveTask();

                setStatus(
                    `全部完成：${task.done.length} 个专辑，` +
                    `下架/不可进入 ${task.unavailable.length} 个，` +
                    `数据异常 ${task.failed.length} 个，` +
                    `校验提醒 ${(task.warnings || []).length} 个。`
                );

                alert(
                    `全部专辑抓取完成\n\n` +
                    `专辑总数：${task.albums.length}\n` +
                    `完成专辑：${task.done.length}\n` +
                    `下架/不可进入：${task.unavailable.length}\n` +
                    `数据异常专辑：${task.failed.length}\n` +
                    `周期校验提醒：${(task.warnings || []).length}\n` +
                    `记录数：${records.length}`
                );

                if (task.autoExport) {
                    await exportExcel(true);
                }
            }
        } catch (error) {
            task.running = false;
            task.error = error.message;
            saveRecords();
            saveTask();

            if (stopFlag) {
                setStatus(`任务已暂停：${task.index}/${task.albums.length}`);
            } else {
                setStatus(`任务暂停：${error.message}`);
                alert(
                    `批量任务已暂停：${error.message}\n` +
                    '点击“继续任务”可重试。'
                );
            }
        } finally {
            processing = false;
            setBusy(false);
            refreshCurrent();
        }
    }

    function hslToArgb(
        hue,
        saturation,
        lightness
    ) {
        const s =
            saturation / 100;

        const l =
            lightness / 100;

        const c =
            (1 -
                Math.abs(
                    2 * l - 1
                )) * s;

        const h =
            (
                (
                    hue % 360
                ) + 360
            ) % 360 / 60;

        const x =
            c *
            (
                1 -
                Math.abs(
                    h % 2 - 1
                )
            );

        let r = 0;
        let g = 0;
        let b = 0;

        if (h < 1) {
            r = c;
            g = x;
        } else if (h < 2) {
            r = x;
            g = c;
        } else if (h < 3) {
            g = c;
            b = x;
        } else if (h < 4) {
            g = x;
            b = c;
        } else if (h < 5) {
            r = x;
            b = c;
        } else {
            r = c;
            b = x;
        }

        const m =
            l - c / 2;

        const hex = value => {
            return Math.round(
                (value + m) * 255
            )
                .toString(16)
                .padStart(2, '0')
                .toUpperCase();
        };

        return (
            hex(r) +
            hex(g) +
            hex(b)
        );
    }

    function albumFill(albumIndex) {
        /*
         * 使用黄金角生成浅色底色。
         * 相邻专辑颜色明显不同，专辑较多时也不会只在两三种颜色间重复。
         */
        const hue =
            albumIndex *
            137.508;

        return hslToArgb(
            hue,
            52,
            88
        );
    }

    function styleRow(
        row,
        period,
        platformIndex,
        metric,
        missing,
        albumColor,
        isTotal = false
    ) {
        row.height = 24;

        row.eachCell(
            {
                includeEmpty: true
            },
            (cell, column) => {
                let fill =
                    platformIndex % 2 === 0
                        ? C.row1
                        : C.row2;

                if (column === 1) {
                    fill =
                        albumColor;
                } else if (
                    column === 2
                ) {
                    fill =
                        periodFill(
                            period
                        );
                } else if (
                    isTotal &&
                    column >= 3
                ) {
                    fill =
                        C.total;
                } else if (
                    missing &&
                    column >= 5
                ) {
                    fill =
                        C.bad;
                }

                cell.font = {
                    name:
                        'Microsoft YaHei',

                    size: 10,

                    bold:
                        isTotal,

                    color: {
                        argb:
                            missing &&
                            column >= 5
                                ? C.red
                                : C.text
                    }
                };

                cell.fill = {
                    type:
                        'pattern',

                    pattern:
                        'solid',

                    fgColor: {
                        argb:
                            fill
                    }
                };

                cell.alignment = {
                    horizontal:
                        'center',

                    vertical:
                        'middle',

                    wrapText:
                        true
                };

                cell.border =
                    thinBorder();
            }
        );

        if (!missing) {
            row.getCell(5)
                .numFmt =
                metric ===
                '播放时长'
                    ? '#,##0.00'
                    : '#,##0';

            row.getCell(6)
                .numFmt =
                '0.00%';
        }
    }

    function exportAlbums() {
        const list = [];

        const seen =
            new Set();
        const unavailableNames = new Set(
            (task.unavailable || []).map(item => {
                return compact(item?.album);
            })
        );

        for (
            const name of
            task.albums || []
        ) {
            if (
                name &&
                !unavailableNames.has(compact(name)) &&
                !seen.has(name)
            ) {
                seen.add(name);

                list.push(name);
            }
        }

        for (const item of records) {
            if (
                item.album &&
                !seen.has(
                    item.album
                )
            ) {
                seen.add(
                    item.album
                );

                list.push(
                    item.album
                );
            }
        }

        return list;
    }

    function fileTime(date) {
        const pad = value =>
            String(value)
                .padStart(
                    2,
                    '0'
                );

        return (
            `${date.getFullYear()}` +
            `${pad(
                date.getMonth() + 1
            )}` +
            `${pad(
                date.getDate()
            )}_` +
            `${pad(
                date.getHours()
            )}` +
            `${pad(
                date.getMinutes()
            )}` +
            `${pad(
                date.getSeconds()
            )}`
        );
    }

    async function exportExcel(
        fromTask = false
    ) {
        if (
            !records.length &&
            !task.albums.length
        ) {
            alert(
                '没有可导出的数据。'
            );

            return;
        }

        const Excel =
            typeof ExcelJS !==
            'undefined'
                ? ExcelJS
                : window.ExcelJS;

        if (!Excel) {
            alert(
                'Excel组件没有加载成功，请检查网络后刷新页面。'
            );

            return;
        }

        if (!fromTask) {
            setBusy(true);
        }

        setStatus(
            '正在生成七个工作表，请稍候……'
        );

        try {
            const map =
                new Map(
                    records
                        .map(
                            normalizeRecord
                        )
                        .map(item => {
                            return [
                                recordKey(item),
                                item
                            ];
                        })
                );

            const albums =
                exportAlbums();

            const workbook =
                new Excel.Workbook();

            workbook.creator =
                'TME声播全部专辑抓取工具';

            buildSheet(
                workbook,
                '四平台播放数据',
                albums,
                map
            );

            buildSheet(
                workbook,
                '昨天',
                albums,
                map,
                '昨天'
            );

            buildSheet(
                workbook,
                '近七天',
                albums,
                map,
                '近七天'
            );

            buildSheet(
                workbook,
                '近30天',
                albums,
                map,
                '近30天'
            );

            buildSheet(
                workbook,
                '累计',
                albums,
                map,
                '累计'
            );

            buildSheet(
                workbook,
                '酷狗QQ近30天',
                albums,
                map,
                '近30天',
                [
                    '酷狗音乐',
                    'QQ音乐'
                ]
            );

            buildUnavailableSheet(
                workbook
            );

            const buffer =
                await workbook.xlsx
                    .writeBuffer();

            const blob =
                new Blob(
                    [buffer],
                    {
                        type:
                            'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
                    }
                );

            const url =
                URL.createObjectURL(
                    blob
                );

            const link =
                document.createElement(
                    'a'
                );

            link.href =
                url;

            link.download =
                `声播_TME全部专辑_四平台播放数据_含下架清单_${fileTime(
                    new Date()
                )}.xlsx`;

            document.body.appendChild(
                link
            );

            link.click();

            link.remove();

            setTimeout(
                () => {
                    URL.revokeObjectURL(
                        url
                    );
                },
                1500
            );

            setStatus(
                `导出完成：${albums.length} 个可抓取专辑，7个工作表；` +
                `另记录下架/不可进入 ${task.unavailable.length} 个。`
            );
        } catch (error) {
            console.error(error);

            setStatus(
                `导出失败：${error.message}`
            );

            alert(
                `导出失败：${error.message}`
            );
        } finally {
            if (!fromTask) {
                setBusy(false);
            }
        }
    }

    function updateProgress() {
        const label =
            document.getElementById(
                'tme-progress'
            );

        const bar =
            document.getElementById(
                'tme-bar'
            );

        if (!label || !bar) {
            return;
        }

        const total =
            task.albums.length;

        const done =
            Math.min(
                task.index,
                total
            );

        label.textContent =
            total
                ? `批量进度：${done}/${total}，` +
                  `下架/不可进入 ${(task.unavailable || []).length} 个，` +
                  `数据异常 ${task.failed.length} 个，` +
                  `校验提醒 ${(task.warnings || []).length} 个`
                : '批量进度：尚未建立任务';

        bar.style.width =
            `${
                total
                    ? Math.round(
                        done /
                        total *
                        100
                    )
                    : 0
            }%`;

        updateCount();
    }

            body.style.display =
                hidden
                    ? 'block'
                    : 'none';

            event.currentTarget
                .textContent =
                hidden
                    ? '−'
                    : '+';
        };

        document.getElementById(
            'tme-skip'
        ).checked =
            task.skipDone;

        document.getElementById(
            'tme-auto'
        ).checked =
            task.autoExport;

        refreshCurrent();

        updateProgress();
    }

    let initStarted = false;

    async function init() {
        if (!document.body) {
            setTimeout(
                init,
                500
            );

            return;
        }

        if (initStarted) {
            return;
        }

        initStarted = true;

        try {
            await hydrateStorage();
        } catch (error) {
            console.error('[TME] IndexedDB初始化失败', error);
            initStarted = false;

            alert(
                `大容量本地存储初始化失败：${error.message}\n\n` +
                '请确认浏览器没有禁用站点存储，并关闭其他同站点页面后刷新。'
            );
            return;
        }

        createPanel();

        reconcileTaskDiagnostics();
        saveTask();

        new MutationObserver(() => {
            /* 抓取过程中页面变化频繁，不再额外执行全页专辑识别。 */
            if (processing || task.running) {
                return;
            }

            clearTimeout(observerTimer);
            observerTimer = setTimeout(refreshCurrent, 1200);
        }).observe(document.body, {
            childList: true,
            subtree: true
        });

        setStatus(
            '9.6.2已加载。抓取记录和任务进度已迁移到IndexedDB大容量存储，不再受localStorage约5MB配额限制；旧数据会自动迁移并释放旧配额。其余扫描、下架、累计/总计逻辑保持不变。'
        );

        if (
            task.running &&
            task.index <
            task.albums.length
        ) {
            setTimeout(
                runTask,
                2500
            );
        }
    }

    init();
})();
